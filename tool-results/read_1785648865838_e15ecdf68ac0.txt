    49→
    50→function formatINRfinance(paise: number): string {
    51→  const rupees = paise / 100;
    52→  if (rupees >= 10000000) return `₹${(rupees / 10000000).toFixed(2)} Cr`;
    53→  if (rupees >= 100000) return `₹${(rupees / 100000).toFixed(2)} L`;
    54→  if (rupees >= 1000) return `₹${(rupees / 1000).toFixed(1)}K`;
    55→  return `₹${rupees.toLocaleString("en-IN")}`;
    56→}
    57→
    58→function formatTime(iso: string): string {
    59→  return new Date(iso).toLocaleTimeString("en-IN", {
    60→    hour: "2-digit",
    61→    minute: "2-digit",
    62→    hour12: true,
    63→  });
    64→}
    65→
    66→function greeting(): string {
    67→  const h = new Date().getHours();
    68→  if (h < 12) return "Good morning";
    69→  if (h < 17) return "Good afternoon";
    70→  return "Good evening";
    71→}
