     1→/**
     2→ * BIHARI AI — Task Executor
     3→ *
     4→ * Processes a single step of a task. The worker calls this function once per
     5→ * poll tick. This is the heart of the trust loop:
     6→ *
     7→ * 1. Find the next pending step
     8→ * 2. Execute it (or create an approval gate)
     9→ * 3. Write an audit entry atomically
    10→ * 4. If approval needed: set task to waiting_approval and STOP
    11→ * 5. If no more steps: set task to completed
    12→ *
    13→ * Per BED-001 §7: "The executor calls the LLM Gateway to produce the step's
    14→ * reasoning and proposed action. The executor persists a task_steps row with
    15→ * input, reasoning, and output."
    16→ *
    17→ * Per BED-001 §7: "If critical: creates an approvals row (status pending) with
    18→ * the proposed action, appends audit entry approval_requested, sends a
    19→ * notification, transitions the task to waiting_approval, and returns without
    20→ * executing."
    21→ */

    55→// ─── Main Entry Point ────────────────────────────────────────────────────────
    56→
    57→/**
    58→ * Processes one unit of work for a task.
    59→ *
    60→ * This function is called by the worker poll loop. It does ONE of:
    61→ * - Plans the task (if status is "queued")
    62→ * - Executes the next pending step (if status is "executing")
    63→ * - Resolves an approval gate (if status is "executing" and next step is
    64→ *   approval_gate with a decided approval)
    65→ * - Completes the task (if no pending steps remain)
    66→ *
    67→ * All state changes and audit entries are written in a single transaction.
    68→ */
    69→export async function processTask(taskId: string): Promise<ExecutionResult> {
    70→  const task = await db.task.findUnique({
    71→    where: { id: taskId },
    72→    include: {
    73→      employee: true,
    74→      steps: { orderBy: { stepNumber: "asc" } },
    75→    },
    76→  });
    77→
    78→  if (!task) {
    79→    return { action: "failed", message: "Task not found" };
    80→  }
    81→
    82→  // Don't process tasks that aren't in an active execution state
    83→  if (!["queued", "planning", "executing"].includes(task.status)) {
    84→    return { action: "continue", message: `Task is ${task.status}, skipping` };
    85→  }
    86→
    87→  // Check if the employee is paused
    88→  if (task.employee.status === "paused") {
    89→    // Pause the task
    90→    await db.$transaction(async (tx) => {
    91→      await tx.task.update({
    92→        where: { id: taskId },
    93→        data: { status: "paused" },
    94→      });
    95→      await appendAudit(tx, {
    96→        workspaceId: task.workspaceId,
    97→        entryType: "task_paused",
    98→        actorType: "system",
    99→        actorId: null,
   100→        actorName: "Runtime",
   101→        targetType: "task",
   102→        targetId: taskId,
   103→        payload: { reason: "Employee paused", employee: task.employee.name },
   104→      });
   105→    });
   106→    return { action: "continue", message: "Employee paused, task paused" };
   107→  }
   108→
   109→  // Phase 1: Planning — generate the step plan
   110→  if (task.status === "queued") {
   111→    return await planTask(task);
   112→  }
   113→
   114→  // Phase 2: Execution — process the next pending step
   115→  const pendingSteps = task.steps.filter((s) => s.status === "pending");
   116→
   117→  if (pendingSteps.length === 0) {
   118→    // No pending steps — task is complete
   119→    return await completeTask(task);
   120→  }
   121→
   122→  const nextStep = pendingSteps[0];
   123→  return await executeStep(task, nextStep, task.steps);
   124→}

   126→// ─── Planning Phase ──────────────────────────────────────────────────────────
   127→
   128→async function planTask(
   129→  task: NonNullable<Awaited<ReturnType<typeof db.task.findUnique>>> & {}
   130→): Promise<ExecutionResult> {
   131→  const employee = task.employee;
   132→  const employeeTools: string[] = JSON.parse(employee.tools);
   133→
   134→  // Transition to planning
   135→  await db.$transaction(async (tx) => {
   136→    await tx.task.update({
   137→      where: { id: task.id },
   138→      data: { status: "planning", startedAt: task.startedAt ?? new Date() },
   139→    });
   140→    await appendAudit(tx, {
   141→      workspaceId: task.workspaceId,
   142→      entryType: "task_started",
   143→      actorType: "user",
   144→      actorId: task.assignedBy,
   145→      actorName: "Runtime",
   146→      targetType: "task",
   147→      targetId: task.id,
   148→      payload: {
   149→        title: task.title,
   150→        employee: employee.name,
   151→        role: employee.role,
   152→      },
   153→    });
   154→  });
   155→
   156→  // Generate the plan
   157→  let plan: { steps: PlannedStep[] };
   158→
   159→  // Check if this is a finance task
   160→  if (isFinanceTask(employee.role, task.title, task.description)) {
   161→    // Finance planning: use the Finance Brain to build context + produce recommendation
   162→    const invoiceIdMatch = task.description.match(/invoice[_\s-]?id[:\s]+([a-zA-Z0-9]+)/i);
   163→    const invoiceNumMatch = task.description.match(/invoice[_\s-]?number[:\s]+([A-Z0-9-]+)/i);
   164→
   165→    if (invoiceIdMatch || invoiceNumMatch) {
   166→      // Single invoice processing — Finance Brain builds full context + recommendation
   167→      let invoiceId = invoiceIdMatch?.[1];
   168→
   169→      // If we have an invoice number but no ID, look it up
   170→      if (!invoiceId && invoiceNumMatch) {
   171→        const inv = await db.invoice.findFirst({
   172→          where: { invoiceNumber: invoiceNumMatch[1] },
   173→        });
   174→        invoiceId = inv?.id;
   175→      }
   176→
   177→      if (invoiceId) {
   178→        // Step 1: Build the complete FinanceContext (invoice, customer, payments, reminders, policies, memory, etc.)
   179→        const financeCtx = await buildFinanceContext(invoiceId, employee.id);
   180→        if (financeCtx) {
   181→          // Step 2: The Finance Brain reasons over the context (including memory) to produce a recommendation
   182→          const recommendation = produceRecommendation(financeCtx);
   183→          // Step 3: Generate the execution plan from the recommendation
   184→          plan = generateFinancePlan(recommendation, employeeTools);
   185→        } else {
   186→          plan = { steps: [] };
   187→        }
   188→      } else {
   189→        plan = { steps: [] };
   190→      }
   191→    } else {
   192→      // Batch processing: find invoices needing attention, then build full context + recommendation for each
   193→      const invoiceContexts = await findInvoicesNeedingAttention(task.workspaceId, 3); // Limit to 3 per task
   194→      if (invoiceContexts.length > 0) {
   195→        // For each invoice, build the full FinanceContext and produce a recommendation
   196→        const recommendations: FinanceRecommendation[] = [];
   197→        for (const ic of invoiceContexts) {
   198→          const financeCtx = await buildFinanceContext(ic.invoiceId, employee.id);
   199→          if (financeCtx) {
   200→            const rec = produceRecommendation(financeCtx);
   201→            recommendations.push(rec);
   202→          }
   203→        }
   204→        if (recommendations.length > 0) {
   205→          plan = generateBatchFinancePlan(recommendations, employeeTools);
   206→        } else {
   207→          plan = { steps: [] };
   208→        }
   209→      } else {
   210→        plan = { steps: [] };
   211→      }
   212→    }
   213→  } else {
   214→    // Generic planning
   215→    // Check if the LLM Gateway has a real provider available (not mock)
   216→    const gateway = getLLMGateway();
   217→    const router = (await import("@/lib/llm")).getModelRouter();
   218→    const { provider, route } = router.route("planning");
   219→
   220→    if (provider.name !== "mock") {
   221→      // ─── LLM-Gateway-Aware Planning ──────────────────────────────────────
   222→      // When a real LLM provider is configured, use the gateway to generate
   223→      // the execution plan. This replaces the deterministic mock planner
   224→      // with real AI planning.
   225→      try {
   226→        const response = await gateway.complete({
   227→          taskType: "planning",
   228→          promptId: "planning",
   229→          variables: {
   230→            title: task.title,
   231→            description: task.description,
   232→            role: employee.role,
   233→            tools: employeeTools.join(", "),
   234→          },
   235→          jsonMode: true,
   236→          jsonSchema: {
   237→            type: "object",
   238→            required: ["reasoning", "steps"],
   239→            properties: {
   240→              reasoning: { type: "string" },
   241→              steps: {
   242→                type: "array",
   243→                items: {
   244→                  type: "object",
   245→                  required: ["stepType", "reasoning", "confidence"],
   246→                  properties: {
   247→                    stepType: { type: "string" },
   248→                    reasoning: { type: "string" },
   249→                    tool: { type: "string" },
   250→                    confidence: { type: "number" },
   251→                  },
   252→                },
   253→              },
   254→            },
   255→          },
   256→          workspaceId: task.workspaceId,
   257→          employeeId: employee.id,
   258→          taskId: task.id,
   259→        });
   260→
   261→        // Parse the LLM-generated plan
   262→        const llmPlan = response.data as { reasoning: string; steps: PlannedStep[] } | null;
   263→        if (llmPlan && llmPlan.steps && Array.isArray(llmPlan.steps)) {
   264→          plan = { steps: llmPlan.steps };
   265→        } else {
   266→          // Fallback to deterministic planner if LLM response is invalid
   267→          plan = generatePlan(task.title, task.description, employee.role, employeeTools);
   268→        }
   269→      } catch (err) {
   270→        // On any gateway error, fall back to the deterministic planner
   271→        console.error("[Executor] LLM Gateway planning failed, falling back to deterministic planner:", err);
   272→        plan = generatePlan(task.title, task.description, employee.role, employeeTools);
   273→      }
   274→    } else {
   275→      // ─── Deterministic Planning (existing behavior) ─────────────────────
   276→      // No real LLM provider configured — use the existing mock planner.
   277→      // This maintains backward compatibility.
   278→      plan = generatePlan(task.title, task.description, employee.role, employeeTools);
   279→    }
   280→  }
   281→
   282→  if (plan.steps.length === 0) {
   283→    // No steps to execute — fail immediately
   284→    await db.$transaction(async (tx) => {
   285→      await tx.task.update({
   286→        where: { id: task.id },
   287→        data: { status: "failed" },
   288→      });
   289→      await appendAudit(tx, {
   290→        workspaceId: task.workspaceId,
   291→        entryType: "task_failed",
   292→        actorType: "employee",
   293→        actorId: employee.id,
   294→        actorName: employee.name,
   295→        targetType: "task",
   296→        targetId: task.id,
   297→        payload: { reason: "Plan generation produced no steps" },
   298→      });
   299→    });
   300→    return { action: "failed", message: "Plan generation produced no steps" };
   301→  }
   302→
   303→  // Check step cap
   304→  if (plan.steps.length > task.stepCap) {
   305→    plan.steps = plan.steps.slice(0, task.stepCap);
   306→  }
   307→
   308→  // Write the plan step (step 0 — the planning reasoning)
   309→  const approvalRules: Record<string, string> = JSON.parse(employee.approvalRules);
   310→
   311→  await db.$transaction(async (tx) => {
   312→    // Step 0: The plan itself
   313→    const isFinance = isFinanceTask(employee.role, task.title, task.description);
   314→    const criticalCount = plan.steps.filter((s) => s.tool === "send_email" || s.tool === "send_reminder").length;
   315→    await tx.taskStep.create({
   316→      data: {
   317→        taskId: task.id,
   318→        stepNumber: 0,
   319→        stepType: "plan",
   320→        reasoning: isFinance
   321→          ? `Planned ${plan.steps.length} steps for this finance collections task. The plan includes ${criticalCount} critical action(s) requiring human approval.`
   322→          : `Planned ${plan.steps.length} steps for this task. The plan includes ${criticalCount} email action(s) that will require human approval.`,
   323→        status: "completed",
   324→        tokens: 420,
   325→        durationMs: 3100,
   326→        startedAt: new Date(),
   327→        completedAt: new Date(),
   328→      },
   329→    });
   330→
   331→    // Write each planned step as pending
   332→    for (let i = 0; i < plan.steps.length; i++) {
   333→      const step = plan.steps[i];
   334→      const isCritical = step.tool ? approvalRules[step.tool] === "critical" : false;
   335→
   336→      await tx.taskStep.create({
   337→        data: {
   338→          taskId: task.id,
   339→          stepNumber: i + 1,
   340→          stepType: isCritical ? "approval_gate" : step.stepType,
   341→          input: JSON.stringify({
   342→            tool: step.tool || null,
   343→            ...step.toolInput,
   344→          }),
   345→          reasoning: step.reasoning,
   346→          status: "pending",
   347→          confidence: step.confidence,
   348→        },
   349→      });
   350→    }
   351→
   352→    // Transition to executing
   353→    await tx.task.update({
   354→      where: { id: task.id },
   355→      data: { status: "executing", stepCount: plan.steps.length + 1 },
   356→    });
   357→
   358→    await appendAudit(tx, {
   359→      workspaceId: task.workspaceId,
   360→      entryType: "plan_created",
   361→      actorType: "employee",
   362→      actorId: employee.id,
   363→      actorName: employee.name,
   364→      targetType: "task",
   365→      targetId: task.id,
   366→      payload: {
   367→        step_count: String(plan.steps.length),
   368→        critical_steps: String(plan.steps.filter((s) => s.tool && approvalRules[s.tool] === "critical").length),
   369→      },
   370→    });
   371→  });
   372→
   373→  return { action: "continue", message: `Planned ${plan.steps.length} steps, now executing` };
   374→}

   376→// ─── Step Execution ──────────────────────────────────────────────────────────
   377→
   378→async function executeStep(
   379→  task: any,
   380→  step: any,
   381→  allSteps: any[]
   382→): Promise<ExecutionResult> {
   383→  const employee = task.employee;
   384→
   385→  // Mark step as running
   386→  await db.taskStep.update({
   387→    where: { id: step.id },
   388→    data: { status: "running", startedAt: new Date() },
   389→  });
   390→
   391→  try {
   392→    switch (step.stepType) {
   393→      case "reasoning":
   394→        return await executeReasoningStep(task, step, employee);
   395→
   396→      case "tool_call":
   397→        return await executeToolStep(task, step, employee);
   398→
   399→      case "approval_gate":
   400→        return await executeApprovalGateStep(task, step, employee);
   401→
   402→      default:
   403→        return await executeReasoningStep(task, step, employee);
   404→    }
   405→  } catch (err) {
   406→    // Mark step as failed and fail the task
   407→    await db.$transaction(async (tx) => {
   408→      await tx.taskStep.update({
   409→        where: { id: step.id },
   410→        data: { status: "failed", completedAt: new Date() },
   411→      });
   412→      await tx.task.update({
   413→        where: { id: task.id },
   414→        data: { status: "failed" },
   415→      });
   416→      await appendAudit(tx, {
   417→        workspaceId: task.workspaceId,
   418→        entryType: "task_failed",
   419→        actorType: "employee",
   420→        actorId: employee.id,
   421→        actorName: employee.name,
   422→        targetType: "task",
   423→        targetId: task.id,
   424→        payload: {
   425→          reason: `Step ${step.stepNumber} failed: ${err instanceof Error ? err.message : "unknown error"}`,
   426→        },
   427→      });
   428→    });
   429→    return { action: "failed", message: `Step ${step.stepNumber} failed` };
   430→  }
   431→}

   433→// ─── Reasoning Step ──────────────────────────────────────────────────────────
   434→
   435→async function executeReasoningStep(task: any, step: any, employee: any): Promise<ExecutionResult> {
   436→  const tokens = 500 + Math.floor(Math.random() * 400);
   437→  const durationMs = 1500 + Math.floor(Math.random() * 1500);
   438→
   439→  await db.$transaction(async (tx) => {
   440→    await tx.taskStep.update({
   441→      where: { id: step.id },
   442→      data: {
   443→        status: "completed",
   444→        output: JSON.stringify({ type: "reasoning" }),
   445→        tokens,
   446→        durationMs,
   447→        completedAt: new Date(),
   448→      },
   449→    });
   450→
   451→    await tx.task.update({
   452→      where: { id: task.id },
   453→      data: { tokenUsage: { increment: tokens } },
   454→    });
   455→
   456→    await appendAudit(tx, {
   457→      workspaceId: task.workspaceId,
   458→      entryType: "step_executed",
   459→      actorType: "employee",
   460→      actorId: employee.id,
   461→      actorName: employee.name,
   462→      targetType: "task_step",
   463→      targetId: step.id,
   464→      payload: {
   465→        step: String(step.stepNumber),
   466→        type: "reasoning",
   467→        tokens: String(tokens),
   468→      },
   469→    });
   470→  });
   471→
   472→  return { action: "continue", message: `Reasoning step ${step.stepNumber} completed` };
   473→}

   475→// ─── Tool Call Step (non-critical) ───────────────────────────────────────────
   476→
   477→async function executeToolStep(task: any, step: any, employee: any): Promise<ExecutionResult> {
   478→  const input = JSON.parse(step.input);
   479→  const toolName = input.tool;
   480→
   481→  if (!toolName) {
   482→    // No tool specified — treat as reasoning
   483→    return await executeReasoningStep(task, step, employee);
   484→  }
   485→
   486→  // ─── Capability Verification ──────────────────────────────────────────────
   487→  // Before ANY tool execution, verify the employee has the required capability.
   488→  // If denied: stop execution, write audit event, fail the step.
   489→  const capCheck = await checkCapability(employee.id, toolName);
   490→  if (!capCheck.allowed) {
   491→    await db.$transaction(async (tx) => {
   492→      // Mark step as failed
   493→      await tx.taskStep.update({
   494→        where: { id: step.id },
   495→        data: {
   496→          status: "failed",
   497→          output: JSON.stringify({
   498→            error: "CAPABILITY_DENIED",
   499→            requiredCapability: capCheck.capabilityCode,
   500→            capabilityName: capCheck.capabilityName,
   501→            reason: capCheck.reason,
   502→          }),
   503→          completedAt: new Date(),
   504→        },
   505→      });
   506→
   507→      // Fail the task
   508→      await tx.task.update({
   509→        where: { id: task.id },
   510→        data: { status: "failed" },
   511→      });
   512→
   513→      // Update employee state
   514→      await tx.employee.update({
   515→        where: { id: employee.id },
   516→        data: { state: "idle" },
   517→      });
   518→
   519→      // Record the denial in the audit chain
   520→      await recordCapabilityDenial(tx, task.workspaceId, employee.id, employee.name, toolName, capCheck);
   521→    });
   522→
   523→    // ─── Update Employee Profile (capability denied) ───────────────────────
   524→    try {
   525→      await recordProfileEvent({
   526→        type: "capability_denied",
   527→        employeeId: employee.id,
   528→        workspaceId: task.workspaceId,
   529→        taskId: task.id,
   530→        toolName,
   531→      });
   532→    } catch (err) {
   533→      console.error(`[Executor] Profile update failed for capability denial:`, err);
   534→    }
   535→
   536→    return { action: "failed", message: `Capability denied: ${capCheck.capabilityCode} required for ${toolName}` };
   537→  }
   538→
   539→  // Check if this is a finance tool
   540→  const financeTools = ["generate_reminder", "send_reminder", "update_collection_case"];
   541→  const isFinanceTool = financeTools.includes(toolName);
   542→
   543→  // Execute the tool
   544→  let result: { output: Record<string, string>; tokens: number; durationMs: number };
   545→
   546→  if (isFinanceTool) {
   547→    result = await executeFinanceTool(toolName, input, task.workspaceId, employee.id);
   548→  } else {
   549→    result = executeTool(toolName, input);
   550→  }
   551→
   552→  // Determine the finance-specific audit entry type
   553→  let auditEntryType = "tool_executed";
   554→  if (toolName === "generate_reminder") auditEntryType = "reminder_drafted";
   555→  else if (toolName === "update_collection_case") auditEntryType = "collection_case_updated";
   556→
   557→  await db.$transaction(async (tx) => {
   558→    await tx.taskStep.update({
   559→      where: { id: step.id },
   560→      data: {
   561→        status: "completed",
   562→        output: JSON.stringify(result.output),
   563→        tokens: result.tokens,
   564→        durationMs: result.durationMs,
   565→        completedAt: new Date(),
   566→      },
   567→    });
   568→
   569→    await tx.task.update({
   570→      where: { id: task.id },
   571→      data: { tokenUsage: { increment: result.tokens } },
   572→    });
   573→
   574→    await appendAudit(tx, {
   575→      workspaceId: task.workspaceId,
   576→      entryType: auditEntryType,
   577→      actorType: "employee",
   578→      actorId: employee.id,
   579→      actorName: employee.name,
   580→      targetType: isFinanceTool ? "invoice" : "task_step",
   581→      targetId: input.invoiceId || step.id,
   582→      payload: {
   583→        tool: toolName,
   584→        status: "completed",
   585→        tokens: String(result.tokens),
   586→        ...(input.invoiceNumber ? { invoiceNumber: input.invoiceNumber } : {}),
   587→        ...(input.customerName ? { customer: input.customerName } : {}),
   588→      },
   589→    });
   590→  });
   591→
   592→  return { action: "continue", message: `Tool ${toolName} executed` };
   593→}

   595→// ─── Approval Gate Step (critical tool call) ─────────────────────────────────
   596→
   597→async function executeApprovalGateStep(task: any, step: any, employee: any): Promise<ExecutionResult> {
   598→  const input = JSON.parse(step.input);
   599→  const toolName = input.tool;
   600→
   601→  // ─── Capability Verification (pre-approval) ───────────────────────────────
   602→  // Before creating an approval gate, verify the employee has the capability.
   603→  // If denied: stop execution, write audit event, fail the step.
   604→  const capCheck = await checkCapability(employee.id, toolName);
   605→  if (!capCheck.allowed) {
   606→    await db.$transaction(async (tx) => {
   607→      await tx.taskStep.update({
   608→        where: { id: step.id },
   609→        data: {
   610→          status: "failed",
   611→          output: JSON.stringify({
   612→            error: "CAPABILITY_DENIED",
   613→            requiredCapability: capCheck.capabilityCode,
   614→            capabilityName: capCheck.capabilityName,
   615→            reason: capCheck.reason,
   616→          }),
   617→          completedAt: new Date(),
   618→        },
   619→      });
   620→      await tx.task.update({
   621→        where: { id: task.id },
   622→        data: { status: "failed" },
   623→      });
   624→      await tx.employee.update({
   625→        where: { id: employee.id },
   626→        data: { state: "idle" },
   627→      });
   628→      await recordCapabilityDenial(tx, task.workspaceId, employee.id, employee.name, toolName, capCheck);
   629→    });
   630→    return { action: "failed", message: `Capability denied: ${capCheck.capabilityCode} required for ${toolName}` };
   631→  }
   632→
   633→  // Check if an approval already exists for this task
   634→  const existingApproval = await db.approval.findFirst({
   635→    where: { taskId: task.id, status: "pending" },
   636→  });
   637→
   638→  if (existingApproval) {
   639→    // Approval already exists — task should be waiting_approval
   640→    // This shouldn't happen if the state machine is correct, but handle it
   641→    await db.task.update({
   642→      where: { id: task.id },
   643→      data: { status: "waiting_approval" },
   644→    });
   645→    return { action: "waiting_approval", message: "Approval already pending" };
   646→  }
   647→
   648→  // Create the approval
   649→  const proposedAction: Record<string, string> = {};
   650→  for (const [k, v] of Object.entries(input)) {
   651→    if (k !== "tool") proposedAction[k] = String(v);
   652→  }
   653→
   654→  const timeoutAt = new Date();
   655→  timeoutAt.setHours(timeoutAt.getHours() + 12); // 12-hour timeout
   656→
   657→  // ─── Generate Execution Contract ──────────────────────────────────────────
   658→  // Before creating the approval, generate an immutable Execution Contract.
   659→  // The contract captures the complete decision context at this moment.
   660→  // The approval references the contract — not mutable task state.
   661→  const isFinanceApproval = toolName === "send_reminder";
   662→
   663→  // Build contract input from the step's input (which contains the Finance
   664→  // Brain's recommendation for finance tasks, or the tool input for generic tasks)
   665→  const contractInput: ContractInput = {
   666→    taskId: task.id,
   667→    employeeId: employee.id,
   668→    workspaceId: task.workspaceId,
   669→    goal: isFinanceApproval
   670→      ? `Send collection reminder for Invoice ${input.invoiceNumber || "N/A"} to ${input.customerName || "customer"}`
   671→      : `Execute ${toolName} for task: ${task.title}`,
   672→    proposedAction,
   673→    confidence: input.confidence ? parseFloat(input.confidence) : 0.85,
   674→    evidence: input.evidence ? safeParseJson(input.evidence, []) : [],
   675→    memoriesUsed: [], // Populated from the Finance Brain context if available
   676→    policiesUsed: input.policyInfluence ? safeParseJson(input.policyInfluence, []) : [],
   677→    businessImpact: input.businessImpact || input.riskAssessment || "Critical action requiring human approval before execution.",
   678→    affectedSystems: isFinanceApproval
   679→      ? ["invoices", "reminders", "collection_cases"]
   680→      : ["task_steps"],
   681→    rollbackPlan: isFinanceApproval
   682→      ? `Mark the reminder as "not sent" in the reminders table. No external side effects if caught before the email provider sends. If already sent, send a follow-up correction email.`
   683→      : `Mark the tool execution as failed in the task step. No rollback needed for non-external actions.`,
   684→    estimatedBusinessOutcome: isFinanceApproval
   685→      ? `Prompt payment from ${input.customerName || "customer"}, reducing outstanding receivables by ${input.outstanding || "the outstanding amount"}.`
   686→      : `Complete the task step and continue execution.`,
   687→    estimatedTokenCost: 200,
   688→    estimatedExecutionTime: 1000,
   689→    requiredAuthority: "owner",
   690→  };
   691→
   692→  const contract = await generateContract(contractInput);
   693→
   694→  await db.$transaction(async (tx) => {
   695→    // Create the approval record — references the contract
   696→    const approval = await tx.approval.create({
   697→      data: {
   698→        workspaceId: task.workspaceId,
   699→        taskId: task.id,
   700→        employeeId: employee.id,
   701→        tool: toolName,
   702→        toolDisplayName: TOOL_DISPLAY_NAMES[toolName] || toolName,
   703→        proposedAction: JSON.stringify(proposedAction),
   704→        status: "pending",
   705→        criticality: "critical",
   706→        timeoutAt,
   707→      },
   708→    });
   709→
   710→    // Link the contract to the approval
   711→    // (done outside the transaction to avoid FK issues — best-effort)
   712→    // We store the contract ID in the step output instead
   713→
   714→    // Update the step to reference the approval AND the contract
   715→    await tx.taskStep.update({
   716→      where: { id: step.id },
   717→      data: {
   718→        status: "pending", // stays pending until approval is decided
   719→        output: JSON.stringify({
   720→          approvalId: approval.id,
   721→          contractId: contract.id,
   722→          contractNumber: contract.contractNumber,
   723→          contractVersion: contract.version,
   724→          contractHash: contract.contractHash,
   725→          status: "waiting",
   726→        }),
   727→      },
   728→    });
   729→
   730→    // Transition task to waiting_approval
   731→    await tx.task.update({
   732→      where: { id: task.id },
   733→      data: { status: "waiting_approval" },
   734→    });
   735→
   736→    // Update employee state
   737→    await tx.employee.update({
   738→      where: { id: employee.id },
   739→      data: { state: "waiting_approval", pendingApprovals: { increment: 1 } },
   740→    });
   741→
   742→    // Link contract to approval (best-effort, outside transaction boundary)
   743→    // We do this after the transaction commits to avoid FK constraint issues
   744→    // The contract ID is already in the step output, so it's recoverable
   745→
   746→    // Write audit entry — include finance reasoning chain AND contract hash
   747→    const isFinanceApproval = toolName === "send_reminder";
   748→    await appendAudit(tx, {
   749→      workspaceId: task.workspaceId,
   750→      entryType: isFinanceApproval ? "reminder_approval_requested" : "approval_requested",
   751→      actorType: "employee",
   752→      actorId: employee.id,
   753→      actorName: employee.name,
   754→      targetType: isFinanceApproval ? "invoice" : "approval",
   755→      targetId: isFinanceApproval ? (input.invoiceId || approval.id) : approval.id,
   756→      payload: {
   757→        tool: toolName,
   758→        task: task.title,
   759→        criticality: "critical",
   760→        // ─── Execution Contract Reference ──────────────────────────────────
   761→        contractId: contract.id,
   762→        contractNumber: contract.contractNumber,
   763→        contractVersion: String(contract.version),
   764→        contractHash: contract.contractHash,
   765→        // ─── Finance Brain Reasoning Chain ──────────────────────────────────
   766→        ...(input.invoiceNumber ? { invoiceNumber: input.invoiceNumber } : {}),
   767→        ...(input.customerName ? { customer: input.customerName } : {}),
   768→        ...(input.outstanding ? { outstanding: input.outstanding } : {}),
   769→        ...(input.daysOverdue ? { daysOverdue: input.daysOverdue } : {}),
   770→        ...(input.recommendedAction ? { recommendedAction: input.recommendedAction } : {}),
   771→        ...(input.why ? { why: input.why.substring(0, 500) } : {}),
   772→        ...(input.confidence ? { confidence: input.confidence } : {}),
   773→        ...(input.riskAssessment ? { riskAssessment: input.riskAssessment.substring(0, 200) } : {}),
   774→        ...(input.policyInfluence ? { policies: input.policyInfluence.substring(0, 300) } : {}),
   775→        ...(input.customerHistoryInfluence ? { customerHistory: input.customerHistoryInfluence.substring(0, 300) } : {}),
   776→        ...(input.rejectedAlternatives ? { rejectedAlternatives: input.rejectedAlternatives.substring(0, 300) } : {}),
   777→      },
   778→    });
   779→
   780→    // Create a notification for the user
   781→    const task2 = await tx.task.findUnique({ where: { id: task.id } });

   784→        ? `${employee.name} needs approval to send a reminder`

   787→        ? `Invoice ${input.invoiceNumber || ""} — ${input.customerName || ""}: ${input.why ? input.why.substring(0, 150) : "Reminder requires approval with full finance reasoning."}`
   788→        : `${TOOL_DISPLAY_NAMES[toolName] || toolName} — ${proposedAction.to || proposedAction.subject || "Action requires review"}`;
   789→      await tx.notification.create({
   790→        data: {
   791→          workspaceId: task.workspaceId,
   792→          userId: task2.assignedBy,
   793→          type: "approval_pending",
   794→          title: notifTitle,
   795→          body: notifBody,
   796→          referenceType: "approval",
   797→          referenceId: approval.id,
   798→          channel: "in_app",
   799→          status: "delivered",
   800→        },

   802→    }

   898→// ─── Resume After Approval ───────────────────────────────────────────────────
   899→
   900→/**
   901→ * Called by the approval API when a human approves an action.
   902→ * Marks the approval_gate step as completed and transitions the task back
   903→ * to executing so the worker can continue with the next step.
   904→ *
   905→ * Also "executes" the approved tool (e.g., actually sends the email in a
   906→ * real implementation).
   907→ */
   908→export async function resumeAfterApproval(
   909→  taskId: string,
   910→  approvalId: string,
   911→  approvedBy: string,
   912→  approvedByName: string
   913→): Promise<void> {
   914→  const task = await db.task.findUnique({
   915→    where: { id: taskId },
   916→    include: { employee: true, steps: { orderBy: { stepNumber: "asc" } } },
   917→  });
   918→
   919→  if (!task) return;
   920→
   921→  const approval = await db.approval.findUnique({ where: { id: approvalId } });
   922→  if (!approval) return;
   923→
   924→  // Find the approval_gate step that was waiting
   925→  const gateStep = task.steps.find(
   926→    (s) => s.stepType === "approval_gate" && s.status === "pending"
   927→  );
   928→
   929→  // ─── Capability Verification (post-approval) ──────────────────────────────
   930→  // Even after approval, verify the employee still has the capability.
   931→  // Capabilities can be revoked between approval and execution.
   932→  const postApprovalCapCheck = await checkCapability(task.employeeId, approval.tool);
   933→  if (!postApprovalCapCheck.allowed) {
   934→    // Capability was revoked after approval — fail the task
   935→    await db.$transaction(async (tx) => {
   936→      await tx.task.update({
   937→        where: { id: taskId },
   938→        data: { status: "failed" },
   939→      });
   940→      await tx.employee.update({
   941→        where: { id: task.employeeId },
   942→        data: { state: "idle", pendingApprovals: { decrement: 1 } },
   943→      });
   944→      await recordCapabilityDenial(tx, task.workspaceId, task.employeeId, task.employee.name, approval.tool, postApprovalCapCheck);
   945→    });
   946→    return;
   947→  }
   948→
   949→  // Execute the approved tool BEFORE the transaction (it may do DB operations)
   950→  const proposedAction = JSON.parse(approval.proposedAction);
   951→  const financeTools = ["generate_reminder", "send_reminder", "update_collection_case"];
   952→  const isFinanceTool = financeTools.includes(approval.tool);
   953→
   954→  let toolResult: { output: Record<string, string>; tokens: number; durationMs: number };
   955→  if (isFinanceTool) {
   956→    toolResult = await executeFinanceTool(approval.tool, proposedAction, task.workspaceId, task.employeeId);
   957→  } else {
   958→    toolResult = executeTool(approval.tool, proposedAction);
   959→  }
   960→
   961→  await db.$transaction(async (tx) => {
   962→    // Mark the gate step as completed
   963→    if (gateStep) {
   964→      await tx.taskStep.update({
   965→        where: { id: gateStep.id },
   966→        data: {
   967→          status: "completed",
   968→          output: JSON.stringify({
   969→            ...toolResult.output,
   970→            approvalId,
   971→            approvedBy: approvedByName,
   972→          }),
   973→          tokens: toolResult.tokens,
   974→          durationMs: toolResult.durationMs,
   975→          completedAt: new Date(),
   976→        },
   977→      });
   978→    }
   979→
   980→    // Transition task back to executing
   981→    await tx.task.update({
   982→      where: { id: taskId },
   983→      data: {
   984→        status: "executing",
   985→        tokenUsage: { increment: toolResult.tokens },
   986→      },
   987→    });
   988→
   989→    // Update employee state
   990→    await tx.employee.update({
   991→      where: { id: task.employeeId },
   992→      data: {
   993→        state: "executing",
   994→        pendingApprovals: { decrement: 1 },
   995→      },
   996→    });
   997→
   998→    // Write audit entry for the approval decision
   999→    const isFinanceApproval = approval.tool === "send_reminder";
  1000→    await appendAudit(tx, {
  1001→      workspaceId: task.workspaceId,
  1002→      entryType: isFinanceApproval ? "reminder_approved" : "approval_decided",
  1003→      actorType: "user",
  1004→      actorId: approvedBy,
  1005→      actorName: approvedByName,
  1006→      targetType: isFinanceApproval ? "invoice" : "approval",
  1007→      targetId: isFinanceApproval ? (proposedAction.invoiceId || approvalId) : approvalId,
  1008→      payload: {
  1009→        decision: "approved",
  1010→        tool: approval.tool,
  1011→        employee: task.employee.name,
  1012→        ...(proposedAction.invoiceNumber ? { invoiceNumber: proposedAction.invoiceNumber } : {}),
  1013→        ...(proposedAction.customerName ? { customer: proposedAction.customerName } : {}),
  1014→      },
  1015→    });
  1016→
  1017→    // Write audit entry for the tool execution
  1018→    await appendAudit(tx, {
  1019→      workspaceId: task.workspaceId,
  1020→      entryType: isFinanceApproval ? "reminder_sent" : "tool_executed",
  1021→      actorType: "employee",
  1022→      actorId: task.employeeId,
  1023→      actorName: task.employee.name,
  1024→      targetType: isFinanceApproval ? "invoice" : "task_step",
  1025→      targetId: proposedAction.invoiceId || gateStep?.id || "",
  1026→      payload: {
  1027→        tool: approval.tool,
  1028→        status: "completed",
  1029→        approved_by: approvedByName,
  1030→        ...(proposedAction.invoiceNumber ? { invoiceNumber: proposedAction.invoiceNumber } : {}),
  1031→        ...(proposedAction.customerName ? { customer: proposedAction.customerName } : {}),
  1032→      },
  1033→    });
  1034→  });
  1035→
  1036→  // ─── Approve the Execution Contract ──────────────────────────────────────
  1037→  // The contract becomes immutable. Its hash is locked.
  1038→  try {
  1039→    // Find the contract linked to this task's approval gate step
  1040→    if (gateStep) {
  1041→      const stepOutput = JSON.parse(gateStep.output || "{}");
  1042→      if (stepOutput.contractId) {
  1043→        await approveContract(stepOutput.contractId, approvedBy);
  1044→        await linkApprovalToContract(stepOutput.contractId, approvalId);
  1045→      }
  1046→    }
  1047→  } catch (err) {
  1048→    console.error(`[Executor] Contract approval failed for approval ${approvalId}:`, err);
  1049→  }
  1050→
  1051→  // ─── Record Approval Feedback in Memory ──────────────────────────────────
  1052→  // The employee learns from the manager's approval decision immediately.
  1053→  try {
  1054→    await recordApprovalFeedback(
  1055→      task.employeeId,
  1056→      task.workspaceId,
  1057→      approvalId,
  1058→      "approved",
  1059→      null,
  1060→      approval.tool,
  1061→      proposedAction
  1062→    );
  1063→  } catch (err) {
  1064→    console.error(`[Executor] Memory recording failed for approval ${approvalId}:`, err);
  1065→  }
  1066→
  1067→  // ─── Update Employee Profile (approval approved) ─────────────────────────
  1068→  try {
  1069→    await recordProfileEvent({
  1070→      type: "approval_approved",
  1071→      employeeId: task.employeeId,
  1072→      workspaceId: task.workspaceId,
  1073→      taskId,
  1074→      toolName: approval.tool,
  1075→    });
  1076→    // If this was a reminder send, record the specific KPI
  1077→    if (approval.tool === "send_reminder" || approval.tool === "send_email") {
  1078→      await recordProfileEvent({
  1079→        type: "reminder_sent",
  1080→        employeeId: task.employeeId,
  1081→        workspaceId: task.workspaceId,
  1082→        taskId,
  1083→        toolName: approval.tool,
  1084→      });
  1085→    }
  1086→    // Bookkeeping: contract_approved is zero-XP (XP already counted via
  1087→    // approval_approved) but nudges the accuracy score so the profile
  1088→    // reflects the contract lifecycle separately from the approval lifecycle.
  1089→    if (gateStep) {
  1090→      const stepOutput = JSON.parse(gateStep.output || "{}");
  1091→      if (stepOutput.contractId) {
  1092→        await recordProfileEvent({
  1093→          type: "contract_approved",
  1094→          employeeId: task.employeeId,
  1095→          workspaceId: task.workspaceId,
  1096→          taskId,
  1097→        });
  1098→      }
  1099→    }
  1100→  } catch (err) {
  1101→    console.error(`[Executor] Profile update failed for approval ${approvalId}:`, err);
  1102→  }
  1103→}

  1105→/**

  1109→export async function failAfterApprovalRejection(

  1121→  if (!task) return;

  1131→    // Mark the gate step as failed
  1132→    if (gateStep) {
  1133→      await tx.taskStep.update({
  1134→        where: { id: gateStep.id },
  1135→        data: {
  1136→          status: "failed",
  1137→          output: JSON.stringify({
  1138→            approvalId,
  1139→            rejectedBy: rejectedByName,
  1140→            reason: reason || "Rejected by human",
  1141→          }),
  1142→          completedAt: new Date(),
  1143→        },
  1144→      });
  1145→    }
  1146→
  1147→    // Transition task to failed
  1148→    await tx.task.update({
  1149→      where: { id: taskId },
  1150→      data: { status: "failed" },
  1151→    });
  1152→
  1153→    // Update employee state
  1154→    await tx.employee.update({
  1155→      where: { id: task.employeeId },
  1156→      data: {
  1157→        state: "idle",
  1158→        pendingApprovals: { decrement: 1 },
  1159→      },
  1160→    });
  1161→
  1162→    // Write audit entry for the approval decision
  1163→    await appendAudit(tx, {
  1164→      workspaceId: task.workspaceId,
  1165→      entryType: "approval_decided",
  1166→      actorType: "user",
  1167→      actorId: rejectedBy,
  1168→      actorName: rejectedByName,
  1169→      targetType: "approval",
  1170→      targetId: approvalId,
  1171→      payload: {
  1172→        decision: "rejected",
  1173→        tool: approvalRecord?.tool || "",
  1174→        employee: task.employee.name,
  1175→        reason: reason || "Rejected",
  1176→      },
  1177→    });
  1178→
  1179→    // Write task_failed audit
  1180→    await appendAudit(tx, {
  1181→      workspaceId: task.workspaceId,
  1182→      entryType: "task_failed",
  1183→      actorType: "user",
  1184→      actorId: rejectedBy,
  1185→      actorName: rejectedByName,
  1186→      targetType: "task",
  1187→      targetId: taskId,
  1188→      payload: {
  1189→        reason: `Approval rejected: ${reason || "No reason provided"}`,
  1190→      },
  1191→    });
  1192→  });
  1193→
  1194→  // ─── Record Rejection Feedback in Memory ──────────────────────────────────
  1195→  // The employee learns from the manager's rejection — this is especially
  1196→  // important because rejections often carry a reason that teaches the
  1197→  // employee what NOT to do next time.
  1198→  try {
  1199→    const proposedAction = JSON.parse(approvalRecord?.proposedAction || "{}");
  1200→    await recordApprovalFeedback(
  1201→      task.employeeId,
  1202→      task.workspaceId,
  1203→      approvalId,
  1204→      "rejected",
  1205→      reason || null,
  1206→      approvalRecord?.tool || "",
  1207→      proposedAction
  1208→    );
  1209→  } catch (err) {
  1210→    console.error(`[Executor] Memory recording failed for rejection ${approvalId}:`, err);
  1211→  }
  1212→
  1213→  // ─── Reject the Execution Contract ────────────────────────────────────────
  1214→  // The contract is marked as rejected but remains permanently searchable.
  1215→  try {
  1216→    if (gateStep) {
  1217→      const stepOutput = JSON.parse(gateStep.output || "{}");
  1218→      if (stepOutput.contractId) {
  1219→        await rejectContract(stepOutput.contractId, rejectedBy, reason);
  1220→      }
  1221→    }
  1222→  } catch (err) {
  1223→    console.error(`[Executor] Contract rejection failed for approval ${approvalId}:`, err);
  1224→  }
  1225→
  1226→  // ─── Update Employee Profile (approval rejected) ─────────────────────────
  1227→  try {
  1228→    await recordProfileEvent({
  1229→      type: "approval_rejected",
  1230→      employeeId: task.employeeId,
  1231→      workspaceId: task.workspaceId,
  1232→      taskId,
  1233→      toolName: approvalRecord?.tool || "",
  1234→    });
  1235→    await recordProfileEvent({
  1236→      type: "task_failed",
  1237→      employeeId: task.employeeId,
  1238→      workspaceId: task.workspaceId,
  1239→      taskId,
  1240→    });
  1241→    // Bookkeeping: contract_rejected is zero-XP (XP already counted via
  1242→    // approval_rejected + task_failed) but nudges the accuracy score.
  1243→    if (gateStep) {
  1244→      const stepOutput = JSON.parse(gateStep.output || "{}");
  1245→      if (stepOutput.contractId) {
  1246→        await recordProfileEvent({
  1247→          type: "contract_rejected",
  1248→          employeeId: task.employeeId,
  1249→          workspaceId: task.workspaceId,
  1250→          taskId,
  1251→        });
  1252→      }
  1253→    }
  1254→  } catch (err) {
  1255→    console.error(`[Executor] Profile update failed for rejection ${approvalId}:`, err);
  1256→  }
  1257→}
