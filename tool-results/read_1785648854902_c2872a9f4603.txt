   149→   149→  // ─── CSV parsing ─────────────────────────────────────────────────────────
   150→   150→  function parseCsv(text: string): ParsedInvoice[] {
   151→   151→    const lines = text.trim().split(/\r?\n/).filter((l) => l.trim());
   152→   152→    if (lines.length < 2) {
   153→   153→      throw new Error("CSV must have a header row and at least one data row.");
   154→   154→    }
   155→   155→
   156→   156→    const headers = lines[0].split(",").map((h) => h.trim().toLowerCase());
   157→   157→    const required = ["customer_name", "customer_email", "invoice_number", "issue_date", "due_date", "subtotal", "tax"];
   158→   158→    for (const req of required) {
   159→   159→      if (!headers.includes(req)) {
   160→   160→        throw new Error(`Missing required column: ${req}. Expected columns: ${required.join(", ")}`);
   161→   161→      }
   162→   162→    }
   163→   163→
   164→   164→    const invoices: ParsedInvoice[] = [];
   165→   165→    for (let i = 1; i < lines.length; i++) {
   166→   166→      const cols = parseCsvLine(lines[i]);
   167→   167→      if (cols.length < headers.length) {
   168→   168→        throw new Error(`Row ${i + 1} has ${cols.length} columns, expected ${headers.length}.`);
   169→   169→      }
   170→   170→
   171→   171→      const row: Record<string, string> = {};
   172→   172→      headers.forEach((h, idx) => { row[h] = cols[idx]?.trim() || ""; });
   173→   173→
   174→   174→      // Convert amounts from rupees to paise (1 INR = 100 paise)
   175→   175→      const subtotal = Math.round(parseFloat(row.subtotal) * 100);
   176→   176→      const tax = Math.round(parseFloat(row.tax) * 100);
   177→   177→
   178→   178→      if (isNaN(subtotal) || isNaN(tax)) {
   179→   179→        throw new Error(`Row ${i + 1}: subtotal and tax must be numbers.`);
   180→   180→      }
   181→   181→
   182→   182→      invoices.push({
   183→   183→        customerName: row.customer_name,
   184→   184→        customerEmail: row.customer_email,
   185→   185→        invoiceNumber: row.invoice_number,
   186→   186→        issueDate: row.issue_date,
   187→   187→        dueDate: row.due_date,
   188→   188→        subtotal,
   189→   189→        tax,
   190→   190→      });
   191→   191→    }
   192→   192→
   193→   193→    return invoices;
   194→   194→  }
   195→   195→
   196→   196→  function parseCsvLine(line: string): string[] {
   197→   197→    const result: string[] = [];
   198→   198→    let current = "";
   199→   199→    let inQuotes = false;
   200→   200→    for (let i = 0; i < line.length; i++) {
   201→   201→      const char = line[i];
   202→   202→      if (char === '"') {
   203→   203→        inQuotes = !inQuotes;
   204→   204→      } else if (char === "," && !inQuotes) {
   205→   205→        result.push(current);
   206→   206→        current = "";
   207→   207→      } else {
   208→   208→        current += char;
   209→   209→      }
   210→   210→    }
   211→   211→    result.push(current);
   212→   212→    return result;
   213→   213→  }
