     1→     1→/**
     2→     2→ * BIHARI AI — Autonomous Learning & Skill Evolution Engine (EMP-002)
     3→     3→ *
     4→     4→ * Transforms the Employee Profile from a static record into a continuously
     5→     5→ * learning workforce. Every completed task generates a deterministic Outcome
     6→     6→ * Evaluation, which drives:
     7→     7→ *
     8→     8→ *   1. Skill Reinforcement  — skills level up from real outcomes, not usage++
     9→     9→ *   2. Pattern Learning     — reusable patterns detected from repeated outcomes
    10→    10→ *   3. Weakness Detection   — repeated failures reduce trust
    11→    11→ *   4. Strength Detection   — repeated successes increase trust
    12→    12→ *   5. Business Outcome History — append-only ledger of measurable impact
    13→    13→ *   6. Career Timeline      — chronological log of every career event
    14→    14→ *   7. Achievement Unlocks  — milestone-based recognition
    15→    15→ *
    16→    16→ * Design principles (NON-NEGOTIABLE):
    17→    17→ *   - NO fake learning. NO random XP. NO LLM-generated scores.
    18→    18→ *   - Everything comes from MEASURABLE BUSINESS OUTCOMES.
    19→    19→ *   - Normalized tables — no JSON blobs for core entities.
    20→    20→ *   - Append-only history. Business outcomes never overwrite.
    21→    21→ *   - Incremental updates only — no expensive scans.
    22→    22→ *   - Generic — works for Finance, HR, Sales Ops, Procurement, etc.
    23→    23→ *
    24→    24→ * Architecture:
    25→    25→ *   The engine is called ONCE per task completion from the executor's
    26→    26→ *   `completeTask()` function. It runs AFTER `recordProfileEvent` so the
    27→    27→ *   profile counters are already updated. The engine then:
    28→    28→ *     1. Loads the task with steps + approvals + finance domain data
    29→    29→ *     2. Builds an OutcomeEvaluation (deterministic scorecard)
    30→    30→ *     3. Applies skill reinforcements based on the outcome
    31→    31→ *     4. Detects patterns (customer payment behavior, reminder effectiveness)
    32→    32→ *     5. Detects weaknesses (high rejection rate, low confidence, etc.)
    33→    33→ *     6. Detects strengths (high approval rate, fast execution, etc.)
    34→    34→ *     7. Records business outcomes (append-only ledger)
    35→    35→ *     8. Appends career timeline entries
    36→    36→ *     9. Checks for achievement unlocks
    37→    37→ *    10. Emits audit events for every evolution
    38→    38→ *
    39→    39→ * All of this happens in best-effort try/catch blocks — learning failures
    40→    40→ * must NEVER break task completion.
    41→    41→ */

    61→    61→//   +3 = major success (payment recovered, customer responded)
    62→    62→//   +2 = good outcome (approval approved, reminder sent)
    63→    63→//   +1 = minor success (task completed without issues)
    64→    64→//   -1 = minor issue (single rejection, single override)
    65→    65→//   -2 = moderate failure (rollback needed, multiple rejections)
    66→    66→//   -3 = major failure (task failed, capability denied repeatedly)

   107→   107→// ─── 1. Outcome Evaluation ───────────────────────────────────────────────────
   108→   108→
   109→   109→/**
   110→   110→ * Evaluates a completed task and produces a deterministic OutcomeEvaluation.
   111→   111→ *
   112→   112→ * This is the SINGLE entry point called from the executor after task
   113→   113→ * completion. It runs the full learning pipeline:
   114→   114→ *   evaluate → reinforce → patterns → weaknesses → strengths → outcomes → timeline → achievements
   115→   115→ *
   116→   116→ * All steps are best-effort. A failure in any step logs an error but does
   117→   117→ * not block the others.
   118→   118→ */
   119→   119→export async function evaluateAndLearn(data: TaskOutcomeData): Promise<void> {
   120→   120→  const { taskId, employeeId, workspaceId } = data;
   121→   121→
   122→   122→  // Idempotency: skip if an evaluation already exists for this task
   123→   123→  const existing = await db.outcomeEvaluation.findUnique({ where: { taskId } });
   124→   124→  if (existing) {
   125→   125→    return;
   126→   126→  }
   127→   127→
   128→   128→  // Load the task with everything we need
   129→   129→  const task = await db.task.findUnique({
   130→   130→    where: { id: taskId },
   131→   131→    include: {
   132→   132→      steps: { orderBy: { stepNumber: "asc" } },
   133→   133→      approvals: true,
   134→   134→      employee: true,
   135→   135→    },
   136→   136→  });
   137→   137→
   138→   138→  if (!task) {
   139→   139→    console.error(`[Learning] Task not found: ${taskId}`);
   140→   140→    return;
   141→   141→  }
   142→   142→
   143→   143→  // Only evaluate completed tasks (not failed/stopped)
   144→   144→  if (task.status !== "completed") {
   145→   145→    return;
   146→   146→  }
   147→   147→
   148→   148→  const evaluation = await buildOutcomeEvaluation(task);
   149→   149→  if (!evaluation) return;
   150→   150→
   151→   151→  // Run the learning pipeline (each step is independent & best-effort)
   152→   152→  try { await reinforceSkillsFromOutcome(task, evaluation); } catch (e) { console.error("[Learning] reinforceSkills failed:", e); }
   153→   153→  try { await detectPatternsFromOutcome(task, evaluation); } catch (e) { console.error("[Learning] detectPatterns failed:", e); }
   154→   154→  try { await detectWeaknesses(employeeId, workspaceId); } catch (e) { console.error("[Learning] detectWeaknesses failed:", e); }
   155→   155→  try { await detectStrengths(employeeId, workspaceId); } catch (e) { console.error("[Learning] detectStrengths failed:", e); }
   156→   156→  try { await recordBusinessOutcomesFromTask(task, evaluation); } catch (e) { console.error("[Learning] recordBusinessOutcomes failed:", e); }
   157→   157→  try { await appendTimelineFromOutcome(task, evaluation); } catch (e) { console.error("[Learning] appendTimeline failed:", e); }
   158→   158→  try { await checkAchievementUnlocks(employeeId, workspaceId, evaluation); } catch (e) { console.error("[Learning] checkAchievements failed:", e); }
   159→   159→}

   161→   161→/**
   162→   162→ * Builds the deterministic OutcomeEvaluation scorecard.
   163→   163→ * Pure function — no side effects except the DB write.
   164→   164→ */
   165→   165→async function buildOutcomeEvaluation(task: any): Promise<any | null> {
   166→   166→  const { id: taskId, workspaceId, employeeId } = task;
   167→   167→
   168→   168→  // ─── Execution quality (from task + steps + approvals) ──────────────────
   169→   169→  const stepCount = task.steps.length;
   170→   170→  const toolCallCount = task.steps.filter((s: any) => s.stepType === "tool_call").length;
   171→   171→  const approvalGateCount = task.steps.filter((s: any) => s.stepType === "approval_gate").length;
   172→   172→  const approvals = task.approvals || [];
   173→   173→  const approvalRejections = approvals.filter((a: any) => a.decision === "rejected").length;
   174→   174→  const humanOverrides = approvals.filter((a: any) => a.originalAction !== null).length;
   175→   175→  const tokenUsage = task.tokenUsage || 0;
   176→   176→
   177→   177→  // Execution time
   178→   178→  const executionTimeMs =
   179→   179→    task.startedAt && task.completedAt
   180→   180→      ? new Date(task.completedAt).getTime() - new Date(task.startedAt).getTime()
   181→   181→      : 0;
   182→   182→
   183→   183→  // Capability denials + step failures (scan step outputs)
   184→   184→  let capabilityDenials = 0;
   185→   185→  let stepFailureCount = 0;
   186→   186→  for (const step of task.steps) {
   187→   187→    if (step.status === "failed") stepFailureCount++;
   188→   188→    if (step.output && step.output.includes("capability_denied")) {
   189→   189→      capabilityDenials++;
   190→   190→    }
   191→   191→  }
   192→   192→
   193→   193→  // ─── Business outcome (from finance domain — generic approach) ──────────
   194→   194→  const { paymentReceived, paymentAmount, invoiceResolved, customerResponded, reminderSentCount, invoiceIds, customerIds } =
   195→   195→    await detectBusinessOutcomes(task, workspaceId);
   196→   196→
   197→   197→  // ─── SLA ────────────────────────────────────────────────────────────────
   198→   198→  // SLA target: 24 hours for finance tasks (configurable per role in future)
   199→   199→  const slaTargetHours = 24;
   200→   200→  const slaActualHours = executionTimeMs / (1000 * 60 * 60);
   201→   201→  const slaAchieved = slaActualHours <= slaTargetHours && stepFailureCount === 0;
   202→   202→
   203→   203→  // ─── Confidence accuracy ────────────────────────────────────────────────
   204→   204→  const completedSteps = task.steps.filter((s: any) => s.status === "completed" && s.confidence !== null);
   205→   205→  const avgConfidence =
   206→   206→    completedSteps.length > 0
   207→   207→      ? completedSteps.reduce((sum: number, s: any) => sum + (s.confidence || 0), 0) / completedSteps.length
   208→   208→      : 0.85;
   209→   209→  const expectedSuccess = avgConfidence >= 0.7;
   210→   210→  const actualSuccess = task.status === "completed" && approvalRejections === 0;
   211→   211→  const confidenceAccuracy = expectedSuccess === actualSuccess ? 1.0 : 0.0;
   212→   212→
   213→   213→  // ─── Human corrections ──────────────────────────────────────────────────
   214→   214→  const humanCorrections = approvalRejections + humanOverrides + capabilityDenials;
   215→   215→
   216→   216→  // ─── Quality score (deterministic formula, 0-100) ───────────────────────
   217→   217→  let qualityScore = 0;
   218→   218→  if (actualSuccess) qualityScore += 40;
   219→   219→  if (humanCorrections === 0) qualityScore += 20;
   220→   220→  else if (humanCorrections === 1) qualityScore += 10;
   221→   221→  if (slaAchieved) qualityScore += 15;
   222→   222→  if (confidenceAccuracy === 1.0) qualityScore += 15;
   223→   223→  if (paymentReceived || customerResponded) qualityScore += 10;
   224→   224→
   225→   225→  // ─── Outcome summary (deterministic, built from facts) ──────────────────
   226→   226→  const outcomeParts: string[] = [];
   227→   227→  outcomeParts.push(`Task completed with ${stepCount} steps (${toolCallCount} tool calls, ${approvalGateCount} approval gates)`);
   228→   228→  if (approvalRejections > 0) outcomeParts.push(`${approvalRejections} approval rejection(s)`);
   229→   229→  if (humanOverrides > 0) outcomeParts.push(`${humanOverrides} human override(s)`);
   230→   230→  if (capabilityDenials > 0) outcomeParts.push(`${capabilityDenials} capability denial(s)`);
   231→   231→  if (reminderSentCount > 0) outcomeParts.push(`${reminderSentCount} reminder(s) sent`);
   232→   232→  if (paymentReceived) outcomeParts.push(`Payment received: ₹${(paymentAmount / 100).toFixed(2)}`);
   233→   233→  if (customerResponded) outcomeParts.push("Customer responded to reminder");
   234→   234→  if (slaAchieved) outcomeParts.push(`SLA achieved (${slaActualHours.toFixed(1)}h / ${slaTargetHours}h)`);
   235→   235→  else outcomeParts.push(`SLA missed (${slaActualHours.toFixed(1)}h / ${slaTargetHours}h)`);
   236→   236→  const outcomeSummary = outcomeParts.join(". ");
   237→   237→
   238→   238→  // ─── Persist the evaluation ─────────────────────────────────────────────
   239→   239→  const evaluation = await db.outcomeEvaluation.create({
   240→   240→    data: {
   241→   241→      workspaceId,
   242→   242→      employeeId,
   243→   243→      taskId,
   244→   244→      stepCount,
   245→   245→      toolCallCount,
   246→   246→      approvalGateCount,
   247→   247→      approvalRejections,
   248→   248→      humanOverrides,
   249→   249→      capabilityDenials,
   250→   250→      tokenUsage,
   251→   251→      executionTimeMs,
   252→   252→      rollbackNeeded: false,
   253→   253→      paymentReceived,
   254→   254→      paymentAmount,
   255→   255→      invoiceResolved,
   256→   256→      customerResponded,
   257→   257→      reminderSentCount,
   258→   258→      slaAchieved,
   259→   259→      slaTargetHours,
   260→   260→      slaActualHours: Math.round(slaActualHours * 100) / 100,
   261→   261→      errorCount: stepFailureCount,
   262→   262→      stepFailureCount,
   263→   263→      expectedSuccess,
   264→   264→      actualSuccess,
   265→   265→      confidenceAccuracy,
   266→   266→      qualityScore,
   267→   267→      humanCorrections,
   268→   268→      outcomeSummary,
   269→   269→    },
   270→   270→  });
   271→   271→
   272→   272→  // Audit: profile_updated (the outcome evaluation IS a profile update)
   273→   273→  await emitAudit(workspaceId, employeeId, "profile_updated", "outcome_evaluation", evaluation.id, {
   274→   274→    taskId,
   275→   275→    qualityScore: String(qualityScore),
   276→   276→    actualSuccess: String(actualSuccess),
   277→   277→    humanCorrections: String(humanCorrections),
   278→   278→    paymentReceived: String(paymentReceived),
   279→   279→    paymentAmount: String(paymentAmount),
   280→   280→  });
   281→   281→
   282→   282→  // Attach the computed (non-persisted) fields for downstream functions.
   283→   283→  // These are not DB columns but are needed by recordBusinessOutcomesFromTask,
   284→   284→  // detectPatternsFromOutcome, etc.
   285→   285→  return {
   286→   286→    ...evaluation,
   287→   287→    invoiceIds,
   288→   288→    customerIds,
   289→   289→  };
   290→   290→}
   291→   291→
   292→   292→/**
   293→   293→ * Detects business outcomes for a task by examining finance domain data.
   294→   294→ * GENERIC approach: scan the task's steps for invoice IDs and check if
   295→   295→ * payments were received or customers responded. Returns empty results for
   296→   296→ * non-finance tasks (which is correct — they have no finance outcomes).
   297→   297→ */
   298→   298→async function detectBusinessOutcomes(task: any, workspaceId: string): Promise<{
   299→   299→  paymentReceived: boolean;
   300→   300→  paymentAmount: number;
   301→   301→  invoiceResolved: boolean;
   302→   302→  customerResponded: boolean;
   303→   303→  reminderSentCount: number;
   304→   304→  invoiceIds: string[];
   305→   305→  customerIds: string[];
   306→   306→}> {
   307→   307→  const result = {
   308→   308→    paymentReceived: false,
   309→   309→    paymentAmount: 0,
   310→   310→    invoiceResolved: false,
   311→   311→    customerResponded: false,
   312→   312→    reminderSentCount: 0,
   313→   313→    invoiceIds: [] as string[],
   314→   314→    customerIds: [] as string[],
   315→   315→  };
   316→   316→
   317→   317→  const invoiceIds = new Set<string>();
   318→   318→  const customerIds = new Set<string>();
   319→   319→
   320→   320→  // Scan steps for invoiceId / customerId / reminderId in their input/output JSON
   321→   321→  for (const step of task.steps) {
   322→   322→    try {
   323→   323→      const input = JSON.parse(step.input || "{}");
   324→   324→      if (input.invoiceIds && Array.isArray(input.invoiceIds)) {
   325→   325→        input.invoiceIds.forEach((id: string) => invoiceIds.add(id));
   326→   326→      }
   327→   327→      if (input.invoiceId) invoiceIds.add(input.invoiceId);
   328→   328→      if (input.customerId) customerIds.add(input.customerId);
   329→   329→      if (input.tool === "send_reminder" || input.tool === "generate_reminder") {
   330→   330→        result.reminderSentCount++;
   331→   331→      }
   332→   332→    } catch {}
   333→   333→
   334→   334→    try {
   335→   335→      const output = JSON.parse(step.output || "{}");
   336→   336→      if (output.invoiceId) invoiceIds.add(output.invoiceId);
   337→   337→      if (output.customerId) customerIds.add(output.customerId);
   338→   338→      if (output.reminderId) {
   339→   339→        // This step created a reminder — check if it was sent & responded
   340→   340→        const reminder = await db.reminder.findUnique({ where: { id: output.reminderId } });
   341→   341→        if (reminder) {
   342→   342→          if (reminder.status === "sent") result.reminderSentCount = Math.max(result.reminderSentCount, 1);
   343→   343→          if (reminder.status === "responded" || reminder.respondedAt) {
   344→   344→            result.customerResponded = true;
   345→   345→            customerIds.add(reminder.customerId);
   346→   346→          }
   347→   347→          invoiceIds.add(reminder.invoiceId);
   348→   348→          customerIds.add(reminder.customerId);
   349→   349→        }
   350→   350→      }
   351→   351→    } catch {}
   352→   352→  }
   353→   353→
   354→   354→  result.invoiceIds = Array.from(invoiceIds);
   355→   355→  result.customerIds = Array.from(customerIds);
   356→   356→
   357→   357→  // Check for payments received on these invoices AFTER the task started
   358→   358→  if (invoiceIds.size > 0) {
   359→   359→    const taskStartTime = task.startedAt || task.createdAt;
   360→   360→    const payments = await db.payment.findMany({
   361→   361→      where: {
   362→   362→        workspaceId,
   363→   363→        invoiceId: { in: result.invoiceIds },
   364→   364→        paymentDate: { gte: taskStartTime },
   365→   365→        status: "completed",
   366→   366→      },
   367→   367→    });
   368→   368→
   369→   369→    if (payments.length > 0) {
   370→   370→      result.paymentReceived = true;
   371→   371→      result.paymentAmount = payments.reduce((sum, p) => sum + p.amount, 0);
   372→   372→
   373→   373→      const invoices = await db.invoice.findMany({
   374→   374→        where: { id: { in: result.invoiceIds } },
   375→   375→        select: { id: true, outstanding: true },
   376→   376→      });
   377→   377→      result.invoiceResolved = invoices.length > 0 && invoices.every((inv) => inv.outstanding <= 0);
   378→   378→    }
   379→   379→  }
   380→   380→
   381→   381→  return result;
   382→   382→}
   383→   383→
   384→   384→// ─── 2. Skill Reinforcement Engine ───────────────────────────────────────────
   385→   385→
   386→   386→/**
   387→   387→ * Applies skill reinforcements based on the outcome evaluation.
   388→   388→ * Every skill exercised in the task gets reinforced (positively or negatively).
   389→   389→ */
   390→   390→async function reinforceSkillsFromOutcome(task: any, evaluation: any): Promise<void> {
   391→   391→  const { employeeId, workspaceId, taskId } = task;
   392→   392→  const employee = task.employee;
   393→   393→  if (!employee) return;
   394→   394→
   395→   395→  // Determine which skills were exercised in this task
   396→   396→  const skillDefs = getSkillsForRole(employee.role);
   397→   397→  const taskText = `${task.title} ${task.description}`.toLowerCase();
   398→   398→  const toolsUsed = new Set<string>();
   399→   399→  for (const step of task.steps) {
   400→   400→    try {
   401→   401→      const input = JSON.parse(step.input);
   402→   402→      if (input.tool) toolsUsed.add(input.tool);
   403→   403→    } catch {}
   404→   404→  }
   405→   405→
   406→   406→  const exercisedSkills: string[] = [];
   407→   407→  for (const skillDef of skillDefs) {
   408→   408→    const usedByKeyword = skillDef.keywords.some((kw) => taskText.includes(kw.toLowerCase()));
   409→   409→    const usedByTool = skillDef.tools.some((t) => toolsUsed.has(t));
   410→   410→    if (usedByKeyword || usedByTool) {
   411→   411→      exercisedSkills.push(skillDef.name);
   412→   412→    }
   413→   413→  }
   414→   414→
   415→   415→  if (exercisedSkills.length === 0) return;
   416→   416→
   417→   417→  // Build reinforcement rules based on the outcome
   418→   418→  const rules = buildReinforcementRules(evaluation);
   419→   419→
   420→   420→  // Apply each rule to each exercised skill
   421→   421→  for (const skillName of exercisedSkills) {
   422→   422→    for (const rule of rules) {
   423→   423→      await db.skillReinforcement.create({
   424→   424→        data: {
   425→   425→          workspaceId,
   426→   426→          employeeId,
   427→   427→          skillName,
   428→   428→          taskId,
   429→   429→          outcomeId: evaluation.id,
   430→   430→          reason: rule.reason,
   431→   431→          reinforcement: rule.reinforcement,
   432→   432→          notes: rule.notes,
   433→   433→        },
   434→   434→      });
   435→   435→    }
   436→   436→
   437→   437→    // Recompute the skill's level from the reinforcement ledger
   438→   438→    await recomputeSkillLevel(employeeId, workspaceId, skillName, evaluation);
   439→   439→  }
   440→   440→}

   442→   442→/**
   443→   443→ * Builds the list of reinforcement rules based on the outcome evaluation.
   444→   444→ */
   445→   445→function buildReinforcementRules(evaluation: any): ReinforcementRule[] {
   446→   446→  const rules: ReinforcementRule[] = [];
   447→   447→
   448→   448→  if (evaluation.actualSuccess) {
   449→   449→    rules.push({ reason: "task_completed", reinforcement: 1, notes: "Task completed without failures" });
   450→   450→  }
   451→   451→  if (evaluation.paymentReceived) {
   452→   452→    rules.push({
   453→   453→      reason: "payment_recovered",
   454→   454→      reinforcement: 3,
   455→   455→      notes: `Payment of ₹${(evaluation.paymentAmount / 100).toFixed(2)} received after this task`,
   456→   456→    });
   457→   457→  }
   458→   458→  if (evaluation.customerResponded) {
   459→   459→    rules.push({ reason: "customer_responded", reinforcement: 2, notes: "Customer responded to a reminder sent in this task" });
   460→   460→  }
   461→   461→  if (evaluation.slaAchieved) {
   462→   462→    rules.push({ reason: "sla_achieved", reinforcement: 1, notes: `Completed within SLA (${evaluation.slaActualHours}h / ${evaluation.slaTargetHours}h)` });
   463→   463→  }
   464→   464→  if (evaluation.humanCorrections === 0 && evaluation.qualityScore >= 80) {
   465→   465→    rules.push({ reason: "high_quality_execution", reinforcement: 2, notes: `Quality score ${evaluation.qualityScore}/100 with zero human corrections` });
   466→   466→  }
   467→   467→  if (evaluation.approvalRejections > 0) {
   468→   468→    rules.push({
   469→   469→      reason: "approval_rejected",
   470→   470→      reinforcement: -1 * Math.min(3, evaluation.approvalRejections),
   471→   471→      notes: `${evaluation.approvalRejections} approval rejection(s)`,
   472→   472→    });
   473→   473→  }
   474→   474→  if (evaluation.humanOverrides > 0) {
   475→   475→    rules.push({
   476→   476→      reason: "human_override",
   477→   477→      reinforcement: -1 * Math.min(2, evaluation.humanOverrides),
   478→   478→      notes: `${evaluation.humanOverrides} human override(s)`,
   479→   479→    });
   480→   480→  }
   481→   481→  if (evaluation.capabilityDenials > 0) {
   482→   482→    rules.push({
   483→   483→      reason: "capability_denied",
   484→   484→      reinforcement: -2 * evaluation.capabilityDenials,
   485→   485→      notes: `${evaluation.capabilityDenials} capability denial(s)`,
   486→   486→    });
   487→   487→  }
   488→   488→  if (!evaluation.slaAchieved && evaluation.slaActualHours > 0) {
   489→   489→    rules.push({ reason: "sla_missed", reinforcement: -1, notes: `SLA missed (${evaluation.slaActualHours}h / ${evaluation.slaTargetHours}h)` });
   490→   490→  }
   491→   491→  if (evaluation.confidenceAccuracy === 0.0) {
   492→   492→    rules.push({ reason: "confidence_inaccurate", reinforcement: -1, notes: "Employee's confidence did not match the actual outcome" });
   493→   493→  }
   494→   494→  if (evaluation.stepFailureCount > 0) {
   495→   495→    rules.push({
   496→   496→      reason: "step_failure",
   497→   497→      reinforcement: -1 * Math.min(2, evaluation.stepFailureCount),
   498→   498→      notes: `${evaluation.stepFailureCount} step failure(s)`,
   499→   499→    });
   500→   500→  }
   501→   501→
   502→   502→  return rules;
   503→   503→}

   505→   505→/**
   506→   506→ * Recomputes a skill's level from the reinforcement ledger.
   507→   507→ * Level formula:
   508→   508→ *   totalReinforcement = sum of all reinforcement values
   509→   509→ *   level = clamp(1, 10, 1 + floor(max(0, totalReinforcement) / 5))
   510→   510→ *   confidence = clamp(0.5, 0.99, 0.5 + (positive_count * 0.03) - (negative_count * 0.05))
   511→   511→ *
   512→   512→ * If the level increased, emit a skill_promoted audit event + timeline entry.
   513→   513→ */
   514→   514→async function recomputeSkillLevel(
   515→   515→  employeeId: string,
   516→   516→  workspaceId: string,
   517→   517→  skillName: string,
   518→   518→  evaluation: any
   519→   519→): Promise<void> {
   520→   520→  const reinforcements = await db.skillReinforcement.findMany({
   521→   521→    where: { employeeId, skillName },
   522→   522→    orderBy: { createdAt: "asc" },
   523→   523→  });
   524→   524→
   525→   525→  if (reinforcements.length === 0) return;
   526→   526→
   527→   527→  const totalReinforcement = reinforcements.reduce((sum, r) => sum + r.reinforcement, 0);
   528→   528→  const positiveCount = reinforcements.filter((r) => r.reinforcement > 0).length;
   529→   529→  const negativeCount = reinforcements.filter((r) => r.reinforcement < 0).length;
   530→   530→
   531→   531→  const newLevel = Math.max(1, Math.min(10, 1 + Math.floor(Math.max(0, totalReinforcement) / 5)));
   532→   532→  const newConfidence = Math.max(0.5, Math.min(0.99, 0.5 + positiveCount * 0.03 - negativeCount * 0.05));
   533→   533→  const newUsageCount = reinforcements.length;
   534→   534→
   535→   535→  const existing = await db.employeeSkill.findUnique({
   536→   536→    where: { employeeId_name: { employeeId, name: skillName } },
   537→   537→  });
   538→   538→
   539→   539→  const previousLevel = existing?.level || 1;
   540→   540→
   541→   541→  if (existing) {
   542→   542→    await db.employeeSkill.update({
   543→   543→      where: { id: existing.id },
   544→   544→      data: {
   545→   545→        level: newLevel,
   546→   546→        confidence: newConfidence,
   547→   547→        usageCount: newUsageCount,
   548→   548→        lastUsedAt: new Date(),
   549→   549→      },
   550→   550→    });
   551→   551→  } else {
   552→   552→    await db.employeeSkill.create({
   553→   553→      data: {
   554→   554→        employeeId,
   555→   555→        workspaceId,
   556→   556→        name: skillName,
   557→   557→        level: newLevel,
   558→   558→        confidence: newConfidence,
   559→   559→        usageCount: newUsageCount,
   560→   560→        lastUsedAt: new Date(),
   561→   561→      },
   562→   562→    });
   563→   563→  }
   564→   564→
   565→   565→  // If level increased, emit skill_promoted audit + timeline
   566→   566→  if (newLevel > previousLevel) {
   567→   567→    await emitAudit(workspaceId, employeeId, "skill_promoted", "skill", skillName, {
   568→   568→      skillName,
   569→   569→      previousLevel: String(previousLevel),
   570→   570→      newLevel: String(newLevel),
   571→   571→      totalReinforcement: String(totalReinforcement),
   572→   572→    });
   573→   573→
   574→   574→    await db.careerTimelineEntry.create({
   575→   575→      data: {
   576→   576→        workspaceId,
   577→   577→        employeeId,
   578→   578→        entryType: "skill_promoted",
   579→   579→        title: `${skillName} → Level ${newLevel}`,
   580→   580→        description: `Skill "${skillName}" promoted from Level ${previousLevel} to Level ${newLevel} based on ${reinforcements.length} reinforcements (net +${totalReinforcement}).`,
   581→   581→        metadata: JSON.stringify({ skillName, previousLevel, newLevel, totalReinforcement }),
   582→   582→        levelAtTime: 0,
   583→   583→        xpAtTime: 0,
   584→   584→        trustAtTime: 0,
   585→   585→        taskId: evaluation.taskId,
   586→   586→        outcomeId: evaluation.id,
   587→   587→      },
   588→   588→    });
   589→   589→  } else if (!existing && newLevel === 1) {
   590→   590→    // First time this skill was learned
   591→   591→    await db.careerTimelineEntry.create({
   592→   592→      data: {
   593→   593→        workspaceId,
   594→   594→        employeeId,
   595→   595→        entryType: "skill_learned",
   596→   596→        title: `Learned: ${skillName}`,
   597→   597→        description: `New skill "${skillName}" acquired with initial reinforcement +${totalReinforcement}.`,
   598→   598→        metadata: JSON.stringify({ skillName, initialReinforcement: totalReinforcement }),
   599→   599→        levelAtTime: 0,
   600→   600→        xpAtTime: 0,
   601→   601→        trustAtTime: 0,
   602→   602→        taskId: evaluation.taskId,
   603→   603→        outcomeId: evaluation.id,
   604→   604→      },
   605→   605→    });
   606→   606→  }
   607→   607→
   608→   608→  // Sync the skills JSON on the profile (for quick access by the UI)
   609→   609→  await syncSkillsJson(employeeId);
   610→   610→}

   630→   630→// ─── 3. Pattern Learning ─────────────────────────────────────────────────────
   631→   631→
   632→   632→/**
   633→   633→ * Detects reusable patterns from the outcome evaluation.
   634→   634→ * Patterns are STORED ONLY — not yet integrated with the Planner.
   635→   635→ */
   636→   636→async function detectPatternsFromOutcome(task: any, evaluation: any): Promise<void> {
   637→   637→  const { employeeId, workspaceId } = task;
   638→   638→
   639→   639→  // ─── Pattern 1: Customer payment behavior ───────────────────────────────
   640→   640→  if (evaluation.paymentReceived && evaluation.invoiceIds.length > 0) {
   641→   641→    for (const invoiceId of evaluation.invoiceIds) {
   642→   642→      const invoice = await db.invoice.findUnique({
   643→   643→        where: { id: invoiceId },
   644→   644→        include: { customer: true, reminders: { orderBy: { createdAt: "asc" } } },
   645→   645→      });
   646→   646→      if (!invoice) continue;
   647→   647→
   648→   648→      const reminderCount = invoice.reminders.filter((r) => r.status === "sent").length;
   649→   649→      if (reminderCount === 0) continue;
   650→   650→
   651→   651→      const patternKey = reminderCount === 1 ? "pays_after_first_reminder" : `pays_after_${reminderCount}_reminders`;
   652→   652→      const patternDescription = `${invoice.customer.name} paid invoice ${invoice.invoiceNumber} after ${reminderCount} reminder(s).`;
   653→   653→
   654→   654→      await upsertPattern({
   655→   655→        workspaceId,
   656→   656→        employeeId,
   657→   657→        patternType: "customer_payment_behavior",
   658→   658→        entityType: "customer",
   659→   659→        entityId: invoice.customerId,
   660→   660→        entityLabel: invoice.customer.name,
   661→   661→        pattern: patternKey,
   662→   662→        description: patternDescription,
   663→   663→        confidence: Math.min(0.99, 0.5 + 0.1 * reminderCount),
   664→   664→      });
   665→   665→    }
   666→   666→  }
   667→   667→
   668→   668→  // ─── Pattern 2: Reminder effectiveness ──────────────────────────────────
   669→   669→  if (evaluation.customerResponded && evaluation.customerIds.length > 0) {
   670→   670→    for (const customerId of evaluation.customerIds) {
   671→   671→      const customer = await db.customer.findUnique({ where: { id: customerId } });
   672→   672→      if (!customer) continue;
   673→   673→
   674→   674→      await upsertPattern({
   675→   675→        workspaceId,
   676→   676→        employeeId,
   677→   677→        patternType: "reminder_effectiveness",
   678→   678→        entityType: "customer",
   679→   679→        entityId: customerId,
   680→   680→        entityLabel: customer.name,
   681→   681→        pattern: "responds_to_reminders",
   682→   682→        description: `${customer.name} responds to reminder emails.`,
   683→   683→        confidence: 0.7,
   684→   684→      });
   685→   685→    }
   686→   686→  }
   687→   687→
   688→   688→  // ─── Pattern 3: Invoice risk indicator ──────────────────────────────────
   689→   689→  if (evaluation.approvalRejections > 0 && evaluation.invoiceIds.length > 0) {
   690→   690→    for (const invoiceId of evaluation.invoiceIds) {
   691→   691→      const invoice = await db.invoice.findUnique({
   692→   692→        where: { id: invoiceId },
   693→   693→        include: { customer: true },
   694→   694→      });
   695→   695→      if (!invoice) continue;
   696→   696→
   697→   697→      // Invoices above ₹5 lakh (5000000 paise = ₹50000... actually ₹5L = 500000*100 = 50000000)
   698→   698→      if (invoice.total >= 5000000) {
   699→   699→        await upsertPattern({
   700→   700→          workspaceId,
   701→   701→          employeeId,
   702→   702→          patternType: "invoice_risk_indicator",
   703→   703→          entityType: "invoice",
   704→   704→          entityId: invoiceId,
   705→   705→          entityLabel: invoice.invoiceNumber,
   706→   706→          pattern: "high_value_invoice_needs_manager_review",
   707→   707→          description: `Invoice ${invoice.invoiceNumber} (₹${(invoice.total / 100).toFixed(2)}) required manager review.`,
   708→   708→          confidence: 0.6,
   709→   709→        });
   710→   710→      }
   711→   711→    }
   712→   712→  }
   713→   713→}
   714→   714→
   715→   715→/**
   716→   716→ * Upserts a pattern: if it exists, increment observationCount and update confidence;
   717→   717→ * otherwise create it. Emits a pattern_learned audit on first detection.
   718→   718→ */
   719→   719→async function upsertPattern(params: {
   720→   720→  workspaceId: string;
   721→   721→  employeeId: string;
   722→   722→  patternType: string;
   723→   723→  entityType: string;
   724→   724→  entityId: string;
   725→   725→  entityLabel: string;
   726→   726→  pattern: string;
   727→   727→  description: string;
   728→   728→  confidence: number;
   729→   729→}): Promise<void> {
   730→   730→  const existing = await db.learningPattern.findUnique({
   731→   731→    where: {
   732→   732→      employeeId_patternType_entityType_entityId_pattern: {
   733→   733→        employeeId: params.employeeId,
   734→   734→        patternType: params.patternType,
   735→   735→        entityType: params.entityType,
   736→   736→        entityId: params.entityId,
   737→   737→        pattern: params.pattern,
   738→   738→      },
   739→   739→    },
   740→   740→  });
   741→   741→
   742→   742→  if (existing) {
   743→   743→    const newCount = existing.observationCount + 1;
   744→   744→    const newConfidence = Math.min(0.99, existing.confidence + 0.05);
   745→   745→    await db.learningPattern.update({
   746→   746→      where: { id: existing.id },
   747→   747→      data: {
   748→   748→        observationCount: newCount,
   749→   749→        confidence: newConfidence,
   750→   750→        lastObservedAt: new Date(),
   751→   751→        description: params.description,
   752→   752→      },
   753→   753→    });
   754→   754→  } else {
   755→   755→    const created = await db.learningPattern.create({
   756→   756→      data: {
   757→   757→        workspaceId: params.workspaceId,
   758→   758→        employeeId: params.employeeId,
   759→   759→        patternType: params.patternType,
   760→   760→        entityType: params.entityType,
   761→   761→        entityId: params.entityId,
   762→   762→        entityLabel: params.entityLabel,
   763→   763→        pattern: params.pattern,
   764→   764→        description: params.description,
   765→   765→        confidence: params.confidence,
   766→   766→        observationCount: 1,
   767→   767→      },
   768→   768→    });
   769→   769→
   770→   770→    await emitAudit(params.workspaceId, params.employeeId, "pattern_learned", "learning_pattern", created.id, {
   771→   771→      patternType: params.patternType,
   772→   772→      pattern: params.pattern,
   773→   773→      entityLabel: params.entityLabel,
   774→   774→      description: params.description,
   775→   775→    });
   776→   776→
   777→   777→    await db.careerTimelineEntry.create({
   778→   778→      data: {
   779→   779→        workspaceId: params.workspaceId,
   780→   780→        employeeId: params.employeeId,
   781→   781→        entryType: "pattern_learned",
   782→   782→        title: `Pattern: ${params.pattern.replace(/_/g, " ")}`,
   783→   783→        description: params.description,
   784→   784→        metadata: JSON.stringify({
   785→   785→          patternType: params.patternType,
   786→   786→          pattern: params.pattern,
   787→   787→          entityType: params.entityType,
   788→   788→          entityId: params.entityId,
   789→   789→        }),
   790→   790→        levelAtTime: 0,
   791→   791→        xpAtTime: 0,
   792→   792→        trustAtTime: 0,
   793→   793→      },
   794→   794→    });
   795→   795→  }
   796→   796→}
   797→   797→
   798→   798→// ─── 4. Weakness Detection ───────────────────────────────────────────────────
   799→   799→
   800→   800→/**
   801→   801→ * Detects weaknesses by scanning the last N outcome evaluations.
   802→   802→ * Called after every task completion (incremental).
   803→   803→ */
   804→   804→async function detectWeaknesses(employeeId: string, workspaceId: string): Promise<void> {
   805→   805→  const recentEvaluations = await db.outcomeEvaluation.findMany({
   806→   806→    where: { employeeId },
   807→   807→    orderBy: { createdAt: "desc" },
   808→   808→    take: 10,
   809→   809→  });
   810→   810→
   811→   811→  if (recentEvaluations.length < 3) return;
   812→   812→
   813→   813→  const totalTasks = recentEvaluations.length;
   814→   814→  const tasksWithRejections = recentEvaluations.filter((e) => e.approvalRejections > 0).length;
   815→   815→  const tasksWithOverrides = recentEvaluations.filter((e) => e.humanOverrides > 0).length;
   816→   816→  const tasksWithSlaMisses = recentEvaluations.filter((e) => !e.slaAchieved).length;
   817→   817→  const tasksWithDenials = recentEvaluations.filter((e) => e.capabilityDenials > 0).length;
   818→   818→  const avgConfidenceAccuracy = recentEvaluations.reduce((s, e) => s + e.confidenceAccuracy, 0) / totalTasks;
   819→   819→  const avgExecutionTime = recentEvaluations.reduce((s, e) => s + e.executionTimeMs, 0) / totalTasks;
   820→   820→  const avgSlaTarget = recentEvaluations[0]?.slaTargetHours || 24;
   821→   821→
   822→   822→  const weaknessChecks = [
   823→   823→    {
   824→   824→      type: "high_rejection_rate",
   825→   825→      description: `${tasksWithRejections}/${totalTasks} recent tasks had approval rejections (${Math.round((tasksWithRejections / totalTasks) * 100)}%)`,
   826→   826→      metricValue: tasksWithRejections / totalTasks,
   827→   827→      threshold: 0.3,
   828→   828→      trustImpact: -5.0,
   829→   829→      inverted: false,
   830→   830→    },
   831→   831→    {
   832→   832→      type: "frequent_human_edits",
   833→   833→      description: `${tasksWithOverrides}/${totalTasks} recent tasks had human overrides (${Math.round((tasksWithOverrides / totalTasks) * 100)}%)`,
   834→   834→      metricValue: tasksWithOverrides / totalTasks,
   835→   835→      threshold: 0.4,
   836→   836→      trustImpact: -3.0,
   837→   837→      inverted: false,
   838→   838→    },
   839→   839→    {
   840→   840→      type: "repeated_sla_misses",
   841→   841→      description: `${tasksWithSlaMisses}/${totalTasks} recent tasks missed SLA (${Math.round((tasksWithSlaMisses / totalTasks) * 100)}%)`,
   842→   842→      metricValue: tasksWithSlaMisses / totalTasks,
   843→   843→      threshold: 0.3,
   844→   844→      trustImpact: -4.0,
   845→   845→      inverted: false,
   846→   846→    },
   847→   847→    {
   848→   848→      type: "low_confidence_accuracy",
   849→   849→      description: `Average confidence accuracy is ${(avgConfidenceAccuracy * 100).toFixed(0)}% (threshold 50%)`,
   850→   850→      metricValue: avgConfidenceAccuracy,
   851→   851→      threshold: 0.5,
   852→   852→      trustImpact: -3.0,
   853→   853→      inverted: true,
   854→   854→    },
   855→   855→    {
   856→   856→      type: "slow_execution",
   857→   857→      description: `Average execution time is ${(avgExecutionTime / (1000 * 60 * 60)).toFixed(1)}h (threshold ${avgSlaTarget * 2}h)`,
   858→   858→      metricValue: avgExecutionTime / (1000 * 60 * 60),
   859→   859→      threshold: avgSlaTarget * 2,
   860→   860→      trustImpact: -2.0,
   861→   861→      inverted: false, // higher execution time = worse
   862→   862→    },
   863→   863→  ];
   864→   864→
   865→   865→  if (tasksWithDenials > 1) {
   866→   866→    weaknessChecks.push({
   867→   867→      type: "high_capability_denials",
   868→   868→      description: `${tasksWithDenials} capability denial(s) in recent ${totalTasks} tasks`,
   869→   869→      metricValue: tasksWithDenials,
   870→   870→      threshold: 1,
   871→   871→      trustImpact: -6.0,
   872→   872→      inverted: false, // higher denials = worse
   873→   873→    });
   874→   874→  }
   875→   875→
   876→   876→  for (const check of weaknessChecks) {
   877→   877→    const triggered = check.inverted
   878→   878→      ? check.metricValue < check.threshold
   879→   879→      : check.metricValue > check.threshold;
   880→   880→
   881→   881→    if (triggered) {
   882→   882→      await upsertWeakness({
   883→   883→        workspaceId,
   884→   884→        employeeId,
   885→   885→        weaknessType: check.type,
   886→   886→        description: check.description,
   887→   887→        metricValue: check.metricValue,
   888→   888→        threshold: check.threshold,

  1345→  1345→/**
  1346→  1346→ * Checks for achievement unlocks based on the employee's current state.
  1347→  1347→ */
  1348→  1348→async function checkAchievementUnlocks(
  1349→  1349→  employeeId: string,
  1350→  1350→  workspaceId: string,
  1351→  1351→  evaluation: any
  1352→  1352→): Promise<void> {
  1353→  1353→  await seedDefaultAchievements(workspaceId);
  1354→  1354→
  1355→  1355→  const profile = await db.employeeProfile.findUnique({ where: { employeeId } });
  1356→  1356→  if (!profile) return;
  1357→  1357→
  1358→  1358→  const [achievements, unlocked] = await Promise.all([
  1359→  1359→    db.achievement.findMany({ where: { workspaceId } }),
  1360→  1360→    db.employeeAchievement.findMany({ where: { employeeId }, select: { achievementId: true } }),
  1361→  1361→  ]);
  1362→  1362→  const unlockedIds = new Set(unlocked.map((u) => u.achievementId));
  1363→  1363→
  1364→  1364→  for (const achievement of achievements) {
  1365→  1365→    if (unlockedIds.has(achievement.id)) continue;
  1366→  1366→
  1367→  1367→    let shouldUnlock = false;
  1368→  1368→    let evidence = "";
  1369→  1369→
  1370→  1370→    switch (achievement.triggerType) {
  1371→  1371→      case "task_count":
  1372→  1372→        if (profile.completedTasks >= achievement.triggerThreshold) {
  1373→  1373→          shouldUnlock = true;
  1374→  1374→          evidence = `Completed ${profile.completedTasks} tasks (threshold: ${achievement.triggerThreshold})`;
  1375→  1375→        }
  1376→  1376→        break;
  1377→  1377→      case "money_recovered":
  1378→  1378→        if (profile.moneyRecovered >= achievement.triggerThreshold) {
  1379→  1379→          shouldUnlock = true;
  1380→  1380→          evidence = `Recovered ₹${(profile.moneyRecovered / 100).toFixed(2)} (threshold: ₹${(achievement.triggerThreshold / 100).toFixed(2)})`;
  1381→  1381→        }
  1382→  1382→        break;
  1383→  1383→      case "level_reached":
  1384→  1384→        if (profile.level >= achievement.triggerThreshold) {
  1385→  1385→          shouldUnlock = true;
  1386→  1386→          evidence = `Reached Level ${profile.level} (threshold: Level ${achievement.triggerThreshold})`;
  1387→  1387→        }
  1388→  1388→        break;
  1389→  1389→      case "skill_level":
  1390→  1390→        if (achievement.triggerSkillName) {
  1391→  1391→          const skill = await db.employeeSkill.findUnique({
  1392→  1392→            where: { employeeId_name: { employeeId, name: achievement.triggerSkillName } },
  1393→  1393→          });
  1394→  1394→          if (skill && skill.level >= achievement.triggerThreshold) {
  1395→  1395→            shouldUnlock = true;
  1396→  1396→            evidence = `Skill "${achievement.triggerSkillName}" reached Level ${skill.level} (threshold: Level ${achievement.triggerThreshold})`;
  1397→  1397→          }
  1398→  1398→        }
  1399→  1399→        break;
  1400→  1400→      case "perfect_quality": {
  1401→  1401→        const perfectCount = await db.outcomeEvaluation.count({
  1402→  1402→          where: { employeeId, qualityScore: 100 },
  1403→  1403→        });
  1404→  1404→        if (perfectCount >= achievement.triggerThreshold) {
  1405→  1405→          shouldUnlock = true;
  1406→  1406→          evidence = `${perfectCount} tasks with perfect quality score (threshold: ${achievement.triggerThreshold})`;
  1407→  1407→        }
  1408→  1408→        break;
  1409→  1409→      }
  1410→  1410→      case "streak": {
  1411→  1411→        const recentEvals = await db.outcomeEvaluation.findMany({
  1412→  1412→          where: { employeeId },
  1413→  1413→          orderBy: { createdAt: "desc" },
  1414→  1414→          take: achievement.triggerThreshold,
  1415→  1415→        });
  1416→  1416→        const allSuccess = recentEvals.length >= achievement.triggerThreshold &&
  1417→  1417→          recentEvals.every((e) => e.actualSuccess);
  1418→  1418→        if (allSuccess) {
  1419→  1419→          shouldUnlock = true;
  1420→  1420→          evidence = `${achievement.triggerThreshold} consecutive successful tasks`;
  1421→  1421→        }
  1422→  1422→        break;
  1423→  1423→      }
  1424→  1424→    }
  1425→  1425→
  1426→  1426→    if (shouldUnlock) {
  1427→  1427→      await db.employeeAchievement.create({
  1428→  1428→        data: {
  1429→  1429→          workspaceId,
  1430→  1430→          employeeId,
  1431→  1431→          achievementId: achievement.id,
  1432→  1432→          evidence,
  1433→  1433→          taskId: evaluation.taskId,
  1434→  1434→          outcomeId: evaluation.id,
  1435→  1435→        },
  1436→  1436→      });
  1437→  1437→
  1438→  1438→      await emitAudit(workspaceId, employeeId, "achievement_unlocked", "achievement", achievement.id, {
  1439→  1439→        code: achievement.code,
  1440→  1440→        name: achievement.name,
  1441→  1441→        description: achievement.description,
  1442→  1442→        trustImpact: String(achievement.trustImpact),
  1443→  1443→        evidence,
  1444→  1444→      });
  1445→  1445→
  1446→  1446→      await db.careerTimelineEntry.create({
  1447→  1447→        data: {
  1448→  1448→          workspaceId,
  1449→  1449→          employeeId,
  1450→  1450→          entryType: "achievement_unlocked",
  1451→  1451→          title: `Achievement: ${achievement.name}`,
  1452→  1452→          description: `${achievement.description} — ${evidence}`,
  1453→  1453→          metadata: JSON.stringify({
  1454→  1454→            code: achievement.code,
  1455→  1455→            name: achievement.name,
  1456→  1456→            trustImpact: achievement.trustImpact,
  1457→  1457→            evidence,
  1458→  1458→          }),
  1459→  1459→          levelAtTime: profile.level,
  1460→  1460→          xpAtTime: profile.experiencePoints,
  1461→  1461→          trustAtTime: profile.trustScore,
  1462→  1462→          taskId: evaluation.taskId,
  1463→  1463→          outcomeId: evaluation.id,
  1464→  1464→        },
  1465→  1465→      });
  1466→  1466→    }
  1467→  1467→  }
  1468→  1468→}
