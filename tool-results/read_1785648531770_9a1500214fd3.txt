     1→/**
     2→ * BIHARI AI — AI Employee Communication Engine (COMM-001)
     3→ *
     4→ * The universal messaging layer for every AI Employee in BIHARI AI.
     5→ * Allows AI Employees to communicate with humans, other AI Employees,
     6→ * and future external systems using structured business communication.
     7→ *
     8→ * This is NOT a chat system. It is structured, explainable, auditable
     9→ * business communication.
    10→ *
    11→ * Design principles (NON-NEGOTIABLE):
    12→ *   1. Humans never receive raw AI reasoning — the engine transforms
    13→ *      internal reasoning into professional business language.
    14→ *   2. Every communication is explainable (why, evidence, confidence, impact).
    15→ *   3. Threaded conversations with replies, acknowledgements, resolution.
    16→ *   4. Deterministic scoring to avoid spam, merge similar alerts, throttle.
    17→ *   5. Every communication becomes permanent memory (Learning Engine).
    18→ *   6. Every communication generates audit events (Audit Chain).
    19→ *   7. Generic and domain-independent — works for every employee type.
    20→ *
    21→ * Architecture:
    22→ *   The engine composes with — does NOT duplicate — the existing systems:
    23→ *   - Audit Chain: every communication generates audit events
    24→ *   - Memory Engine: communication outcomes become permanent memory
    25→ *   - Profile Engine: communication events update the employee profile
    26→ *   - Learning Engine: communication effectiveness is scored over time
    27→ *   - Capability Engine: not gated (communication is a universal right)
    28→ *   - Approval Engine: approval_request type creates a linked Approval
    29→ *
    30→ * The engine is called from:
    31→ *   - The Executor (when an employee needs to communicate during a task)
    32→ *   - The Approval API (when a human responds to a communication)
    33→ *   - Employee-to-employee coordination (when employees hand off work)
    34→ *   - Future external system adapters
    35→ */

   112→// ─── Throttling + Deduplication Config ───────────────────────────────────────
   113→
   114→// Throttle window: if the same sender sends the same communicationType +
   115→// subject to the same receiver within this window (ms), it's throttled.
   116→const THROTTLE_WINDOW_MS = 5 * 60 * 1000; // 5 minutes
   117→
   118→// Dedup window: if a message with the same (sender, receiver, communicationType,
   119→// subject) exists within this window and is NOT resolved, the new one is a duplicate.
   120→const DEDUP_WINDOW_MS = 30 * 60 * 1000; // 30 minutes
   121→
   122→// ─── 1. Create Communication ─────────────────────────────────────────────────
   123→
   124→/**
   125→ * Creates a structured business communication.
   126→ *
   127→ * This is the SINGLE entry point for all AI Employee communication.
   128→ * The function:
   129→ *   1. Transforms the input into professional business language (already done by caller)
   130→ *   2. Checks for duplicates (same sender + receiver + type + subject in the dedup window)
   131→ *   3. Checks for throttling (same sender + receiver + type + subject in the throttle window)
   132→ *   4. Computes a deterministic quality score (0-100)
   133→ *   5. Creates or appends to a thread
   134→ *   6. Writes the communication record
   135→ *   7. Emits an audit event
   136→ *   8. Records a memory (so the employee remembers sending it)
   137→ *
   138→ * Returns the created communication, or the original if this is a duplicate.
   139→ */
   140→export async function createCommunication(input: CommunicationInput): Promise<{
   141→  id: string;
   142→  isDuplicate: boolean;
   143→  duplicateOfId?: string;
   144→  isThrottled: boolean;
   145→  qualityScore: number;
   146→}> {
   147→  const priority = input.priority || "medium";
   148→  const confidence = input.confidence ?? 0.85;
   149→
   150→  // ─── Step 1: Check for duplicates ────────────────────────────────────────
   151→  const dedupSince = new Date(Date.now() - DEDUP_WINDOW_MS);
   152→  const existing = await db.employeeCommunication.findFirst({
   153→    where: {
   154→      workspaceId: input.workspaceId,
   155→      senderEmployeeId: input.senderEmployeeId || null,
   156→      senderUserId: input.senderUserId || null,
   157→      receiverType: input.receiverType,
   158→      receiverEmployeeId: input.receiverEmployeeId || null,
   159→      receiverUserId: input.receiverUserId || null,
   160→      communicationType: input.communicationType,
   161→      subject: input.subject,
   162→      status: { notIn: ["resolved", "ignored"] },
   163→      createdAt: { gte: dedupSince },
   164→    },
   165→    orderBy: { createdAt: "desc" },
   166→  });
   167→
   168→  if (existing) {
   169→    // This is a duplicate — don't create a new message, just return the original
   170→    return {
   171→      id: existing.id,
   172→      isDuplicate: true,
   173→      duplicateOfId: existing.id,
   174→      isThrottled: false,
   175→      qualityScore: existing.qualityScore,
   176→    };
   177→  }
   178→
   179→  // ─── Step 2: Check for throttling ────────────────────────────────────────
   180→  const throttleSince = new Date(Date.now() - THROTTLE_WINDOW_MS);
   181→  const throttled = await db.employeeCommunication.findFirst({
   182→    where: {
   183→      workspaceId: input.workspaceId,
   184→      senderEmployeeId: input.senderEmployeeId || null,
   185→      senderUserId: input.senderUserId || null,
   186→      receiverType: input.receiverType,
   187→      receiverEmployeeId: input.receiverEmployeeId || null,
   188→      receiverUserId: input.receiverUserId || null,
   189→      communicationType: input.communicationType,
   190→      subject: input.subject,
   191→      createdAt: { gte: throttleSince },
   192→    },
   193→  });
   194→
   195→  const isThrottled = !!throttled;
   196→
   197→  // ─── Step 3: Compute deterministic quality score (0-100) ─────────────────
   198→  const qualityScore = computeQualityScore(input, confidence);
   199→
   200→  // ─── Step 4: Create or append to thread ──────────────────────────────────
   201→  let threadId = input.threadId;
   202→  if (!threadId && !input.parentCommunicationId) {
   203→    // Start a new thread for top-level messages
   204→    const thread = await db.communicationThread.create({
   205→      data: {
   206→        workspaceId: input.workspaceId,
   207→        subject: input.subject,
   208→        startedByEmployeeId: input.senderEmployeeId || null,
   209→        startedByUserId: input.senderUserId || null,
   210→        relatedTaskId: input.relatedTaskId || null,
   211→        relatedCustomerId: input.relatedCustomerId || null,
   212→        relatedInvoiceId: input.relatedInvoiceId || null,
   213→        status: "active",
   214→        priority,
   215→        communicationType: input.communicationType,
   216→        messageCount: 1,
   217→        participantCount: 1,
   218→        lastMessageAt: new Date(),
   219→      },
   220→    });
   221→    threadId = thread.id;
   222→  } else if (threadId) {
   223→    // Append to existing thread — update counts + lastMessageAt
   224→    await db.communicationThread.update({
   225→      where: { id: threadId },
   226→      data: {
   227→        messageCount: { increment: 1 },
   228→        lastMessageAt: new Date(),
   229→        // Escalate thread priority if this message is higher
   230→        priority: escalatePriority(priority, (await db.communicationThread.findUnique({ where: { id: threadId } }))?.priority || "medium"),
   231→      },
   232→    });
   233→  }
   234→
   235→  // ─── Step 5: Create the communication record ─────────────────────────────
   236→  const communication = await db.employeeCommunication.create({
   237→    data: {
   238→      workspaceId: input.workspaceId,
   239→      threadId: threadId || null,
   240→      senderEmployeeId: input.senderEmployeeId || null,
   241→      senderUserId: input.senderUserId || null,
   242→      senderName: input.senderName,
   243→      senderType: input.senderType,
   244→      receiverType: input.receiverType,
   245→      receiverEmployeeId: input.receiverEmployeeId || null,
   246→      receiverUserId: input.receiverUserId || null,
   247→      receiverName: input.receiverName,
   248→      communicationType: input.communicationType,
   249→      priority,
   250→      subject: input.subject,
   251→      summary: input.summary,
   252→      explanation: input.explanation,
   253→      relatedTaskId: input.relatedTaskId || null,
   254→      relatedContractId: input.relatedContractId || null,
   255→      relatedCustomerId: input.relatedCustomerId || null,
   256→      relatedInvoiceId: input.relatedInvoiceId || null,
   257→      relatedApprovalId: input.relatedApprovalId || null,
   258→      relatedReminderId: input.relatedReminderId || null,
   259→      whyExists: input.whyExists,
   260→      evidence: JSON.stringify(input.evidence || []),
   261→      confidence,
   262→      businessImpact: input.businessImpact,
   263→      recommendedAction: input.recommendedAction,
   264→      expectedOutcome: input.expectedOutcome,
   265→      attachments: JSON.stringify(input.attachments || []),
   266→      actionButtons: JSON.stringify(input.actionButtons || []),
   267→      status: isThrottled ? "sent" : "sent", // throttled messages are still sent but flagged
   268→      qualityScore,
   269→      isDuplicate: false,
   270→      isThrottled,
   271→    },
   272→  });
   273→
   274→  // ─── Step 6: Emit audit event ────────────────────────────────────────────
   275→  try {
   276→    await db.$transaction(async (tx) => {
   277→      await appendAudit(tx, {
   278→        workspaceId: input.workspaceId,
   279→        entryType: "communication_created",
   280→        actorType: input.senderType as "user" | "employee" | "system",
   281→        actorId: input.senderEmployeeId || input.senderUserId || null,
   282→        actorName: input.senderName,
   283→        targetType: "communication",
   284→        targetId: communication.id,
   285→        payload: {
   286→          communicationType: input.communicationType,
   287→          priority,
   288→          subject: input.subject,
   289→          receiverType: input.receiverType,
   290→          receiverName: input.receiverName,
   291→          threadId: threadId || "",
   292→          qualityScore: String(qualityScore),
   293→          isThrottled: String(isThrottled),
   294→        },
   295→      });
   296→    });
   297→  } catch (err) {
   298→    console.error("[Communication] Audit emission failed:", err);
   299→  }
   300→
   301→  // ─── Step 7: Record memory (so the employee remembers sending it) ────────
   302→  if (input.senderEmployeeId) {
   303→    try {
   304→      const { recordMemory } = await import("@/lib/memory/service");
   305→      await recordMemory(input.senderEmployeeId, input.workspaceId, {
   306→        memoryType: "communication_preference",
   307→        entityType: "communication",
   308→        entityId: communication.id,
   309→        entityLabel: input.subject,
   310→        key: `${input.communicationType}_${input.receiverType}`,
   311→        value: {
   312→          subject: input.subject,
   313→          communicationType: input.communicationType,
   314→          priority,
   315→          receiverType: input.receiverType,
   316→          receiverName: input.receiverName,
   317→          qualityScore: String(qualityScore),
   318→          sentAt: new Date().toISOString(),
   319→        },
   320→        source: "communication_engine",
   321→      });
   322→    } catch (err) {
   323→      console.error("[Communication] Memory recording failed:", err);
   324→    }
   325→  }
   326→
   327→  return {
   328→    id: communication.id,
   329→    isDuplicate: false,
   330→    isThrottled,
   331→    qualityScore,
   332→  };
   333→}
   334→
   335→// ─── 2. Quality Score Computation (Deterministic) ────────────────────────────
   336→
   337→/**
   338→ * Computes a deterministic quality score (0-100) for a communication.
   339→ *
   340→ * Components (each contributes to a 0-100 score):
   341→ *   30% — Has all explainability fields (whyExists, evidence, businessImpact, recommendedAction, expectedOutcome)
   342→ *   20% — Confidence (higher = better)
   343→ *   15% — Priority alignment (critical/warning types should have high priority)
   344→ *   15% — Has business context (related entities)
   345→ *   10% — Has action buttons (actionable)
   346→ *   10% — Subject + summary quality (length check)
   347→ */
   348→function computeQualityScore(input: CommunicationInput, confidence: number): number {
   349→  let score = 0;
   350→
   351→  // Explainability completeness (30%)
   352→  if (input.whyExists && input.whyExists.length > 10) score += 6;
   353→  if (input.evidence && input.evidence.length > 0) score += 6;
   354→  if (input.businessImpact && input.businessImpact.length > 5) score += 6;
   355→  if (input.recommendedAction && input.recommendedAction.length > 5) score += 6;
   356→  if (input.expectedOutcome && input.expectedOutcome.length > 5) score += 6;
   357→
   358→  // Confidence (20%)
   359→  score += Math.round(confidence * 20);
   360→
   361→  // Priority alignment (15%)
   362→  const highPriorityTypes = ["escalation", "critical_alert", "warning", "approval_request"];
   363→  const isHighPriorityType = highPriorityTypes.includes(input.communicationType);
   364→  const priority = input.priority || "medium";
   365→  if (isHighPriorityType && (priority === "high" || priority === "critical")) score += 15;
   366→  else if (!isHighPriorityType && (priority === "low" || priority === "medium")) score += 15;
   367→  else if (isHighPriorityType || priority === "high" || priority === "critical") score += 8;
   368→
   369→  // Business context (15%)
   370→  const hasContext = input.relatedTaskId || input.relatedCustomerId || input.relatedInvoiceId ||
   371→    input.relatedContractId || input.relatedApprovalId;
   372→  if (hasContext) score += 15;
   373→
   374→  // Actionable (10%)
   375→  if (input.actionButtons && input.actionButtons.length > 0) score += 10;
   376→
   377→  // Subject + summary quality (10%)
   378→  if (input.subject && input.subject.length >= 10 && input.subject.length <= 120) score += 5;
   379→  if (input.summary && input.summary.length >= 20 && input.summary.length <= 300) score += 5;
   380→
   381→  return Math.min(100, score);
   382→}

   386→/**
   387→ * Replies to an existing communication. Creates a new message in the same thread.
   388→ */
   389→export async function replyToCommunication(
   390→  communicationId: string,
   391→  replyInput: Omit<CommunicationInput, "threadId" | "parentCommunicationId">
   392→): Promise<{ id: string; isDuplicate: boolean; isThrottled: boolean; qualityScore: number }> {
   393→  const original = await db.employeeCommunication.findUnique({
   394→    where: { id: communicationId },
   395→    select: { threadId: true, workspaceId: true },
   396→  });
   397→
   398→  if (!original) {
   399→    throw new Error("Original communication not found");
   400→  }
   401→
   402→  return createCommunication({
   403→    ...replyInput,
   404→    workspaceId: original.workspaceId,
   405→    threadId: original.threadId || undefined,
   406→    parentCommunicationId: communicationId,
   407→  });
   408→}

   472→/**
   473→ * Resolves a communication (the issue is handled, no further action needed).

   475→export async function resolveCommunication(
   476→  communicationId: string,
   477→  resolvedBy: string,
   478→  resolvedByName: string,
   479→  resolutionNote?: string
   480→): Promise<void> {
   481→  const comm = await db.employeeCommunication.findUnique({
   482→    where: { id: communicationId },
   483→    select: { workspaceId: true, threadId: true, senderEmployeeId: true, createdAt: true },
   484→  });
   485→  if (!comm) return;
   486→
   487→  const now = new Date();
   488→  const responseTimeMs = comm.createdAt ? now.getTime() - new Date(comm.createdAt).getTime() : null;
   489→
   490→  await db.employeeCommunication.update({
   491→    where: { id: communicationId },
   492→    data: {
   493→      status: "resolved",
   494→      resolvedAt: now,
   495→      responseTimeMs,
   496→      responseAction: "resolve",
   497→    },
   498→  });
   499→
   500→  // If this message is in a thread, resolve the thread too
   501→  if (comm.threadId) {
   502→    await db.communicationThread.update({
   503→      where: { id: comm.threadId },
   504→      data: {
   505→        status: "resolved",
   506→        resolvedAt: now,
   507→      },
   508→    }).catch(() => {});
   509→  }
   510→
   511→  await emitAudit(comm.workspaceId, "communication_resolved", "communication", communicationId, {
   512→    resolvedBy,
   513→    resolvedByName,
   514→    resolutionNote: resolutionNote || "",
   515→    responseTimeMs: String(responseTimeMs || 0),
   516→  });
   517→
   518→  // Record memory for the sender employee
   519→  if (comm.senderEmployeeId) {
   520→    await recordCommunicationMemory(comm.senderEmployeeId, comm.workspaceId, communicationId, "resolved", responseTimeMs);
   521→  }
   522→}

   556→        status: "escalated",

   562→  await emitAudit(comm.workspaceId, "communication_escalated", "communication", communicationId, {

   566→    newPriority,
   567→  });
   568→
   569→  // Record memory
   570→  if (comm.senderEmployeeId) {
   571→    await recordCommunicationMemory(comm.senderEmployeeId, comm.workspaceId, communicationId, "escalated", null);
   572→  }
   573→}

   575→// ─── 5. Employee-to-Employee Communication ───────────────────────────────────

   577→/**

   589→export async function sendEmployeeCoordination(
   590→  fromEmployeeId: string,
   591→  toEmployeeId: string,
   592→  workspaceId: string,
   593→  params: {
   594→    subject: string;
   595→    summary: string;

   646→/**
   647→ * Records a communication outcome as permanent memory.
   648→ * This enables the Learning Engine to track:
   649→ *   - Communication effectiveness (was it acknowledged/resolved/ignored?)
   650→ *   - Response times (how long did the receiver take?)
   651→ *   - Human acceptance (approved vs rejected)
   652→ *   - False alerts (ignored messages)
   653→ *   - Successful escalations
   654→ */
   655→async function recordCommunicationMemory(
   656→  employeeId: string,
   657→  workspaceId: string,
   658→  communicationId: string,
   659→  outcome: string,
   660→  responseTimeMs: number | null
   661→): Promise<void> {
   662→  try {
   663→    const { recordMemory } = await import("@/lib/memory/service");
   664→    await recordMemory(employeeId, workspaceId, {
   665→      memoryType: "strategy_effectiveness",
   666→      entityType: "communication",
   667→      entityId: communicationId,
   668→      entityLabel: `Communication ${outcome}`,
   669→      key: `comm_outcome_${outcome}`,
   670→      value: {
   671→        outcome,
   672→        responseTimeMs: String(responseTimeMs || 0),
   673→        recordedAt: new Date().toISOString(),
   674→      },
   675→      source: "communication_engine",
   676→    });
   677→  } catch (err) {
   678→    console.error("[Communication] Memory recording failed:", err);
   679→  }

   697→export async function listCommunications(workspaceId: string, query: CommunicationQuery = {}) {
   698→  const where: any = { workspaceId };
   699→
   700→  if (query.status && query.status !== "all") {
   701→    where.status = query.status;
   702→  }
   703→  if (query.priority && query.priority !== "all") {
   704→    where.priority = query.priority;
   705→  }
   706→  if (query.communicationType && query.communicationType !== "all") {
   707→    where.communicationType = query.communicationType;
   708→  }
   709→  if (query.receiverType && query.receiverType !== "all") {
   710→    where.receiverType = query.receiverType;
   711→  }
   712→  if (query.employeeId) {
   713→    where.OR = [
   714→      { senderEmployeeId: query.employeeId },
   715→      { receiverEmployeeId: query.employeeId },
   716→    ];
   717→  }
   718→  if (query.customerId) {
   719→    where.relatedCustomerId = query.customerId;
   720→  }
   721→  if (query.taskId) {
   722→    where.relatedTaskId = query.taskId;
   723→  }
   724→  if (query.invoiceId) {
   725→    where.relatedInvoiceId = query.invoiceId;
   726→  }
   727→  if (query.search) {
   728→    where.OR = [
   729→      { subject: { contains: query.search } },
   730→      { summary: { contains: query.search } },
   731→      { explanation: { contains: query.search } },
   732→    ];
   733→  }
   734→
   735→  return db.employeeCommunication.findMany({
   736→    where,
   737→    orderBy: { createdAt: "desc" },
   738→    take: Math.min(100, query.limit || 50),
   739→  });
   740→}

   763→export async function getCommunicationStats(workspaceId: string) {
   764→  const [total, unread, critical, waiting, resolved, escalated, byType, byPriority] = await Promise.all([
   765→    db.employeeCommunication.count({ where: { workspaceId } }),
   766→    db.employeeCommunication.count({ where: { workspaceId, status: { in: ["sent", "delivered"] } } }),
   767→    db.employeeCommunication.count({ where: { workspaceId, priority: "critical", status: { notIn: ["resolved", "ignored"] } } }),
   768→    db.employeeCommunication.count({ where: { workspaceId, status: { in: ["acknowledged"] } } }),
   769→    db.employeeCommunication.count({ where: { workspaceId, status: "resolved" } }),
   770→    db.employeeCommunication.count({ where: { workspaceId, status: "escalated" } }),
   771→    db.employeeCommunication.groupBy({ by: ["communicationType"], where: { workspaceId }, _count: true }),
   772→    db.employeeCommunication.groupBy({ by: ["priority"], where: { workspaceId }, _count: true }),
   773→  ]);
   774→
   775→  // Compute average response time for resolved communications
   776→  const resolvedComms = await db.employeeCommunication.findMany({
   777→    where: { workspaceId, status: "resolved", responseTimeMs: { not: null } },
   778→    select: { responseTimeMs: true },
   779→    take: 100,
   780→  });
   781→  const avgResponseTimeMs = resolvedComms.length > 0
   782→    ? resolvedComms.reduce((s, c) => s + (c.responseTimeMs || 0), 0) / resolvedComms.length
   783→    : null;
   784→
   785→  return {
   786→    total,
   787→    unread,
   788→    critical,
   789→    waiting,
   790→    resolved,
   791→    escalated,
   792→    avgResponseTimeMs,
   793→    byType: byType.reduce((acc, b) => ({ ...acc, [b.communicationType]: b._count }), {}),
   794→    byPriority: byPriority.reduce((acc, b) => ({ ...acc, [b.priority]: b._count }), {}),
   795→  };
   796→}

   800→function escalatePriority(newPriority: string, currentPriority: string): string {
   801→  const order = { low: 0, medium: 1, high: 2, critical: 3 };
   802→  const newLevel = (order as any)[newPriority] || 1;
   803→  const currentLevel = (order as any)[currentPriority] || 1;
   804→  return newLevel > currentLevel ? newPriority : currentPriority;
   805→}
