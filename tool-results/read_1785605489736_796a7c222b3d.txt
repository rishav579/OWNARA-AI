     3→     3→ *
     4→     4→ * Every AI Employee has one persistent profile representing its professional career.
     5→     5→ * Think: LinkedIn profile + performance review + resume + trust history.
     6→     6→ *

     8→     8→ * - Task completion (XP, KPIs, skills, trust)

    17→    17→ *
    18→    18→ * Architecture:
    19→    19→ * - EmployeeProfile: the persistent career record (one per employee)
    20→    20→ * - EmployeeSkill: individual skills with level, confidence, usage count

    74→    74→  memory_created: 1,

    95→    95→  let result = LEVELS[0];
    96→    96→  for (const l of LEVELS) {
    97→    97→    if (xp >= l.minXp) result = l;
    98→    98→  }
    99→    99→  return { level: result.level, title: result.title };
   100→   100→}

   167→   167→// ─── Core: Record Profile Event ─────────────────────────────────────────────
   168→   168→
   169→   169→/**
   170→   170→ * Records a profile event and updates the employee's profile accordingly.
   171→   171→ *
   172→   172→ * This is the SINGLE entry point for all profile updates. Every system
   173→   173→ * (executor, approval, memory, contracts, capabilities) calls this function.
   174→   174→ *
   175→   175→ * The function:
   176→   176→ * 1. Awards/penalizes XP based on the event type
   177→   177→ * 2. Updates the level if XP crossed a threshold
   178→   178→ * 3. Updates KPIs (tasks, emails, money, etc.)
   179→   179→ * 4. Updates trust score (composite)
   180→   180→ * 5. Updates quality metrics
   181→   181→ * 6. Infers and updates skills
   182→   182→ * 7. Updates memory/capability counts
   183→   183→ */
   184→   184→export async function recordProfileEvent(event: ProfileUpdateEvent): Promise<void> {
   185→   185→  const profile = await db.employeeProfile.findUnique({
   186→   186→    where: { employeeId: event.employeeId },
   187→   187→  });
   188→   188→
   189→   189→  if (!profile) {
   190→   190→    // Auto-initialize if profile doesn't exist
   191→   191→    const employee = await db.employee.findUnique({
   192→   192→      where: { id: event.employeeId },
   193→   193→    });
   194→   194→    if (!employee) return;
   195→   195→
   196→   196→    await initProfile(
   197→   197→      event.employeeId,
   198→   198→      event.workspaceId,
   199→   199→      employee.role,
   200→   200→      employee.role === "finance_employee" ? "Finance" : "General"
   201→   201→    );
   202→   202→
   203→   203→    return recordProfileEvent(event); // Retry after init
   204→   204→  }
   205→   205→
   206→   206→  // ─── Calculate XP delta ──────────────────────────────────────────────────
   207→   207→  let xpDelta = XP_REWARDS[event.type] || 0;
   208→   208→
   209→   209→  // Money recovered bonus: +15 base + 1 per ₹1000 recovered
   210→   210→  if (event.type === "money_recovered" && event.amount) {
   211→   211→    xpDelta = 15 + Math.floor(event.amount / 100000); // 100000 paise = ₹1000
   212→   212→  }
   213→   213→
   214→   214→  const newXp = Math.max(0, profile.experiencePoints + xpDelta);
   215→   215→  const { level, title } = getLevelForXp(newXp);
   216→   216→
   217→   217→  // ─── Calculate field updates ─────────────────────────────────────────────
   218→   218→  const updates: Record<string, any> = {
   219→   219→    experiencePoints: newXp,
   220→   220→    level,
   221→   221→    title,
   222→   222→    version: { increment: 1 },
   223→   223→    updatedAt: new Date(),
   224→   224→  };
   225→   225→
   226→   226→  // Task counters
   227→   227→  if (event.type === "task_completed") {
   228→   228→    updates.completedTasks = { increment: 1 };
   229→   229→    updates.successfulTasks = { increment: 1 };
   230→   230→    updates.tasksAutomated = { increment: 1 };
   231→   231→    updates.lastTaskAt = new Date();
   232→   232→    updates.hoursSaved = { increment: 0.5 }; // ~30 min per automated task
   233→   233→    if (event.executionTimeMs) {
   234→   234→      const newAvg = Math.round(
   235→   235→        (profile.averageExecutionTime * profile.completedTasks + event.executionTimeMs) /
   236→   236→        (profile.completedTasks + 1)
   237→   237→      );
   238→   238→      updates.averageExecutionTime = newAvg;
   239→   239→    }
   240→   240→    if (event.confidence) {
   241→   241→      const newAvgConf = (profile.averageConfidence * profile.completedTasks + event.confidence) /
   242→   242→        (profile.completedTasks + 1);
   243→   243→      updates.averageConfidence = newAvgConf;
   244→   244→    }
   245→   245→  }
   246→   246→
   247→   247→  if (event.type === "task_failed") {
   248→   248→    updates.failedTasks = { increment: 1 };
   249→   249→  }
   250→   250→
   251→   251→  // Approval rate — only recomputed on the actual approval decision events.
   252→   252→  // contract_approved/contract_rejected are bookkeeping events that co-occur
   253→   253→  // with approval_approved/approval_rejected; including them here would
   254→   254→  // recompute the same rate twice (harmless but wasteful and semantically wrong).
   255→   255→  if (event.type === "approval_approved") {
   256→   256→    const totalDecisions = profile.successfulTasks + profile.failedTasks + 1;
   257→   257→    const newApprovalRate = (profile.successfulTasks + 1) / totalDecisions;
   258→   258→    updates.approvalRate = Math.min(1.0, newApprovalRate);
   259→   259→  }
   260→   260→
   261→   261→  if (event.type === "approval_rejected") {
   262→   262→    const totalDecisions = profile.successfulTasks + profile.failedTasks + 1;
   263→   263→    const newApprovalRate = profile.successfulTasks / totalDecisions;
   264→   264→    updates.approvalRate = Math.max(0.0, newApprovalRate);
   265→   265→  }
   266→   266→
   267→   267→  // KPIs — emailsSent, customersHandled, and invoicesProcessed are all synced
   268→   268→  // from the Reminder table in updateMemoryAndCapabilityCounts (count of
   269→   269→  // sent reminders, distinct customers, distinct invoices). They are NOT
   270→   270→  // incremented here because:
   271→   271→  // 1. The reminder_sent event fires when the approval is approved, BEFORE
   272→   272→  //    the worker actually executes the send_reminder tool. If the task is
   273→   273→  //    then cancelled or the tool fails, the counter would be wrong.
   274→   274→  // 2. The approval_approved event with toolName==="send_reminder" used to
   275→   275→  //    ALSO increment emailsSent, causing a double-count (2× per reminder).
   276→   276→  // DB-synced counters are the source of truth.
   277→   277→
   278→   278→  if (event.type === "money_recovered" && event.amount) {
   279→   279→    updates.moneyRecovered = { increment: event.amount };
   280→   280→    updates.estimatedBusinessValue = { increment: event.amount };
   281→   281→  }
   282→   282→
   283→   283→  // Memory counts
   284→   284→  if (event.type === "memory_created") {
   285→   285→    updates.memoryCount = { increment: 1 };
   286→   286→  }
   287→   287→  if (event.type === "memory_reinforced") {
   288→   288→    updates.reinforcementCount = { increment: 1 };
   289→   289→  }
   290→   290→
   291→   291→  // Capability denial — increase risk
   292→   292→  if (event.type === "capability_denied") {
   293→   293→    updates.riskScore = Math.min(100, profile.riskScore + 5);
   294→   294→  }
   295→   295→
   296→   296→  // Human override — manager edited the proposed action.
   297→   297→  // Slightly raise humanInterventionRate (capped at 1.0) and nudge risk up.
   298→   298→  if (event.type === "human_override") {
   299→   299→    const newIntervention = Math.min(1.0, profile.humanInterventionRate + 0.05);
   300→   300→    updates.humanInterventionRate = newIntervention;
   301→   301→    updates.riskScore = Math.min(100, profile.riskScore + 2);
   302→   302→  }
   303→   303→
   304→   304→  // Contract outcomes — these are bookkeeping events (zero XP) but they
   305→   305→  // still nudge consistency/accuracy scores so the profile reflects the
   306→   306→  // contract lifecycle separately from the approval lifecycle.
   307→   307→  if (event.type === "contract_approved") {
   308→   308→    // Slight accuracy boost: the contract's reasoning was endorsed.
   309→   309→    updates.accuracyScore = Math.min(1.0, profile.accuracyScore + 0.005);
   310→   310→  }
   311→   311→  if (event.type === "contract_rejected") {
   312→   312→    // Slight accuracy penalty: the contract's reasoning was rejected.
   313→   313→    updates.accuracyScore = Math.max(0.0, profile.accuracyScore - 0.01);
   314→   314→  }
   315→   315→
   316→   316→  // ─── Recalculate trust score ─────────────────────────────────────────────
   317→   317→  const newTrustScore = calculateTrustScore({
   318→   318→    approvalRate: updates.approvalRate ?? profile.approvalRate,
   319→   319→    failedTasks: event.type === "task_failed" ? profile.failedTasks + 1 : profile.failedTasks,
   320→   320→    completedTasks: event.type === "task_completed" ? profile.completedTasks + 1 : profile.completedTasks,
   321→   321→    riskScore: updates.riskScore ?? profile.riskScore,
   322→   322→    humanInterventionRate: updates.humanInterventionRate ?? profile.humanInterventionRate,
   323→   323→    hallucinationRate: profile.hallucinationRate,
   324→   324→  });
   325→   325→  updates.trustScore = newTrustScore;
   326→   326→
   327→   327→  // ─── Apply updates ───────────────────────────────────────────────────────
   328→   328→  await db.employeeProfile.update({
   329→   329→    where: { employeeId: event.employeeId },
   330→   330→    data: updates,
   331→   331→  });
   332→   332→
   333→   333→  // ─── Infer and update skills (ONLY on task completion) ───────────────────
   334→   334→  // Skill inference scans the task's title/description/tools and increments
   335→   335→  // matching skill counters. If we ran this on EVERY event (approval_approved,
   336→   336→  // reminder_sent, contract_approved, etc.), a single task with 2 approval
   337→   337→  // gates would increment skills 7+ times. We only want skills to grow once
   338→   338→  // per task completion.
   339→   339→  if (event.type === "task_completed" && event.taskId) {
   340→   340→    try {
   341→   341→      await inferAndUpdateSkills(event.employeeId, event.workspaceId, event.taskId, event.toolName);
   342→   342→    } catch (err) {
   343→   343→      console.error(`[Profile] Skill inference failed for employee ${event.employeeId}:`, err);
   344→   344→    }
   345→   345→  }
   346→   346→
   347→   347→  // ─── Update memory/capability/customer/email counts ─────────────────────
   348→   348→  // Only sync on events that actually change the synced counters:
   349→   349→  // - memory_created / memory_reinforced → memory counts change
   350→   350→  // - task_completed → all synced counters may have changed
   351→   351→  // - reminder_sent → emailsSent/customersHandled/invoicesProcessed may change
   352→   352→  //   (the reminder_sent event fires when the approval is approved; the
   353→   353→  //   actual reminder row is created/sent by the worker shortly after.
   354→   354→  //   We sync here AND on task_completed to catch both timing windows.)
   355→   355→  // Running this on every event caused 3 DB queries × ~90 events = ~270
   356→   356→  // redundant queries per task.
   357→   357→  if (
   358→   358→    event.type === "memory_created" ||
   359→   359→    event.type === "memory_reinforced" ||
   360→   360→    event.type === "task_completed" ||
   361→   361→    event.type === "reminder_sent"
   362→   362→  ) {
   363→   363→    try {
   364→   364→      await updateMemoryAndCapabilityCounts(event.employeeId, event.workspaceId);
   365→   365→    } catch (err) {
   366→   366→      console.error(`[Profile] Count update failed for employee ${event.employeeId}:`, err);
   367→   367→    }
   368→   368→  }
   369→   369→}
   370→   370→
   371→   371→// ─── Trust Score Calculation ─────────────────────────────────────────────────
   372→   372→
   373→   373→function calculateTrustScore(params: {
   374→   374→  approvalRate: number;
   375→   375→  completedTasks: number;
   376→   376→  failedTasks: number;
   377→   377→  riskScore: number;
   378→   378→  humanInterventionRate: number;
   379→   379→  hallucinationRate: number;
   380→   380→}): number {
   381→   381→  const totalTasks = params.completedTasks + params.failedTasks;
   382→   382→  if (totalTasks === 0) return 85.0;
   383→   383→
   384→   384→  // Weighted components (each contributes to a 0-100 score)
   385→   385→  const approvalScore = params.approvalRate * 35;        // 35% weight
   386→   386→  const successScore = totalTasks > 0
   387→   387→    ? (params.completedTasks / totalTasks) * 25            // 25% weight
   388→   388→    : 25;
   389→   389→  const riskScore = (100 - params.riskScore) * 0.15;     // 15% weight (inverse)
   390→   390→  const interventionScore = (1 - params.humanInterventionRate) * 15; // 15% weight
   391→   391→  const hallucinationScore = (1 - params.hallucinationRate) * 10;   // 10% weight
   392→   392→
   393→   393→  return Math.round((approvalScore + successScore + riskScore + interventionScore + hallucinationScore) * 10) / 10;
   394→   394→}
   395→   395→
   396→   396→// ─── Skill Inference ────────────────────────────────────────────────────────
   397→   397→
   398→   398→/**
   399→   399→ * Infers which skills were used in a task and updates skill levels.
   400→   400→ * Skills are inferred from:
   401→   401→ * - Task title and description keywords
   402→   402→ * - Tools used in the task
   403→   403→ */
   404→   404→async function inferAndUpdateSkills(
   405→   405→  employeeId: string,
   406→   406→  workspaceId: string,
   407→   407→  taskId: string,
   408→   408→  toolName?: string
   409→   409→): Promise<void> {
   410→   410→  const task = await db.task.findUnique({
   411→   411→    where: { id: taskId },
   412→   412→    include: { steps: true },
   413→   413→  });
   414→   414→
   415→   415→  if (!task) return;
   416→   416→
   417→   417→  const employee = await db.employee.findUnique({ where: { id: employeeId } });
   418→   418→  if (!employee) return;
   419→   419→
   420→   420→  const skillDefs = getSkillsForRole(employee.role);
   421→   421→  const taskText = `${task.title} ${task.description}`.toLowerCase();
   422→   422→  const toolsUsed = new Set<string>();
   423→   423→  if (toolName) toolsUsed.add(toolName);
   424→   424→  for (const step of task.steps) {
   425→   425→    try {
   426→   426→      const input = JSON.parse(step.input);
   427→   427→      if (input.tool) toolsUsed.add(input.tool);
   428→   428→    } catch {}
   429→   429→  }
   430→   430→
   431→   431→  // Find which skills were exercised
   432→   432→  for (const skillDef of skillDefs) {
   433→   433→    const usedByKeyword = skillDef.keywords.some((kw) => taskText.includes(kw.toLowerCase()));
   434→   434→    const usedByTool = skillDef.tools.some((t) => toolsUsed.has(t));
   435→   435→
   436→   436→    if (usedByKeyword || usedByTool) {
   437→   437→      // Update or create the skill
   438→   438→      const existing = await db.employeeSkill.findUnique({
   439→   439→        where: { employeeId_name: { employeeId, name: skillDef.name } },
   440→   440→      });
   441→   441→
   442→   442→      if (existing) {
   443→   443→        // Increase level every 5 uses, cap at 10
   444→   444→        const newUsageCount = existing.usageCount + 1;
   445→   445→        const newLevel = Math.min(10, Math.floor(newUsageCount / 5) + 1);
   446→   446→        // Increase confidence, cap at 0.99
   447→   447→        const newConfidence = Math.min(0.99, existing.confidence + 0.05);
   448→   448→

   493→   493→// ─── Memory & Capability Count Sync ──────────────────────────────────────────
   494→   494→
   495→   495→/**
   496→   496→ * Syncs derived counters from canonical sources:
   497→   497→ * - memoryCount, reinforcementCount ← EmployeeMemory table
   498→   498→ * - capabilitiesGranted, criticalCapabilities ← EmployeeCapability table
   499→   499→ * - customersHandled, invoicesProcessed ← Reminder table (distinct customer/invoice IDs)
   500→   500→ *
   501→   501→ * Called after memory events and task completion. NOT called on every profile
   502→   502→ * event (would cause ~270 redundant queries per task).
   503→   503→ *
   504→   504→ * NOTE: Prisma's `count()` does NOT support `include` (it conflicts with the
   505→   505→ * implicit `select: { _count }`). We use `findMany` with a projected `select`
   506→   506→ * on the related Capability row instead.
   507→   507→ *
   508→   508→ * NOTE: customersHandled/invoicesProcessed are derived from the Reminder table
   509→   509→ * (distinct customerId/invoiceId where createdBy = employeeId). This is
   510→   510→ * finance-specific in practice but the query itself is generic — any employee
   511→   511→ * that creates reminders gets credit for the customers/invoices it touched.
   512→   512→ * Employees that never create reminders (HR, Sales Ops, etc.) will have these
   513→   513→ * counters stay at 0, which is correct.
   514→   514→ */
   515→   515→async function updateMemoryAndCapabilityCounts(
   516→   516→  employeeId: string,
   517→   517→  _workspaceId: string
   518→   518→): Promise<void> {
   519→   519→  const [memoryCount, reinforcementAgg, allCaps, reminderAgg, sentReminders, distinctInvoices] = await Promise.all([
   520→   520→    db.employeeMemory.count({ where: { employeeId } }),
   521→   521→    db.employeeMemory.aggregate({
   522→   522→      where: { employeeId },
   523→   523→      _sum: { reinforcementCount: true },
   524→   524→    }),
   525→   525→    db.employeeCapability.findMany({
   526→   526→      where: { employeeId },
   527→   527→      include: { capability: { select: { riskLevel: true } } },
   528→   528→    }),
   529→   529→    // Distinct customers from reminders created by this employee.
   530→   530→    db.reminder.groupBy({
   531→   531→      by: ["customerId"],
   532→   532→      where: { createdBy: employeeId },
   533→   533→    }),
   534→   534→    // Count of reminders actually SENT (status = "sent") by this employee.
   535→   535→    // This is the source of truth for emailsSent — not event increments.
   536→   536→    db.reminder.count({
   537→   537→      where: { createdBy: employeeId, status: "sent" },
   538→   538→    }),
   539→   539→    // Distinct invoices from reminders created by this employee.
   540→   540→    db.reminder.groupBy({
   541→   541→      by: ["invoiceId"],
   542→   542→      where: { createdBy: employeeId },
   543→   543→    }),
   544→   544→  ]);
   545→   545→
   546→   546→  const criticalCaps = allCaps.filter(
   547→   547→    (c) =>
   548→   548→      c.capability.riskLevel === "critical" ||
   549→   549→      c.capability.riskLevel === "high"
   550→   550→  ).length;
   551→   551→
   552→   552→  await db.employeeProfile.update({
   553→   553→    where: { employeeId },
   554→   554→    data: {
   555→   555→      memoryCount,
   556→   556→      reinforcementCount: reinforcementAgg._sum.reinforcementCount || 0,
   557→   557→      capabilitiesGranted: allCaps.length,
   558→   558→      criticalCapabilities: criticalCaps,
   559→   559→      emailsSent: sentReminders,
   560→   560→      customersHandled: reminderAgg.length,
   561→   561→      invoicesProcessed: distinctInvoices.length,
   562→   562→    },
   563→   563→  });
   564→   564→}
   565→   565→
   566→   566→// ─── Profile Retrieval ───────────────────────────────────────────────────────
   567→   567→
   568→   568→/**
   569→   569→ * Recomputes the derived/synced fields of a profile from canonical sources:
   570→   570→ * memory count, reinforcement count, capability counts. Used by backfill
   571→   571→ * scripts and as a consistency check after schema migrations.
   572→   572→ *
   573→   573→ * NOTE: This does NOT recompute XP, trust, KPIs, or skills — those are
   574→   574→ * event-driven and accumulated over time. Only the "synced" counters
   575→   575→ * (which mirror state in other tables) are recomputed.
   576→   576→ */
   577→   577→export async function recalcProfileSyncedCounts(employeeId: string): Promise<void> {
   578→   578→  await updateMemoryAndCapabilityCounts(employeeId, "");
   579→   579→}

   717→   717→function getNextLevelXp(currentXp: number): number {
   718→   718→  const current = getLevelForXp(currentXp);
   719→   719→  const nextLevel = LEVELS.find((l) => l.level === current.level + 1);
   720→   720→  return nextLevel ? nextLevel.minXp : currentXp; // Max level
   721→   721→}
   722→   722→
   723→   723→function getProgressToNextLevel(currentXp: number): number {
   724→   724→  const current = getLevelForXp(currentXp);
   725→   725→  const nextLevel = LEVELS.find((l) => l.level === current.level + 1);
   726→   726→  if (!nextLevel) return 100; // Max level
   727→   727→  const range = nextLevel.minXp - current.minXp;
   728→   728→  const progress = (currentXp - current.minXp) / range;
   729→   729→  return Math.round(progress * 100);
   730→   730→}
