   122→  // ─── CSV parsing ─────────────────────────────────────────────────────────
   123→  function parseCsv(text: string): ParsedInvoice[] {
   124→    const lines = text.trim().split(/\r?\n/).filter((l) => l.trim());
   125→    if (lines.length < 2) {
   126→      throw new Error("CSV must have a header row and at least one data row.");
   127→    }
   128→
   129→    const headers = lines[0].split(",").map((h) => h.trim().toLowerCase());
   130→    const required = ["customer_name", "customer_email", "invoice_number", "issue_date", "due_date", "subtotal", "tax"];
   131→    for (const req of required) {
   132→      if (!headers.includes(req)) {
   133→        throw new Error(`Missing required column: ${req}. Expected columns: ${required.join(", ")}`);
   134→      }
   135→    }
   136→
   137→    const invoices: ParsedInvoice[] = [];
   138→    for (let i = 1; i < lines.length; i++) {
   139→      const cols = parseCsvLine(lines[i]);
   140→      if (cols.length < headers.length) {
   141→        throw new Error(`Row ${i + 1} has ${cols.length} columns, expected ${headers.length}.`);
   142→      }
   143→
   144→      const row: Record<string, string> = {};
   145→      headers.forEach((h, idx) => { row[h] = cols[idx]?.trim() || ""; });
   146→
   147→      // Convert amounts from rupees to paise (1 INR = 100 paise)
   148→      const subtotal = Math.round(parseFloat(row.subtotal) * 100);
   149→      const tax = Math.round(parseFloat(row.tax) * 100);
   150→
   151→      if (isNaN(subtotal) || isNaN(tax)) {
   152→        throw new Error(`Row ${i + 1}: subtotal and tax must be numbers.`);
   153→      }
   154→
   155→      invoices.push({
   156→        customerName: row.customer_name,
   157→        customerEmail: row.customer_email,
   158→        invoiceNumber: row.invoice_number,
   159→        issueDate: row.issue_date,
   160→        dueDate: row.due_date,
   161→        subtotal,
   162→        tax,
   163→      });
   164→    }
   165→
   166→    return invoices;
   167→  }
   168→
   169→  function parseCsvLine(line: string): string[] {
   170→    const result: string[] = [];
   171→    let current = "";
   172→    let inQuotes = false;
   173→    for (let i = 0; i < line.length; i++) {
   174→      const char = line[i];
   175→      if (char === '"') {
   176→        inQuotes = !inQuotes;
   177→      } else if (char === "," && !inQuotes) {
   178→        result.push(current);
   179→        current = "";
   180→      } else {
   181→        current += char;
   182→      }
   183→    }
   184→    result.push(current);
   185→    return result;
   186→  }

   213→        setParseError(err.message);
   214→      }
   215→    };
   216→    reader.readAsText(file);
