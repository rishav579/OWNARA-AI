   742→// ─── 2. Skill Reinforcement ──────────────────────────────────────────────────
   743→// Append-only ledger of every reinforcement event for a skill.
   744→// Instead of "skill++", we record WHY the skill was reinforced.
   745→// The skill's level is derived from the sum of reinforcements.
   746→//
   747→// Reinforcement values:
   748→//   positive (+1, +2, +3) = good outcome attributable to this skill
   749→//   negative (-1, -2, -3) = bad outcome attributable to this skill
   750→//
   751→// Level formula (in engine):
   752→//   totalReinforcement = sum of all reinforcement values for this skill
   753→//   level = clamp(1, 10, 1 + floor(totalReinforcement / 5))
   754→//   (every +5 net reinforcement = +1 level)
   755→model SkillReinforcement {
   756→  id              String   @id @default(cuid())
   757→  workspaceId     String
   758→  employeeId      String
   759→  skillName       String   // e.g. "Collections", "Invoice Analysis"
   760→  taskId          String?  // null if not tied to a task
   761→  outcomeId       String?  // link to OutcomeEvaluation
   762→
   763→  // Why was this skill reinforced? (deterministic reason code)
   764→  reason          String   // e.g. "successful_collection", "payment_recovered",
   765→                           //      "reminder_approved", "rejected", "rollback"
   766→  reinforcement   Int      @default(0)  // +1, +2, +3, -1, -2, -3
   767→  notes           String   @default("") // human-readable detail
   768→
   769→  createdAt       DateTime @default(now())
   770→
   771→  workspace       Workspace @relation(fields: [workspaceId], references: [id])
   772→
   773→  @@index([employeeId, skillName, createdAt])
   774→  @@index([workspaceId, createdAt])
   775→}
