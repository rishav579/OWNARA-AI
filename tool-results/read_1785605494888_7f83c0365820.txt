     1→/**
     2→ * BIHARI AI — Task Executor
     3→ *
     4→ * Processes a single step of a task. The worker calls this function once per
     5→ * poll tick. This is the heart of the trust loop:
     6→ *
     7→ * 1. Find the next pending step
     8→ * 2. Execute it (or create an approval gate)
     9→ * 3. Write an audit entry atomically
    10→ * 4. If approval needed: set task to waiting_approval and STOP
    11→ * 5. If no more steps: set task to completed
    12→ *
    13→ * Per BED-001 §7: "The executor calls the LLM Gateway to produce the step's
    14→ * reasoning and proposed action. The executor persists a task_steps row with
    15→ * input, reasoning, and output."
    16→ *
    17→ * Per BED-001 §7: "If critical: creates an approvals row (status pending) with
    18→ * the proposed action, appends audit entry approval_requested, sends a
    19→ * notification, transitions the task to waiting_approval, and returns without
    20→ * executing."
    21→ */

    55→// ─── Main Entry Point ────────────────────────────────────────────────────────
    56→
    57→/**
    58→ * Processes one unit of work for a task.
    59→ *
    60→ * This function is called by the worker poll loop. It does ONE of:
    61→ * - Plans the task (if status is "queued")
    62→ * - Executes the next pending step (if status is "executing")
    63→ * - Resolves an approval gate (if status is "executing" and next step is
    64→ *   approval_gate with a decided approval)
    65→ * - Completes the task (if no pending steps remain)
    66→ *
    67→ * All state changes and audit entries are written in a single transaction.
    68→ */
    69→export async function processTask(taskId: string): Promise<ExecutionResult> {
    70→  const task = await db.task.findUnique({
    71→    where: { id: taskId },
    72→    include: {
    73→      employee: true,
    74→      steps: { orderBy: { stepNumber: "asc" } },
    75→    },
    76→  });
    77→
    78→  if (!task) {
    79→    return { action: "failed", message: "Task not found" };
    80→  }
    81→
    82→  // Don't process tasks that aren't in an active execution state
    83→  if (!["queued", "planning", "executing"].includes(task.status)) {
    84→    return { action: "continue", message: `Task is ${task.status}, skipping` };
    85→  }
    86→
    87→  // Check if the employee is paused
    88→  if (task.employee.status === "paused") {
    89→    // Pause the task
    90→    await db.$transaction(async (tx) => {
    91→      await tx.task.update({
    92→        where: { id: taskId },
    93→        data: { status: "paused" },
    94→      });
    95→      await appendAudit(tx, {
    96→        workspaceId: task.workspaceId,
    97→        entryType: "task_paused",
    98→        actorType: "system",
    99→        actorId: null,
   100→        actorName: "Runtime",
   101→        targetType: "task",
   102→        targetId: taskId,
   103→        payload: { reason: "Employee paused", employee: task.employee.name },
   104→      });
   105→    });
   106→    return { action: "continue", message: "Employee paused, task paused" };
   107→  }
   108→
   109→  // Phase 1: Planning — generate the step plan
   110→  if (task.status === "queued") {
   111→    return await planTask(task);
   112→  }
   113→
   114→  // Phase 2: Execution — process the next pending step
   115→  const pendingSteps = task.steps.filter((s) => s.status === "pending");
   116→
   117→  if (pendingSteps.length === 0) {
   118→    // No pending steps — task is complete
   119→    return await completeTask(task);
   120→  }
   121→
   122→  const nextStep = pendingSteps[0];
   123→  return await executeStep(task, nextStep, task.steps);
   124→}

   126→// ─── Planning Phase ──────────────────────────────────────────────────────────
   127→
   128→async function planTask(
   129→  task: NonNullable<Awaited<ReturnType<typeof db.task.findUnique>>> & {}
   130→): Promise<ExecutionResult> {
   131→  const employee = task.employee;
   132→  const employeeTools: string[] = JSON.parse(employee.tools);
   133→
   134→  // Transition to planning
   135→  await db.$transaction(async (tx) => {
   136→    await tx.task.update({
   137→      where: { id: task.id },
   138→      data: { status: "planning", startedAt: task.startedAt ?? new Date() },
   139→    });
   140→    await appendAudit(tx, {
   141→      workspaceId: task.workspaceId,
   142→      entryType: "task_started",
   143→      actorType: "user",
   144→      actorId: task.assignedBy,
   145→      actorName: "Runtime",
   146→      targetType: "task",
   147→      targetId: task.id,
   148→      payload: {
   149→        title: task.title,
   150→        employee: employee.name,
   151→        role: employee.role,
   152→      },
   153→    });
   154→  });
   155→
   156→  // Generate the plan
   157→  let plan: { steps: PlannedStep[] };
   158→
   159→  // Check if this is a finance task
   160→  if (isFinanceTask(employee.role, task.title, task.description)) {
   161→    // Finance planning: use the Finance Brain to build context + produce recommendation
   162→    const invoiceIdMatch = task.description.match(/invoice[_\s-]?id[:\s]+([a-zA-Z0-9]+)/i);
   163→    const invoiceNumMatch = task.description.match(/invoice[_\s-]?number[:\s]+([A-Z0-9-]+)/i);
   164→
   165→    if (invoiceIdMatch || invoiceNumMatch) {
   166→      // Single invoice processing — Finance Brain builds full context + recommendation
   167→      let invoiceId = invoiceIdMatch?.[1];
   168→
   169→      // If we have an invoice number but no ID, look it up
   170→      if (!invoiceId && invoiceNumMatch) {
   171→        const inv = await db.invoice.findFirst({
   172→          where: { invoiceNumber: invoiceNumMatch[1] },
   173→        });
   174→        invoiceId = inv?.id;
   175→      }
   176→
   177→      if (invoiceId) {
   178→        // Step 1: Build the complete FinanceContext (invoice, customer, payments, reminders, policies, memory, etc.)
   179→        const financeCtx = await buildFinanceContext(invoiceId, employee.id);
   180→        if (financeCtx) {
   181→          // Step 2: The Finance Brain reasons over the context (including memory) to produce a recommendation
   182→          const recommendation = produceRecommendation(financeCtx);
   183→          // Step 3: Generate the execution plan from the recommendation
   184→          plan = generateFinancePlan(recommendation, employeeTools);
   185→        } else {
   186→          plan = { steps: [] };
   187→        }
   188→      } else {
   189→        plan = { steps: [] };
   190→      }
   191→    } else {
   192→      // Batch processing: find invoices needing attention, then build full context + recommendation for each
   193→      const invoiceContexts = await findInvoicesNeedingAttention(task.workspaceId, 3); // Limit to 3 per task
   194→      if (invoiceContexts.length > 0) {
   195→        // For each invoice, build the full FinanceContext and produce a recommendation
   196→        const recommendations: FinanceRecommendation[] = [];
   197→        for (const ic of invoiceContexts) {
   198→          const financeCtx = await buildFinanceContext(ic.invoiceId, employee.id);
   199→          if (financeCtx) {
   200→            const rec = produceRecommendation(financeCtx);
   201→            recommendations.push(rec);
   202→          }
   203→        }
   204→        if (recommendations.length > 0) {
   205→          plan = generateBatchFinancePlan(recommendations, employeeTools);
   206→        } else {
   207→          plan = { steps: [] };
   208→        }
   209→      } else {
   210→        plan = { steps: [] };
   211→      }
   212→    }
   213→  } else {
   214→    // Generic planning
   215→    // Check if the LLM Gateway has a real provider available (not mock)
   216→    const gateway = getLLMGateway();
   217→    const router = (await import("@/lib/llm")).getModelRouter();
   218→    const { provider, route } = router.route("planning");
   219→
   220→    if (provider.name !== "mock") {
   221→      // ─── LLM-Gateway-Aware Planning ──────────────────────────────────────
   222→      // When a real LLM provider is configured, use the gateway to generate
   223→      // the execution plan. This replaces the deterministic mock planner
   224→      // with real AI planning.
   225→      try {
   226→        const response = await gateway.complete({
   227→          taskType: "planning",
   228→          promptId: "planning",
   229→          variables: {
   230→            title: task.title,
   231→            description: task.description,
   232→            role: employee.role,
   233→            tools: employeeTools.join(", "),
   234→          },
   235→          jsonMode: true,
   236→          jsonSchema: {
   237→            type: "object",
   238→            required: ["reasoning", "steps"],
   239→            properties: {
   240→              reasoning: { type: "string" },
   241→              steps: {
   242→                type: "array",
   243→                items: {
   244→                  type: "object",
   245→                  required: ["stepType", "reasoning", "confidence"],
   246→                  properties: {
   247→                    stepType: { type: "string" },
   248→                    reasoning: { type: "string" },
   249→                    tool: { type: "string" },
   250→                    confidence: { type: "number" },
   251→                  },
   252→                },
   253→              },
   254→            },
   255→          },
   256→          workspaceId: task.workspaceId,
   257→          employeeId: employee.id,
   258→          taskId: task.id,
   259→        });
   260→
   261→        // Parse the LLM-generated plan
   262→        const llmPlan = response.data as { reasoning: string; steps: PlannedStep[] } | null;
   263→        if (llmPlan && llmPlan.steps && Array.isArray(llmPlan.steps)) {
   264→          plan = { steps: llmPlan.steps };
   265→        } else {
   266→          // Fallback to deterministic planner if LLM response is invalid
   267→          plan = generatePlan(task.title, task.description, employee.role, employeeTools);
   268→        }
   269→      } catch (err) {
   270→        // On any gateway error, fall back to the deterministic planner
   271→        console.error("[Executor] LLM Gateway planning failed, falling back to deterministic planner:", err);
   272→        plan = generatePlan(task.title, task.description, employee.role, employeeTools);
   273→      }
   274→    } else {
   275→      // ─── Deterministic Planning (existing behavior) ─────────────────────
   276→      // No real LLM provider configured — use the existing mock planner.
   277→      // This maintains backward compatibility.
   278→      plan = generatePlan(task.title, task.description, employee.role, employeeTools);
   279→    }
   280→  }
   281→
   282→  if (plan.steps.length === 0) {
   283→    // No steps to execute — fail immediately
   284→    await db.$transaction(async (tx) => {
   285→      await tx.task.update({
   286→        where: { id: task.id },
   287→        data: { status: "failed" },
   288→      });
   289→      await appendAudit(tx, {
   290→        workspaceId: task.workspaceId,
   291→        entryType: "task_failed",
   292→        actorType: "employee",
   293→        actorId: employee.id,
   294→        actorName: employee.name,
   295→        targetType: "task",
   296→        targetId: task.id,
   297→        payload: { reason: "Plan generation produced no steps" },
   298→      });
   299→    });
   300→    return { action: "failed", message: "Plan generation produced no steps" };
   301→  }
   302→
   303→  // Check step cap
   304→  if (plan.steps.length > task.stepCap) {
   305→    plan.steps = plan.steps.slice(0, task.stepCap);
   306→  }
   307→
   308→  // Write the plan step (step 0 — the planning reasoning)
   309→  const approvalRules: Record<string, string> = JSON.parse(employee.approvalRules);
   310→
   311→  await db.$transaction(async (tx) => {
   312→    // Step 0: The plan itself
   313→    const isFinance = isFinanceTask(employee.role, task.title, task.description);
   314→    const criticalCount = plan.steps.filter((s) => s.tool === "send_email" || s.tool === "send_reminder").length;
   315→    await tx.taskStep.create({
   316→      data: {
   317→        taskId: task.id,
   318→        stepNumber: 0,
   319→        stepType: "plan",
   320→        reasoning: isFinance
   321→          ? `Planned ${plan.steps.length} steps for this finance collections task. The plan includes ${criticalCount} critical action(s) requiring human approval.`
   322→          : `Planned ${plan.steps.length} steps for this task. The plan includes ${criticalCount} email action(s) that will require human approval.`,
   323→        status: "completed",
   324→        tokens: 420,
   325→        durationMs: 3100,
   326→        startedAt: new Date(),
   327→        completedAt: new Date(),
   328→      },
   329→    });
   330→
   331→    // Write each planned step as pending
   332→    for (let i = 0; i < plan.steps.length; i++) {
   333→      const step = plan.steps[i];
   334→      const isCritical = step.tool ? approvalRules[step.tool] === "critical" : false;
   335→
   336→      await tx.taskStep.create({
   337→        data: {
   338→          taskId: task.id,
   339→          stepNumber: i + 1,
   340→          stepType: isCritical ? "approval_gate" : step.stepType,
   341→          input: JSON.stringify({
   342→            tool: step.tool || null,
   343→            ...step.toolInput,
   344→          }),
   345→          reasoning: step.reasoning,
   346→          status: "pending",
   347→          confidence: step.confidence,
   348→        },
   349→      });
   350→    }
   351→
   352→    // Transition to executing
   353→    await tx.task.update({
   354→      where: { id: task.id },
   355→      data: { status: "executing", stepCount: plan.steps.length + 1 },
   356→    });
   357→
   358→    await appendAudit(tx, {
   359→      workspaceId: task.workspaceId,
   360→      entryType: "plan_created",
   361→      actorType: "employee",
   362→      actorId: employee.id,
   363→      actorName: employee.name,
   364→      targetType: "task",
   365→      targetId: task.id,
   366→      payload: {
   367→        step_count: String(plan.steps.length),
   368→        critical_steps: String(plan.steps.filter((s) => s.tool && approvalRules[s.tool] === "critical").length),
   369→      },
   370→    });
   371→  });
   372→
   373→  return { action: "continue", message: `Planned ${plan.steps.length} steps, now executing` };
   374→}

   376→// ─── Step Execution ──────────────────────────────────────────────────────────
   377→
   378→async function executeStep(
   379→  task: any,
   380→  step: any,
   381→  allSteps: any[]
   382→): Promise<ExecutionResult> {
   383→  const employee = task.employee;
   384→
   385→  // Mark step as running
   386→  await db.taskStep.update({
   387→    where: { id: step.id },
   388→    data: { status: "running", startedAt: new Date() },
   389→  });
   390→
   391→  try {
   392→    switch (step.stepType) {
   393→      case "reasoning":
   394→        return await executeReasoningStep(task, step, employee);
   395→
   396→      case "tool_call":
   397→        return await executeToolStep(task, step, employee);
   398→
   399→      case "approval_gate":
   400→        return await executeApprovalGateStep(task, step, employee);
   401→
   402→      default:
   403→        return await executeReasoningStep(task, step, employee);
   404→    }
   405→  } catch (err) {
   406→    // Mark step as failed and fail the task
   407→    await db.$transaction(async (tx) => {
   408→      await tx.taskStep.update({
   409→        where: { id: step.id },
   410→        data: { status: "failed", completedAt: new Date() },
   411→      });
   412→      await tx.task.update({
   413→        where: { id: task.id },
   414→        data: { status: "failed" },
   415→      });
   416→      await appendAudit(tx, {
   417→        workspaceId: task.workspaceId,
   418→        entryType: "task_failed",
   419→        actorType: "employee",
   420→        actorId: employee.id,
   421→        actorName: employee.name,
   422→        targetType: "task",
   423→        targetId: task.id,
   424→        payload: {
   425→          reason: `Step ${step.stepNumber} failed: ${err instanceof Error ? err.message : "unknown error"}`,
   426→        },
   427→      });
   428→    });
   429→    return { action: "failed", message: `Step ${step.stepNumber} failed` };
   430→  }
   431→}

   433→// ─── Reasoning Step ──────────────────────────────────────────────────────────
   434→
   435→async function executeReasoningStep(task: any, step: any, employee: any): Promise<ExecutionResult> {
   436→  const tokens = 500 + Math.floor(Math.random() * 400);
   437→  const durationMs = 1500 + Math.floor(Math.random() * 1500);
   438→
   439→  await db.$transaction(async (tx) => {
   440→    await tx.taskStep.update({
   441→      where: { id: step.id },
   442→      data: {
   443→        status: "completed",
   444→        output: JSON.stringify({ type: "reasoning" }),
   445→        tokens,
   446→        durationMs,
   447→        completedAt: new Date(),
   448→      },
   449→    });
   450→
   451→    await tx.task.update({
   452→      where: { id: task.id },
   453→      data: { tokenUsage: { increment: tokens } },
   454→    });
   455→
   456→    await appendAudit(tx, {
   457→      workspaceId: task.workspaceId,
   458→      entryType: "step_executed",
   459→      actorType: "employee",
   460→      actorId: employee.id,
   461→      actorName: employee.name,
   462→      targetType: "task_step",
   463→      targetId: step.id,
   464→      payload: {
   465→        step: String(step.stepNumber),
   466→        type: "reasoning",
   467→        tokens: String(tokens),
   468→      },
   469→    });
   470→  });
   471→
   472→  return { action: "continue", message: `Reasoning step ${step.stepNumber} completed` };
   473→}

   475→// ─── Tool Call Step (non-critical) ───────────────────────────────────────────
   476→
   477→async function executeToolStep(task: any, step: any, employee: any): Promise<ExecutionResult> {
   478→  const input = JSON.parse(step.input);
   479→  const toolName = input.tool;
   480→
   481→  if (!toolName) {
   482→    // No tool specified — treat as reasoning
   483→    return await executeReasoningStep(task, step, employee);
   484→  }
   485→
   486→  // ─── Capability Verification ──────────────────────────────────────────────
   487→  // Before ANY tool execution, verify the employee has the required capability.
   488→  // If denied: stop execution, write audit event, fail the step.
   489→  const capCheck = await checkCapability(employee.id, toolName);
   490→  if (!capCheck.allowed) {
   491→    await db.$transaction(async (tx) => {
   492→      // Mark step as failed
   493→      await tx.taskStep.update({
   494→        where: { id: step.id },
   495→        data: {
   496→          status: "failed",
   497→          output: JSON.stringify({
   498→            error: "CAPABILITY_DENIED",
   499→            requiredCapability: capCheck.capabilityCode,
   500→            capabilityName: capCheck.capabilityName,
   501→            reason: capCheck.reason,
   502→          }),
   503→          completedAt: new Date(),
   504→        },
   505→      });
   506→
   507→      // Fail the task
   508→      await tx.task.update({
   509→        where: { id: task.id },
   510→        data: { status: "failed" },
   511→      });
   512→
   513→      // Update employee state
   514→      await tx.employee.update({
   515→        where: { id: employee.id },
   516→        data: { state: "idle" },
   517→      });
   518→
   519→      // Record the denial in the audit chain
   520→      await recordCapabilityDenial(tx, task.workspaceId, employee.id, employee.name, toolName, capCheck);
   521→    });
   522→
   523→    // ─── Update Employee Profile (capability denied) ───────────────────────
   524→    try {
   525→      await recordProfileEvent({
   526→        type: "capability_denied",
   527→        employeeId: employee.id,
   528→        workspaceId: task.workspaceId,
   529→        taskId: task.id,
   530→        toolName,
   531→      });
   532→    } catch (err) {
   533→      console.error(`[Executor] Profile update failed for capability denial:`, err);
   534→    }
   535→
   536→    return { action: "failed", message: `Capability denied: ${capCheck.capabilityCode} required for ${toolName}` };
   537→  }
   538→
   539→  // Check if this is a finance tool
   540→  const financeTools = ["generate_reminder", "send_reminder", "update_collection_case"];
   541→  const isFinanceTool = financeTools.includes(toolName);
   542→
   543→  // Execute the tool
   544→  let result: { output: Record<string, string>; tokens: number; durationMs: number };
   545→
   546→  if (isFinanceTool) {
   547→    result = await executeFinanceTool(toolName, input, task.workspaceId, employee.id);
   548→  } else {
   549→    result = executeTool(toolName, input);
   550→  }
   551→
   552→  // Determine the finance-specific audit entry type
   553→  let auditEntryType = "tool_executed";
   554→  if (toolName === "generate_reminder") auditEntryType = "reminder_drafted";
   555→  else if (toolName === "update_collection_case") auditEntryType = "collection_case_updated";
   556→
   557→  await db.$transaction(async (tx) => {
   558→    await tx.taskStep.update({
   559→      where: { id: step.id },
   560→      data: {
   561→        status: "completed",
   562→        output: JSON.stringify(result.output),
   563→        tokens: result.tokens,
   564→        durationMs: result.durationMs,
   565→        completedAt: new Date(),
   566→      },
   567→    });
   568→
   569→    await tx.task.update({
   570→      where: { id: task.id },
   571→      data: { tokenUsage: { increment: result.tokens } },
   572→    });
   573→
   574→    await appendAudit(tx, {
   575→      workspaceId: task.workspaceId,
   576→      entryType: auditEntryType,
   577→      actorType: "employee",
   578→      actorId: employee.id,
   579→      actorName: employee.name,
   580→      targetType: isFinanceTool ? "invoice" : "task_step",
   581→      targetId: input.invoiceId || step.id,
   582→      payload: {
   583→        tool: toolName,
   584→        status: "completed",
   585→        tokens: String(result.tokens),
   586→        ...(input.invoiceNumber ? { invoiceNumber: input.invoiceNumber } : {}),
   587→        ...(input.customerName ? { customer: input.customerName } : {}),
   588→      },

   597→async function executeApprovalGateStep(task: any, step: any, employee: any): Promise<ExecutionResult> {

   621→        where: { id: task.id },

   899→  //   3. Detect reusable patterns (customer behavior, reminder effectiveness)

   917→  return { action: "completed", message: "Task completed" };

   920→// ─── Resume After Approval ───────────────────────────────────────────────────
   921→
   922→/**
   923→ * Called by the approval API when a human approves an action.
   924→ * Marks the approval_gate step as completed and transitions the task back
   925→ * to executing so the worker can continue with the next step.
   926→ *
   927→ * Also "executes" the approved tool (e.g., actually sends the email in a
   928→ * real implementation).
   929→ */
   930→export async function resumeAfterApproval(
   931→  taskId: string,
   932→  approvalId: string,
   933→  approvedBy: string,
   934→  approvedByName: string
   935→): Promise<void> {
   936→  const task = await db.task.findUnique({
   937→    where: { id: taskId },
   938→    include: { employee: true, steps: { orderBy: { stepNumber: "asc" } } },
   939→  });
   940→
   941→  if (!task) return;
   942→
   943→  const approval = await db.approval.findUnique({ where: { id: approvalId } });
   944→  if (!approval) return;
   945→
   946→  // Find the approval_gate step that was waiting
   947→  const gateStep = task.steps.find(
   948→    (s) => s.stepType === "approval_gate" && s.status === "pending"
   949→  );
   950→
   951→  // ─── Capability Verification (post-approval) ──────────────────────────────
   952→  // Even after approval, verify the employee still has the capability.
   953→  // Capabilities can be revoked between approval and execution.
   954→  const postApprovalCapCheck = await checkCapability(task.employeeId, approval.tool);
   955→  if (!postApprovalCapCheck.allowed) {
   956→    // Capability was revoked after approval — fail the task
   957→    await db.$transaction(async (tx) => {
   958→      await tx.task.update({
   959→        where: { id: taskId },
   960→        data: { status: "failed" },
   961→      });
   962→      await tx.employee.update({
   963→        where: { id: task.employeeId },
   964→        data: { state: "idle", pendingApprovals: { decrement: 1 } },
   965→      });
   966→      await recordCapabilityDenial(tx, task.workspaceId, task.employeeId, task.employee.name, approval.tool, postApprovalCapCheck);
   967→    });
   968→    return;
   969→  }
   970→
   971→  // Execute the approved tool BEFORE the transaction (it may do DB operations)
   972→  const proposedAction = JSON.parse(approval.proposedAction);
   973→  const financeTools = ["generate_reminder", "send_reminder", "update_collection_case"];
   974→  const isFinanceTool = financeTools.includes(approval.tool);
   975→
   976→  let toolResult: { output: Record<string, string>; tokens: number; durationMs: number };
   977→  if (isFinanceTool) {
   978→    toolResult = await executeFinanceTool(approval.tool, proposedAction, task.workspaceId, task.employeeId);
   979→  } else {
   980→    toolResult = executeTool(approval.tool, proposedAction);
   981→  }
   982→

   984→    // Mark the gate step as completed
   985→    if (gateStep) {
   986→      await tx.taskStep.update({
   987→        where: { id: gateStep.id },
   988→        data: {
   989→          status: "completed",
   990→          output: JSON.stringify({
   991→            ...toolResult.output,
   992→            approvalId,
   993→            approvedBy: approvedByName,
   994→          }),
   995→          tokens: toolResult.tokens,
   996→          durationMs: toolResult.durationMs,
   997→          completedAt: new Date(),
   998→        },
   999→      });
  1000→    }
  1001→
  1002→    // Transition task back to executing
  1003→    await tx.task.update({
  1004→      where: { id: taskId },
  1005→      data: {
  1006→        status: "executing",
  1007→        tokenUsage: { increment: toolResult.tokens },
  1008→      },
  1009→    });
  1010→
  1011→    // Update employee state
  1012→    await tx.employee.update({
  1013→      where: { id: task.employeeId },
  1014→      data: {
  1015→        state: "executing",
  1016→        pendingApprovals: { decrement: 1 },
  1017→      },
  1018→    });
  1019→
  1020→    // Write audit entry for the approval decision
  1021→    const isFinanceApproval = approval.tool === "send_reminder";
  1022→    await appendAudit(tx, {
  1023→      workspaceId: task.workspaceId,
  1024→      entryType: isFinanceApproval ? "reminder_approved" : "approval_decided",
  1025→      actorType: "user",
  1026→      actorId: approvedBy,
  1027→      actorName: approvedByName,
  1028→      targetType: isFinanceApproval ? "invoice" : "approval",
  1029→      targetId: isFinanceApproval ? (proposedAction.invoiceId || approvalId) : approvalId,
  1030→      payload: {
  1031→        decision: "approved",
  1032→        tool: approval.tool,
  1033→        employee: task.employee.name,
  1034→        ...(proposedAction.invoiceNumber ? { invoiceNumber: proposedAction.invoiceNumber } : {}),
  1035→        ...(proposedAction.customerName ? { customer: proposedAction.customerName } : {}),
  1036→      },
  1037→    });
  1038→
  1039→    // Write audit entry for the tool execution
  1040→    await appendAudit(tx, {
  1041→      workspaceId: task.workspaceId,
  1042→      entryType: isFinanceApproval ? "reminder_sent" : "tool_executed",
  1043→      actorType: "employee",
  1044→      actorId: task.employeeId,
  1045→      actorName: task.employee.name,
  1046→      targetType: isFinanceApproval ? "invoice" : "task_step",
  1047→      targetId: proposedAction.invoiceId || gateStep?.id || "",
  1048→      payload: {

  1050→        status: "completed",

  1052→        ...(proposedAction.invoiceNumber ? { invoiceNumber: proposedAction.invoiceNumber } : {}),

  1077→      task.employeeId,
  1078→      task.workspaceId,
  1079→      approvalId,
  1080→      "approved",
  1081→      null,
  1082→      approval.tool,
  1083→      proposedAction
  1084→    );

  1107→    }
  1108→    // Bookkeeping: contract_approved is zero-XP (XP already counted via
  1109→    // approval_approved) but nudges the accuracy score so the profile
  1110→    // reflects the contract lifecycle separately from the approval lifecycle.
  1111→    if (gateStep) {
  1112→      const stepOutput = JSON.parse(gateStep.output || "{}");
  1113→      if (stepOutput.contractId) {
  1114→        await recordProfileEvent({

  1127→/**
  1128→ * Called by the approval API when a human rejects an action.
  1129→ * Marks the approval_gate step as failed and transitions the task to failed.
  1130→ */
  1131→export async function failAfterApprovalRejection(
  1132→  taskId: string,
  1133→  approvalId: string,
  1134→  rejectedBy: string,
  1135→  rejectedByName: string,
  1136→  reason?: string
  1137→): Promise<void> {
  1138→  const task = await db.task.findUnique({
  1139→    where: { id: taskId },
  1140→    include: { employee: true, steps: { orderBy: { stepNumber: "asc" } } },
  1141→  });
  1142→
  1143→  if (!task) return;
  1144→
  1145→  // Fetch the approval to get the tool name
  1146→  const approvalRecord = await db.approval.findUnique({ where: { id: approvalId } });
  1147→
  1148→  const gateStep = task.steps.find(
  1149→    (s) => s.stepType === "approval_gate" && s.status === "pending"
  1150→  );
  1151→
  1152→  await db.$transaction(async (tx) => {
  1153→    // Mark the gate step as failed
  1154→    if (gateStep) {
  1155→      await tx.taskStep.update({
  1156→        where: { id: gateStep.id },
  1157→        data: {
  1158→          status: "failed",
  1159→          output: JSON.stringify({
  1160→            approvalId,
  1161→            rejectedBy: rejectedByName,
  1162→            reason: reason || "Rejected by human",
  1163→          }),
  1164→          completedAt: new Date(),
  1165→        },
  1166→      });
  1167→    }
  1168→
  1169→    // Transition task to failed
  1170→    await tx.task.update({
  1171→      where: { id: taskId },
  1172→      data: { status: "failed" },
  1173→    });
  1174→
  1175→    // Update employee state
  1176→    await tx.employee.update({
  1177→      where: { id: task.employeeId },
  1178→      data: {
  1179→        state: "idle",
  1180→        pendingApprovals: { decrement: 1 },
  1181→      },
  1182→    });
  1183→
  1184→    // Write audit entry for the approval decision
  1185→    await appendAudit(tx, {
  1186→      workspaceId: task.workspaceId,
  1187→      entryType: "approval_decided",
  1188→      actorType: "user",
  1189→      actorId: rejectedBy,
  1190→      actorName: rejectedByName,
  1191→      targetType: "approval",
  1192→      targetId: approvalId,
  1193→      payload: {
  1194→        decision: "rejected",
  1195→        tool: approvalRecord?.tool || "",
  1196→        employee: task.employee.name,
  1197→        reason: reason || "Rejected",
  1198→      },
  1199→    });
  1200→
  1201→    // Write task_failed audit
  1202→    await appendAudit(tx, {
  1203→      workspaceId: task.workspaceId,
  1204→      entryType: "task_failed",
  1205→      actorType: "user",
  1206→      actorId: rejectedBy,
  1207→      actorName: rejectedByName,
  1208→      targetType: "task",
  1209→      targetId: taskId,
  1210→      payload: {
  1211→        reason: `Approval rejected: ${reason || "No reason provided"}`,
  1212→      },
  1213→    });
  1214→  });
  1215→
  1216→  // ─── Record Rejection Feedback in Memory ──────────────────────────────────
  1217→  // The employee learns from the manager's rejection — this is especially
  1218→  // important because rejections often carry a reason that teaches the
  1219→  // employee what NOT to do next time.
  1220→  try {
  1221→    const proposedAction = JSON.parse(approvalRecord?.proposedAction || "{}");
  1222→    await recordApprovalFeedback(
  1223→      task.employeeId,
  1224→      task.workspaceId,
  1225→      approvalId,
  1226→      "rejected",
  1227→      reason || null,
  1228→      approvalRecord?.tool || "",
  1229→      proposedAction
  1230→    );
  1231→  } catch (err) {
  1232→    console.error(`[Executor] Memory recording failed for rejection ${approvalId}:`, err);
  1233→  }
  1234→
  1235→  // ─── Reject the Execution Contract ────────────────────────────────────────
  1236→  // The contract is marked as rejected but remains permanently searchable.
  1237→  try {
  1238→    if (gateStep) {
  1239→      const stepOutput = JSON.parse(gateStep.output || "{}");
  1240→      if (stepOutput.contractId) {
  1241→        await rejectContract(stepOutput.contractId, rejectedBy, reason);
  1242→      }
  1243→    }
  1244→  } catch (err) {
  1245→    console.error(`[Executor] Contract rejection failed for approval ${approvalId}:`, err);
  1246→  }
  1247→
  1248→  // ─── Update Employee Profile (approval rejected) ─────────────────────────
  1249→  try {
  1250→    await recordProfileEvent({
  1251→      type: "approval_rejected",
  1252→      employeeId: task.employeeId,
  1253→      workspaceId: task.workspaceId,
  1254→      taskId,
  1255→      toolName: approvalRecord?.tool || "",
  1256→    });
  1257→    await recordProfileEvent({
  1258→      type: "task_failed",
  1259→      employeeId: task.employeeId,
  1260→      workspaceId: task.workspaceId,
  1261→      taskId,
  1262→    });
  1263→    // Bookkeeping: contract_rejected is zero-XP (XP already counted via
  1264→    // approval_rejected + task_failed) but nudges the accuracy score.
  1265→    if (gateStep) {
  1266→      const stepOutput = JSON.parse(gateStep.output || "{}");
  1267→      if (stepOutput.contractId) {
  1268→        await recordProfileEvent({
  1269→          type: "contract_rejected",
  1270→          employeeId: task.employeeId,
  1271→          workspaceId: task.workspaceId,
  1272→          taskId,
  1273→        });
  1274→      }
  1275→    }
  1276→  } catch (err) {
  1277→    console.error(`[Executor] Profile update failed for rejection ${approvalId}:`, err);
  1278→  }
  1279→}
