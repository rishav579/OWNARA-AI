   149→  // ─── CSV parsing ─────────────────────────────────────────────────────────
   150→  function parseCsv(text: string): ParsedInvoice[] {
   151→    const lines = text.trim().split(/\r?\n/).filter((l) => l.trim());
   152→    if (lines.length < 2) {
   153→      throw new Error("CSV must have a header row and at least one data row.");
   154→    }
   155→
   156→    const headers = lines[0].split(",").map((h) => h.trim().toLowerCase());
   157→    const required = ["customer_name", "customer_email", "invoice_number", "issue_date", "due_date", "subtotal", "tax"];
   158→    for (const req of required) {
   159→      if (!headers.includes(req)) {
   160→        throw new Error(`Missing required column: ${req}. Expected columns: ${required.join(", ")}`);
   161→      }
   162→    }
   163→
   164→    const invoices: ParsedInvoice[] = [];
   165→    for (let i = 1; i < lines.length; i++) {
   166→      const cols = parseCsvLine(lines[i]);
   167→      if (cols.length < headers.length) {
   168→        throw new Error(`Row ${i + 1} has ${cols.length} columns, expected ${headers.length}.`);
   169→      }
   170→
   171→      const row: Record<string, string> = {};
   172→      headers.forEach((h, idx) => { row[h] = cols[idx]?.trim() || ""; });
   173→
   174→      // Convert amounts from rupees to paise (1 INR = 100 paise)
   175→      const subtotal = Math.round(parseFloat(row.subtotal) * 100);
   176→      const tax = Math.round(parseFloat(row.tax) * 100);
   177→
   178→      if (isNaN(subtotal) || isNaN(tax)) {
   179→        throw new Error(`Row ${i + 1}: subtotal and tax must be numbers.`);
   180→      }
   181→
   182→      invoices.push({
   183→        customerName: row.customer_name,
   184→        customerEmail: row.customer_email,
   185→        invoiceNumber: row.invoice_number,
   186→        issueDate: row.issue_date,
   187→        dueDate: row.due_date,
   188→        subtotal,
   189→        tax,
   190→      });
   191→    }
   192→
   193→    return invoices;
   194→  }
   195→
   196→  function parseCsvLine(line: string): string[] {
   197→    const result: string[] = [];
   198→    let current = "";
   199→    let inQuotes = false;
   200→    for (let i = 0; i < line.length; i++) {
   201→      const char = line[i];
   202→      if (char === '"') {
   203→        inQuotes = !inQuotes;
   204→      } else if (char === "," && !inQuotes) {
   205→        result.push(current);
   206→        current = "";
   207→      } else {
   208→        current += char;
   209→      }
   210→    }
   211→    result.push(current);
   212→    return result;
   213→  }

   241→      }
   242→    };
