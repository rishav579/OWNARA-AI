     1→/**
     2→ * BIHARI AI — Autonomous Learning & Skill Evolution Engine (EMP-002)
     3→ *
     4→ * Transforms the Employee Profile from a static record into a continuously
     5→ * learning workforce. Every completed task generates a deterministic Outcome
     6→ * Evaluation, which drives:
     7→ *
     8→ *   1. Skill Reinforcement  — skills level up from real outcomes, not usage++
     9→ *   2. Pattern Learning     — reusable patterns detected from repeated outcomes
    10→ *   3. Weakness Detection   — repeated failures reduce trust
    11→ *   4. Strength Detection   — repeated successes increase trust
    12→ *   5. Business Outcome History — append-only ledger of measurable impact
    13→ *   6. Career Timeline      — chronological log of every career event
    14→ *   7. Achievement Unlocks  — milestone-based recognition
    15→ *
    16→ * Design principles (NON-NEGOTIABLE):
    17→ *   - NO fake learning. NO random XP. NO LLM-generated scores.
    18→ *   - Everything comes from MEASURABLE BUSINESS OUTCOMES.
    19→ *   - Normalized tables — no JSON blobs for core entities.
    20→ *   - Append-only history. Business outcomes never overwrite.
    21→ *   - Incremental updates only — no expensive scans.
    22→ *   - Generic — works for Finance, HR, Sales Ops, Procurement, etc.
    23→ *
    24→ * Architecture:
    25→ *   The engine is called ONCE per task completion from the executor's
    26→ *   `completeTask()` function. It runs AFTER `recordProfileEvent` so the
    27→ *   profile counters are already updated. The engine then:
    28→ *     1. Loads the task with steps + approvals + finance domain data
    29→ *     2. Builds an OutcomeEvaluation (deterministic scorecard)
    30→ *     3. Applies skill reinforcements based on the outcome
    31→ *     4. Detects patterns (customer payment behavior, reminder effectiveness)
    32→ *     5. Detects weaknesses (high rejection rate, low confidence, etc.)
    33→ *     6. Detects strengths (high approval rate, fast execution, etc.)
    34→ *     7. Records business outcomes (append-only ledger)
    35→ *     8. Appends career timeline entries
    36→ *     9. Checks for achievement unlocks
    37→ *    10. Emits audit events for every evolution
    38→ *
    39→ * All of this happens in best-effort try/catch blocks — learning failures
    40→ * must NEVER break task completion.
    41→ */
    42→

    54→// ─── Skill Reinforcement Definitions ─────────────────────────────────────────
    55→// Maps outcome events to skill reinforcements. GENERIC — no domain logic.
    56→// The "skill" is inferred from the task's tools/keywords (same logic as the
    57→// profile engine's inferAndUpdateSkills). The reinforcement is applied to
    58→// EVERY skill that was exercised in the task.
    59→//
    60→// Reinforcement values are intentionally small so levels grow gradually:
    61→//   +3 = major success (payment recovered, customer responded)
    62→//   +2 = good outcome (approval approved, reminder sent)
    63→//   +1 = minor success (task completed without issues)
    64→//   -1 = minor issue (single rejection, single override)
    65→//   -2 = moderate failure (rollback needed, multiple rejections)
    66→//   -3 = major failure (task failed, capability denied repeatedly)
    67→
    68→interface ReinforcementRule {
    69→  reason: string;
    70→  reinforcement: number;
    71→  notes: string;
    72→}

   107→// ─── 1. Outcome Evaluation ───────────────────────────────────────────────────

   109→/**
   110→ * Evaluates a completed task and produces a deterministic OutcomeEvaluation.
   111→ *
   112→ * This is the SINGLE entry point called from the executor after task
   113→ * completion. It runs the full learning pipeline:
   114→ *   evaluate → reinforce → patterns → weaknesses → strengths → outcomes → timeline → achievements
   115→ *
   116→ * All steps are best-effort. A failure in any step logs an error but does
   117→ * not block the others.
   118→ */
   119→export async function evaluateAndLearn(data: TaskOutcomeData): Promise<void> {
   120→  const { taskId, employeeId, workspaceId } = data;
   121→
   122→  // Idempotency: skip if an evaluation already exists for this task
   123→  const existing = await db.outcomeEvaluation.findUnique({ where: { taskId } });
   124→  if (existing) {
   125→    return;
   126→  }
   127→
   128→  // Load the task with everything we need
   129→  const task = await db.task.findUnique({
   130→    where: { id: taskId },
   131→    include: {
   132→      steps: { orderBy: { stepNumber: "asc" } },
   133→      approvals: true,
   134→      employee: true,
   135→    },
   136→  });
   137→
   138→  if (!task) {
   139→    console.error(`[Learning] Task not found: ${taskId}`);
   140→    return;
   141→  }
   142→
   143→  // Only evaluate completed tasks (not failed/stopped)
   144→  if (task.status !== "completed") {
   145→    return;
   146→  }
   147→
   148→  const evaluation = await buildOutcomeEvaluation(task);
   149→  if (!evaluation) return;
   150→
   151→  // Run the learning pipeline (each step is independent & best-effort)
   152→  try { await reinforceSkillsFromOutcome(task, evaluation); } catch (e) { console.error("[Learning] reinforceSkills failed:", e); }
   153→  try { await detectPatternsFromOutcome(task, evaluation); } catch (e) { console.error("[Learning] detectPatterns failed:", e); }
   154→  try { await detectWeaknesses(employeeId, workspaceId); } catch (e) { console.error("[Learning] detectWeaknesses failed:", e); }
   155→  try { await detectStrengths(employeeId, workspaceId); } catch (e) { console.error("[Learning] detectStrengths failed:", e); }
   156→  try { await recordBusinessOutcomesFromTask(task, evaluation); } catch (e) { console.error("[Learning] recordBusinessOutcomes failed:", e); }
   157→  try { await appendTimelineFromOutcome(task, evaluation); } catch (e) { console.error("[Learning] appendTimeline failed:", e); }
   158→  try { await checkAchievementUnlocks(employeeId, workspaceId, evaluation); } catch (e) { console.error("[Learning] checkAchievements failed:", e); }
   159→}

   161→/**
   162→ * Builds the deterministic OutcomeEvaluation scorecard.
   163→ * Pure function — no side effects except the DB write.
   164→ */
   165→async function buildOutcomeEvaluation(task: any): Promise<any | null> {
   166→  const { id: taskId, workspaceId, employeeId } = task;
   167→
   168→  // ─── Execution quality (from task + steps + approvals) ──────────────────
   169→  const stepCount = task.steps.length;
   170→  const toolCallCount = task.steps.filter((s: any) => s.stepType === "tool_call").length;
   171→  const approvalGateCount = task.steps.filter((s: any) => s.stepType === "approval_gate").length;
   172→  const approvals = task.approvals || [];
   173→  const approvalRejections = approvals.filter((a: any) => a.decision === "rejected").length;
   174→  const humanOverrides = approvals.filter((a: any) => a.originalAction !== null).length;
   175→  const tokenUsage = task.tokenUsage || 0;
   176→
   177→  // Execution time
   178→  const executionTimeMs =
   179→    task.startedAt && task.completedAt
   180→      ? new Date(task.completedAt).getTime() - new Date(task.startedAt).getTime()
   181→      : 0;
   182→
   183→  // Capability denials + step failures (scan step outputs)
   184→  let capabilityDenials = 0;
   185→  let stepFailureCount = 0;
   186→  for (const step of task.steps) {
   187→    if (step.status === "failed") stepFailureCount++;
   188→    if (step.output && step.output.includes("capability_denied")) {
   189→      capabilityDenials++;
   190→    }
   191→  }
   192→
   193→  // ─── Business outcome (from finance domain — generic approach) ──────────
   194→  const { paymentReceived, paymentAmount, invoiceResolved, customerResponded, reminderSentCount, invoiceIds, customerIds } =
   195→    await detectBusinessOutcomes(task, workspaceId);
   196→
   197→  // ─── SLA ────────────────────────────────────────────────────────────────
   198→  // SLA target: 24 hours for finance tasks (configurable per role in future)
   199→  const slaTargetHours = 24;
   200→  const slaActualHours = executionTimeMs / (1000 * 60 * 60);
   201→  const slaAchieved = slaActualHours <= slaTargetHours && stepFailureCount === 0;
   202→
   203→  // ─── Confidence accuracy ────────────────────────────────────────────────
   204→  const completedSteps = task.steps.filter((s: any) => s.status === "completed" && s.confidence !== null);
   205→  const avgConfidence =
   206→    completedSteps.length > 0
   207→      ? completedSteps.reduce((sum: number, s: any) => sum + (s.confidence || 0), 0) / completedSteps.length
   208→      : 0.85;
   209→  const expectedSuccess = avgConfidence >= 0.7;
   210→  const actualSuccess = task.status === "completed" && approvalRejections === 0;
   211→  const confidenceAccuracy = expectedSuccess === actualSuccess ? 1.0 : 0.0;
   212→
   213→  // ─── Human corrections ──────────────────────────────────────────────────
   214→  const humanCorrections = approvalRejections + humanOverrides + capabilityDenials;
   215→
   216→  // ─── Quality score (deterministic formula, 0-100) ───────────────────────
   217→  let qualityScore = 0;
   218→  if (actualSuccess) qualityScore += 40;
   219→  if (humanCorrections === 0) qualityScore += 20;
   220→  else if (humanCorrections === 1) qualityScore += 10;
   221→  if (slaAchieved) qualityScore += 15;
   222→  if (confidenceAccuracy === 1.0) qualityScore += 15;
   223→  if (paymentReceived || customerResponded) qualityScore += 10;
   224→
   225→  // ─── Outcome summary (deterministic, built from facts) ──────────────────
   226→  const outcomeParts: string[] = [];
   227→  outcomeParts.push(`Task completed with ${stepCount} steps (${toolCallCount} tool calls, ${approvalGateCount} approval gates)`);
   228→  if (approvalRejections > 0) outcomeParts.push(`${approvalRejections} approval rejection(s)`);
   229→  if (humanOverrides > 0) outcomeParts.push(`${humanOverrides} human override(s)`);
   230→  if (capabilityDenials > 0) outcomeParts.push(`${capabilityDenials} capability denial(s)`);
   231→  if (reminderSentCount > 0) outcomeParts.push(`${reminderSentCount} reminder(s) sent`);
   232→  if (paymentReceived) outcomeParts.push(`Payment received: ₹${(paymentAmount / 100).toFixed(2)}`);
   233→  if (customerResponded) outcomeParts.push("Customer responded to reminder");
   234→  if (slaAchieved) outcomeParts.push(`SLA achieved (${slaActualHours.toFixed(1)}h / ${slaTargetHours}h)`);
   235→  else outcomeParts.push(`SLA missed (${slaActualHours.toFixed(1)}h / ${slaTargetHours}h)`);
   236→  const outcomeSummary = outcomeParts.join(". ");
   237→
   238→  // ─── Persist the evaluation ─────────────────────────────────────────────
   239→  const evaluation = await db.outcomeEvaluation.create({
   240→    data: {
   241→      workspaceId,
   242→      employeeId,
   243→      taskId,
   244→      stepCount,
   245→      toolCallCount,
   246→      approvalGateCount,
   247→      approvalRejections,
   248→      humanOverrides,
   249→      capabilityDenials,
   250→      tokenUsage,
   251→      executionTimeMs,
   252→      rollbackNeeded: false,
   253→      paymentReceived,
   254→      paymentAmount,
   255→      invoiceResolved,
   256→      customerResponded,
   257→      reminderSentCount,
   258→      slaAchieved,
   259→      slaTargetHours,
   260→      slaActualHours: Math.round(slaActualHours * 100) / 100,
   261→      errorCount: stepFailureCount,
   262→      stepFailureCount,
   263→      expectedSuccess,
   264→      actualSuccess,
   265→      confidenceAccuracy,
   266→      qualityScore,
   267→      humanCorrections,
   268→      outcomeSummary,
   269→    },
   270→  });
   271→
   272→  // Audit: profile_updated (the outcome evaluation IS a profile update)
   273→  await emitAudit(workspaceId, employeeId, "profile_updated", "outcome_evaluation", evaluation.id, {
   274→    taskId,
   275→    qualityScore: String(qualityScore),
   276→    actualSuccess: String(actualSuccess),
   277→    humanCorrections: String(humanCorrections),
   278→    paymentReceived: String(paymentReceived),
   279→    paymentAmount: String(paymentAmount),
   280→  });
   281→
   282→  // Attach the computed (non-persisted) fields for downstream functions.
   283→  // These are not DB columns but are needed by recordBusinessOutcomesFromTask,
   284→  // detectPatternsFromOutcome, etc.
   285→  return {
   286→    ...evaluation,
   287→    invoiceIds,
   288→    customerIds,
   289→  };
   290→}
   291→
   292→/**
   293→ * Detects business outcomes for a task by examining finance domain data.
   294→ * GENERIC approach: scan the task's steps for invoice IDs and check if
   295→ * payments were received or customers responded. Returns empty results for
   296→ * non-finance tasks (which is correct — they have no finance outcomes).
   297→ */
   298→async function detectBusinessOutcomes(task: any, workspaceId: string): Promise<{
   299→  paymentReceived: boolean;
   300→  paymentAmount: number;
   301→  invoiceResolved: boolean;
   302→  customerResponded: boolean;
   303→  reminderSentCount: number;
   304→  invoiceIds: string[];
   305→  customerIds: string[];
   306→}> {
   307→  const result = {
   308→    paymentReceived: false,
   309→    paymentAmount: 0,
   310→    invoiceResolved: false,
   311→    customerResponded: false,
   312→    reminderSentCount: 0,
   313→    invoiceIds: [] as string[],
   314→    customerIds: [] as string[],
   315→  };
   316→
   317→  const invoiceIds = new Set<string>();
   318→  const customerIds = new Set<string>();
   319→
   320→  // Scan steps for invoiceId / customerId / reminderId in their input/output JSON
   321→  for (const step of task.steps) {
   322→    try {
   323→      const input = JSON.parse(step.input || "{}");
   324→      if (input.invoiceIds && Array.isArray(input.invoiceIds)) {
   325→        input.invoiceIds.forEach((id: string) => invoiceIds.add(id));
   326→      }
   327→      if (input.invoiceId) invoiceIds.add(input.invoiceId);
   328→      if (input.customerId) customerIds.add(input.customerId);
   329→      if (input.tool === "send_reminder" || input.tool === "generate_reminder") {
   330→        result.reminderSentCount++;
   331→      }
   332→    } catch {}
   333→
   334→    try {
   335→      const output = JSON.parse(step.output || "{}");
   336→      if (output.invoiceId) invoiceIds.add(output.invoiceId);
   337→      if (output.customerId) customerIds.add(output.customerId);
   338→      if (output.reminderId) {
   339→        // This step created a reminder — check if it was sent & responded
   340→        const reminder = await db.reminder.findUnique({ where: { id: output.reminderId } });
   341→        if (reminder) {
   342→          if (reminder.status === "sent") result.reminderSentCount = Math.max(result.reminderSentCount, 1);
   343→          if (reminder.status === "responded" || reminder.respondedAt) {
   344→            result.customerResponded = true;
   345→            customerIds.add(reminder.customerId);
   346→          }
   347→          invoiceIds.add(reminder.invoiceId);
   348→          customerIds.add(reminder.customerId);
   349→        }
   350→      }
   351→    } catch {}
   352→  }
   353→
   354→  result.invoiceIds = Array.from(invoiceIds);
   355→  result.customerIds = Array.from(customerIds);
   356→
   357→  // Check for payments received on these invoices AFTER the task started
   358→  if (invoiceIds.size > 0) {
   359→    const taskStartTime = task.startedAt || task.createdAt;
   360→    const payments = await db.payment.findMany({
   361→      where: {
   362→        workspaceId,
   363→        invoiceId: { in: result.invoiceIds },
   364→        paymentDate: { gte: taskStartTime },
   365→        status: "completed",
   366→      },
   367→    });
   368→
   369→    if (payments.length > 0) {
   370→      result.paymentReceived = true;
   371→      result.paymentAmount = payments.reduce((sum, p) => sum + p.amount, 0);
   372→
   373→      const invoices = await db.invoice.findMany({
   374→        where: { id: { in: result.invoiceIds } },
   375→        select: { id: true, outstanding: true },
   376→      });
   377→      result.invoiceResolved = invoices.length > 0 && invoices.every((inv) => inv.outstanding <= 0);
   378→    }
   379→  }
   380→
   381→  return result;
   382→}

   384→// ─── 2. Skill Reinforcement Engine ───────────────────────────────────────────
   385→
   386→/**
   387→ * Applies skill reinforcements based on the outcome evaluation.
   388→ * Every skill exercised in the task gets reinforced (positively or negatively).
   389→ */
   390→async function reinforceSkillsFromOutcome(task: any, evaluation: any): Promise<void> {
   391→  const { employeeId, workspaceId, taskId } = task;
   392→  const employee = task.employee;
   393→  if (!employee) return;
   394→
   395→  // Determine which skills were exercised in this task
   396→  const skillDefs = getSkillsForRole(employee.role);
   397→  const taskText = `${task.title} ${task.description}`.toLowerCase();
   398→  const toolsUsed = new Set<string>();
   399→  for (const step of task.steps) {
   400→    try {
   401→      const input = JSON.parse(step.input);
   402→      if (input.tool) toolsUsed.add(input.tool);
   403→    } catch {}
   404→  }
   405→
   406→  const exercisedSkills: string[] = [];
   407→  for (const skillDef of skillDefs) {
   408→    const usedByKeyword = skillDef.keywords.some((kw) => taskText.includes(kw.toLowerCase()));
   409→    const usedByTool = skillDef.tools.some((t) => toolsUsed.has(t));
   410→    if (usedByKeyword || usedByTool) {
   411→      exercisedSkills.push(skillDef.name);
   412→    }
   413→  }
   414→
   415→  if (exercisedSkills.length === 0) return;
   416→
   417→  // Build reinforcement rules based on the outcome
   418→  const rules = buildReinforcementRules(evaluation);
   419→
   420→  // Apply each rule to each exercised skill
   421→  for (const skillName of exercisedSkills) {
   422→    for (const rule of rules) {
   423→      await db.skillReinforcement.create({
   424→        data: {
   425→          workspaceId,
   426→          employeeId,
   427→          skillName,
   428→          taskId,
   429→          outcomeId: evaluation.id,
   430→          reason: rule.reason,
   431→          reinforcement: rule.reinforcement,
   432→          notes: rule.notes,
   433→        },
   434→      });
   435→    }
   436→
   437→    // Recompute the skill's level from the reinforcement ledger
   438→    await recomputeSkillLevel(employeeId, workspaceId, skillName, evaluation);
   439→  }
   440→}
   441→
   442→/**
   443→ * Builds the list of reinforcement rules based on the outcome evaluation.
   444→ */
   445→function buildReinforcementRules(evaluation: any): ReinforcementRule[] {
   446→  const rules: ReinforcementRule[] = [];
   447→
   448→  if (evaluation.actualSuccess) {
   449→    rules.push({ reason: "task_completed", reinforcement: 1, notes: "Task completed without failures" });
   450→  }
   451→  if (evaluation.paymentReceived) {
   452→    rules.push({
   453→      reason: "payment_recovered",
   454→      reinforcement: 3,
   455→      notes: `Payment of ₹${(evaluation.paymentAmount / 100).toFixed(2)} received after this task`,
   456→    });
   457→  }
   458→  if (evaluation.customerResponded) {
   459→    rules.push({ reason: "customer_responded", reinforcement: 2, notes: "Customer responded to a reminder sent in this task" });
   460→  }
   461→  if (evaluation.slaAchieved) {
   462→    rules.push({ reason: "sla_achieved", reinforcement: 1, notes: `Completed within SLA (${evaluation.slaActualHours}h / ${evaluation.slaTargetHours}h)` });
   463→  }
   464→  if (evaluation.humanCorrections === 0 && evaluation.qualityScore >= 80) {
   465→    rules.push({ reason: "high_quality_execution", reinforcement: 2, notes: `Quality score ${evaluation.qualityScore}/100 with zero human corrections` });
   466→  }
   467→  if (evaluation.approvalRejections > 0) {
   468→    rules.push({
   469→      reason: "approval_rejected",
   470→      reinforcement: -1 * Math.min(3, evaluation.approvalRejections),
   471→      notes: `${evaluation.approvalRejections} approval rejection(s)`,
   472→    });
   473→  }
   474→  if (evaluation.humanOverrides > 0) {
   475→    rules.push({
   476→      reason: "human_override",
   477→      reinforcement: -1 * Math.min(2, evaluation.humanOverrides),
   478→      notes: `${evaluation.humanOverrides} human override(s)`,
   479→    });
   480→  }
   481→  if (evaluation.capabilityDenials > 0) {
   482→    rules.push({
   483→      reason: "capability_denied",
   484→      reinforcement: -2 * evaluation.capabilityDenials,
   485→      notes: `${evaluation.capabilityDenials} capability denial(s)`,
   486→    });
   487→  }
   488→  if (!evaluation.slaAchieved && evaluation.slaActualHours > 0) {
   489→    rules.push({ reason: "sla_missed", reinforcement: -1, notes: `SLA missed (${evaluation.slaActualHours}h / ${evaluation.slaTargetHours}h)` });
   490→  }
   491→  if (evaluation.confidenceAccuracy === 0.0) {
   492→    rules.push({ reason: "confidence_inaccurate", reinforcement: -1, notes: "Employee's confidence did not match the actual outcome" });
   493→  }
   494→  if (evaluation.stepFailureCount > 0) {
   495→    rules.push({
   496→      reason: "step_failure",
   497→      reinforcement: -1 * Math.min(2, evaluation.stepFailureCount),
   498→      notes: `${evaluation.stepFailureCount} step failure(s)`,
   499→    });
   500→  }
   501→
   502→  return rules;
   503→}

   505→/**
   506→ * Recomputes a skill's level from the reinforcement ledger.
   507→ * Level formula:
   508→ *   totalReinforcement = sum of all reinforcement values
   509→ *   level = clamp(1, 10, 1 + floor(max(0, totalReinforcement) / 5))
   510→ *   confidence = clamp(0.5, 0.99, 0.5 + (positive_count * 0.03) - (negative_count * 0.05))
   511→ *
   512→ * If the level increased, emit a skill_promoted audit event + timeline entry.
   513→ */
   514→async function recomputeSkillLevel(
   515→  employeeId: string,
   516→  workspaceId: string,
   517→  skillName: string,
   518→  evaluation: any
   519→): Promise<void> {
   520→  const reinforcements = await db.skillReinforcement.findMany({
   521→    where: { employeeId, skillName },
   522→    orderBy: { createdAt: "asc" },
   523→  });
   524→
   525→  if (reinforcements.length === 0) return;
   526→
   527→  const totalReinforcement = reinforcements.reduce((sum, r) => sum + r.reinforcement, 0);
   528→  const positiveCount = reinforcements.filter((r) => r.reinforcement > 0).length;
   529→  const negativeCount = reinforcements.filter((r) => r.reinforcement < 0).length;
   530→
   531→  const newLevel = Math.max(1, Math.min(10, 1 + Math.floor(Math.max(0, totalReinforcement) / 5)));
   532→  const newConfidence = Math.max(0.5, Math.min(0.99, 0.5 + positiveCount * 0.03 - negativeCount * 0.05));
   533→  const newUsageCount = reinforcements.length;
   534→
   535→  const existing = await db.employeeSkill.findUnique({
   536→    where: { employeeId_name: { employeeId, name: skillName } },
   537→  });
   538→
   539→  const previousLevel = existing?.level || 1;
   540→
   541→  if (existing) {
   542→    await db.employeeSkill.update({
   543→      where: { id: existing.id },
   544→      data: {
   545→        level: newLevel,
   546→        confidence: newConfidence,
   547→        usageCount: newUsageCount,
   548→        lastUsedAt: new Date(),
   549→      },
   550→    });
   551→  } else {
   552→    await db.employeeSkill.create({
   553→      data: {
   554→        employeeId,
   555→        workspaceId,
   556→        name: skillName,
   557→        level: newLevel,
   558→        confidence: newConfidence,
   559→        usageCount: newUsageCount,
   560→        lastUsedAt: new Date(),
   561→      },
   562→    });
   563→  }
   564→
   565→  // If level increased, emit skill_promoted audit + timeline
   566→  if (newLevel > previousLevel) {
   567→    await emitAudit(workspaceId, employeeId, "skill_promoted", "skill", skillName, {
   568→      skillName,
   569→      previousLevel: String(previousLevel),
   570→      newLevel: String(newLevel),
   571→      totalReinforcement: String(totalReinforcement),
   572→    });
   573→
   574→    await db.careerTimelineEntry.create({
   575→      data: {
   576→        workspaceId,
   577→        employeeId,
   578→        entryType: "skill_promoted",
   579→        title: `${skillName} → Level ${newLevel}`,
   580→        description: `Skill "${skillName}" promoted from Level ${previousLevel} to Level ${newLevel} based on ${reinforcements.length} reinforcements (net +${totalReinforcement}).`,
   581→        metadata: JSON.stringify({ skillName, previousLevel, newLevel, totalReinforcement }),
   582→        levelAtTime: 0,
   583→        xpAtTime: 0,
   584→        trustAtTime: 0,
   585→        taskId: evaluation.taskId,
   586→        outcomeId: evaluation.id,
   587→      },
   588→    });
   589→  } else if (!existing && newLevel === 1) {
   590→    // First time this skill was learned
   591→    await db.careerTimelineEntry.create({
   592→      data: {
   593→        workspaceId,
   594→        employeeId,
   595→        entryType: "skill_learned",
   596→        title: `Learned: ${skillName}`,
   597→        description: `New skill "${skillName}" acquired with initial reinforcement +${totalReinforcement}.`,
   598→        metadata: JSON.stringify({ skillName, initialReinforcement: totalReinforcement }),
   599→        levelAtTime: 0,
   600→        xpAtTime: 0,
   601→        trustAtTime: 0,
   602→        taskId: evaluation.taskId,
   603→        outcomeId: evaluation.id,
   604→      },
   605→    });
   606→  }
   607→
   608→  // Sync the skills JSON on the profile (for quick access by the UI)
   609→  await syncSkillsJson(employeeId);
   610→}

   636→async function detectPatternsFromOutcome(task: any, evaluation: any): Promise<void> {
   637→  const { employeeId, workspaceId } = task;
   638→
   639→  // ─── Pattern 1: Customer payment behavior ───────────────────────────────
   640→  if (evaluation.paymentReceived && evaluation.invoiceIds.length > 0) {
   641→    for (const invoiceId of evaluation.invoiceIds) {
   642→      const invoice = await db.invoice.findUnique({
   643→        where: { id: invoiceId },
   644→        include: { customer: true, reminders: { orderBy: { createdAt: "asc" } } },
   645→      });
   646→      if (!invoice) continue;
   647→
   648→      const reminderCount = invoice.reminders.filter((r) => r.status === "sent").length;
   649→      if (reminderCount === 0) continue;
   650→
   651→      const patternKey = reminderCount === 1 ? "pays_after_first_reminder" : `pays_after_${reminderCount}_reminders`;
   652→      const patternDescription = `${invoice.customer.name} paid invoice ${invoice.invoiceNumber} after ${reminderCount} reminder(s).`;
   653→
   654→      await upsertPattern({
   655→        workspaceId,
   656→        employeeId,
   657→        patternType: "customer_payment_behavior",
   658→        entityType: "customer",
   659→        entityId: invoice.customerId,
   660→        entityLabel: invoice.customer.name,
   661→        pattern: patternKey,
   662→        description: patternDescription,
   663→        confidence: Math.min(0.99, 0.5 + 0.1 * reminderCount),
   664→      });
   665→    }
   666→  }
   667→
   668→  // ─── Pattern 2: Reminder effectiveness ──────────────────────────────────
   669→  if (evaluation.customerResponded && evaluation.customerIds.length > 0) {
   670→    for (const customerId of evaluation.customerIds) {
   671→      const customer = await db.customer.findUnique({ where: { id: customerId } });
   672→      if (!customer) continue;
   673→
   674→      await upsertPattern({
   675→        workspaceId,
   676→        employeeId,
   677→        patternType: "reminder_effectiveness",
   678→        entityType: "customer",
   679→        entityId: customerId,
   680→        entityLabel: customer.name,
   681→        pattern: "responds_to_reminders",
   682→        description: `${customer.name} responds to reminder emails.`,
   683→        confidence: 0.7,
   684→      });
   685→    }
   686→  }
   687→
   688→  // ─── Pattern 3: Invoice risk indicator ──────────────────────────────────
   689→  if (evaluation.approvalRejections > 0 && evaluation.invoiceIds.length > 0) {
   690→    for (const invoiceId of evaluation.invoiceIds) {
   691→      const invoice = await db.invoice.findUnique({
   692→        where: { id: invoiceId },
   693→        include: { customer: true },
   694→      });
   695→      if (!invoice) continue;
   696→
   697→      // Invoices above ₹5 lakh (5000000 paise = ₹50000... actually ₹5L = 500000*100 = 50000000)
   698→      if (invoice.total >= 5000000) {
   699→        await upsertPattern({
   700→          workspaceId,
   701→          employeeId,
   702→          patternType: "invoice_risk_indicator",
   703→          entityType: "invoice",
   704→          entityId: invoiceId,
   705→          entityLabel: invoice.invoiceNumber,
   706→          pattern: "high_value_invoice_needs_manager_review",
   707→          description: `Invoice ${invoice.invoiceNumber} (₹${(invoice.total / 100).toFixed(2)}) required manager review.`,
   708→          confidence: 0.6,
   709→        });
   710→      }
   711→    }
   712→  }
   713→}

   804→async function detectWeaknesses(employeeId: string, workspaceId: string): Promise<void> {
   805→  const recentEvaluations = await db.outcomeEvaluation.findMany({
   806→    where: { employeeId },
   807→    orderBy: { createdAt: "desc" },
   808→    take: 10,
   809→  });
   810→
   811→  if (recentEvaluations.length < 3) return;
   812→
   813→  const totalTasks = recentEvaluations.length;
   814→  const tasksWithRejections = recentEvaluations.filter((e) => e.approvalRejections > 0).length;
   815→  const tasksWithOverrides = recentEvaluations.filter((e) => e.humanOverrides > 0).length;
   816→  const tasksWithSlaMisses = recentEvaluations.filter((e) => !e.slaAchieved).length;
   817→  const tasksWithDenials = recentEvaluations.filter((e) => e.capabilityDenials > 0).length;
   818→  const avgConfidenceAccuracy = recentEvaluations.reduce((s, e) => s + e.confidenceAccuracy, 0) / totalTasks;
   819→  const avgExecutionTime = recentEvaluations.reduce((s, e) => s + e.executionTimeMs, 0) / totalTasks;
   820→  const avgSlaTarget = recentEvaluations[0]?.slaTargetHours || 24;
   821→
   822→  const weaknessChecks = [
   823→    {
   824→      type: "high_rejection_rate",
   825→      description: `${tasksWithRejections}/${totalTasks} recent tasks had approval rejections (${Math.round((tasksWithRejections / totalTasks) * 100)}%)`,
   826→      metricValue: tasksWithRejections / totalTasks,
   827→      threshold: 0.3,
   828→      trustImpact: -5.0,
   829→      inverted: false,
   830→    },
   831→    {
   832→      type: "frequent_human_edits",
   833→      description: `${tasksWithOverrides}/${totalTasks} recent tasks had human overrides (${Math.round((tasksWithOverrides / totalTasks) * 100)}%)`,
   834→      metricValue: tasksWithOverrides / totalTasks,
   835→      threshold: 0.4,
   836→      trustImpact: -3.0,
   837→      inverted: false,
   838→    },
   839→    {
   840→      type: "repeated_sla_misses",
   841→      description: `${tasksWithSlaMisses}/${totalTasks} recent tasks missed SLA (${Math.round((tasksWithSlaMisses / totalTasks) * 100)}%)`,
   842→      metricValue: tasksWithSlaMisses / totalTasks,
   843→      threshold: 0.3,
   844→      trustImpact: -4.0,
   845→      inverted: false,
   846→    },
   847→    {
   848→      type: "low_confidence_accuracy",
   849→      description: `Average confidence accuracy is ${(avgConfidenceAccuracy * 100).toFixed(0)}% (threshold 50%)`,
   850→      metricValue: avgConfidenceAccuracy,
   851→      threshold: 0.5,
   852→      trustImpact: -3.0,
   853→      inverted: true,
   854→    },
   855→    {
   856→      type: "slow_execution",
   857→      description: `Average execution time is ${(avgExecutionTime / (1000 * 60 * 60)).toFixed(1)}h (threshold ${avgSlaTarget * 2}h)`,
   858→      metricValue: avgExecutionTime / (1000 * 60 * 60),
   859→      threshold: avgSlaTarget * 2,
   860→      trustImpact: -2.0,
   861→      inverted: false, // higher execution time = worse
   862→    },
   863→  ];
   864→
   865→  if (tasksWithDenials > 1) {
   866→    weaknessChecks.push({
   867→      type: "high_capability_denials",
   868→      description: `${tasksWithDenials} capability denial(s) in recent ${totalTasks} tasks`,
   869→      metricValue: tasksWithDenials,
   870→      threshold: 1,
   871→      trustImpact: -6.0,
   872→      inverted: false, // higher denials = worse
   873→    });
   874→  }
   875→
   876→  for (const check of weaknessChecks) {
   877→    const triggered = check.inverted
   878→      ? check.metricValue < check.threshold
   879→      : check.metricValue > check.threshold;
   880→
   881→    if (triggered) {
   882→      await upsertWeakness({
   883→        workspaceId,
   884→        employeeId,
   885→        weaknessType: check.type,
   886→        description: check.description,
   887→        metricValue: check.metricValue,
   888→        threshold: check.threshold,
   889→        trustImpact: check.trustImpact,
   890→      });
   891→    } else {
   892→      await resolveWeaknessIfApplicable(workspaceId, employeeId, check.type);
   893→    }
   894→  }
   895→}

  1345→/**
  1346→ * Checks for achievement unlocks based on the employee's current state.
  1347→ */
  1348→async function checkAchievementUnlocks(
  1349→  employeeId: string,
  1350→  workspaceId: string,
  1351→  evaluation: any
  1352→): Promise<void> {
  1353→  await seedDefaultAchievements(workspaceId);
  1354→
  1355→  const profile = await db.employeeProfile.findUnique({ where: { employeeId } });
  1356→  if (!profile) return;
  1357→
  1358→  const [achievements, unlocked] = await Promise.all([
  1359→    db.achievement.findMany({ where: { workspaceId } }),
  1360→    db.employeeAchievement.findMany({ where: { employeeId }, select: { achievementId: true } }),
  1361→  ]);
  1362→  const unlockedIds = new Set(unlocked.map((u) => u.achievementId));
  1363→
  1364→  for (const achievement of achievements) {
  1365→    if (unlockedIds.has(achievement.id)) continue;
  1366→
  1367→    let shouldUnlock = false;
  1368→    let evidence = "";
  1369→
  1370→    switch (achievement.triggerType) {
  1371→      case "task_count":
  1372→        if (profile.completedTasks >= achievement.triggerThreshold) {
  1373→          shouldUnlock = true;
  1374→          evidence = `Completed ${profile.completedTasks} tasks (threshold: ${achievement.triggerThreshold})`;
  1375→        }
  1376→        break;
  1377→      case "money_recovered":
  1378→        if (profile.moneyRecovered >= achievement.triggerThreshold) {
  1379→          shouldUnlock = true;
  1380→          evidence = `Recovered ₹${(profile.moneyRecovered / 100).toFixed(2)} (threshold: ₹${(achievement.triggerThreshold / 100).toFixed(2)})`;
  1381→        }
  1382→        break;
  1383→      case "level_reached":
  1384→        if (profile.level >= achievement.triggerThreshold) {
  1385→          shouldUnlock = true;
  1386→          evidence = `Reached Level ${profile.level} (threshold: Level ${achievement.triggerThreshold})`;
  1387→        }
  1388→        break;
  1389→      case "skill_level":
  1390→        if (achievement.triggerSkillName) {
  1391→          const skill = await db.employeeSkill.findUnique({
  1392→            where: { employeeId_name: { employeeId, name: achievement.triggerSkillName } },
  1393→          });
  1394→          if (skill && skill.level >= achievement.triggerThreshold) {
  1395→            shouldUnlock = true;
  1396→            evidence = `Skill "${achievement.triggerSkillName}" reached Level ${skill.level} (threshold: Level ${achievement.triggerThreshold})`;
  1397→          }
  1398→        }
  1399→        break;
  1400→      case "perfect_quality": {
  1401→        const perfectCount = await db.outcomeEvaluation.count({
  1402→          where: { employeeId, qualityScore: 100 },
  1403→        });
  1404→        if (perfectCount >= achievement.triggerThreshold) {
  1405→          shouldUnlock = true;
  1406→          evidence = `${perfectCount} tasks with perfect quality score (threshold: ${achievement.triggerThreshold})`;
  1407→        }
  1408→        break;
  1409→      }
  1410→      case "streak": {
  1411→        const recentEvals = await db.outcomeEvaluation.findMany({
  1412→          where: { employeeId },
  1413→          orderBy: { createdAt: "desc" },
  1414→          take: achievement.triggerThreshold,
  1415→        });
  1416→        const allSuccess = recentEvals.length >= achievement.triggerThreshold &&
  1417→          recentEvals.every((e) => e.actualSuccess);
  1418→        if (allSuccess) {
  1419→          shouldUnlock = true;
  1420→          evidence = `${achievement.triggerThreshold} consecutive successful tasks`;
  1421→        }
  1422→        break;
  1423→      }
  1424→    }
  1425→
  1426→    if (shouldUnlock) {
  1427→      await db.employeeAchievement.create({
  1428→        data: {
  1429→          workspaceId,
  1430→          employeeId,
  1431→          achievementId: achievement.id,
  1432→          evidence,
  1433→          taskId: evaluation.taskId,
  1434→          outcomeId: evaluation.id,
  1435→        },
  1436→      });
  1437→
  1438→      await emitAudit(workspaceId, employeeId, "achievement_unlocked", "achievement", achievement.id, {
  1439→        code: achievement.code,
  1440→        name: achievement.name,
  1441→        description: achievement.description,
  1442→        trustImpact: String(achievement.trustImpact),
  1443→        evidence,
  1444→      });
  1445→
  1446→      await db.careerTimelineEntry.create({
  1447→        data: {
  1448→          workspaceId,
  1449→          employeeId,
  1450→          entryType: "achievement_unlocked",
  1451→          title: `Achievement: ${achievement.name}`,
  1452→          description: `${achievement.description} — ${evidence}`,
  1453→          metadata: JSON.stringify({
  1454→            code: achievement.code,
  1455→            name: achievement.name,
  1456→            trustImpact: achievement.trustImpact,
  1457→            evidence,
  1458→          }),
  1459→          levelAtTime: profile.level,
  1460→          xpAtTime: profile.experiencePoints,
  1461→          trustAtTime: profile.trustScore,
  1462→          taskId: evaluation.taskId,
  1463→          outcomeId: evaluation.id,
  1464→        },
  1465→      });
  1466→    }
  1467→  }
  1468→}
