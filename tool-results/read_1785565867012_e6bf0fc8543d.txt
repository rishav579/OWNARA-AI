   560→// ─── Employee Memory Layer ──────────────────────────────────────────────────
   561→// Generic persistent memory for ALL AI Employees (Finance, HR, Sales, etc.).
   562→// Records learned patterns, customer behavior, approval outcomes, and
   563→// strategy effectiveness. The employee retrieves this before acting and
   564→// updates it after every task.
   565→
   566→model EmployeeMemory {
   567→  id                String   @id @default(cuid())
   568→  workspaceId      String
   569→  employeeId       String
   570→  memoryType       String   // customer_behavior | payment_habits | negotiation_outcome | approval_history | manager_feedback | communication_preference | dispute_history | customer_promise | collection_strategy | strategy_effectiveness
   571→  entityType       String   // customer | invoice | task | approval | generic
   572→  entityId         String   // ID of the entity (customer ID, invoice ID, etc.)
   573→  entityLabel      String   // Human-readable label (e.g., customer name)
   574→  key              String   // Specific key within type (e.g., "average_payment_delay", "preferred_channel")
   575→  value            String   // JSON — the memory content
   576→  confidence       Float    @default(0.5) // 0.0–1.0, increases with reinforcement
   577→  source           String   @default("task") // task | approval | observation | human_feedback
   578→  reinforcementCount Int    @default(1) // how many times this memory has been confirmed

   589→// ─── Execution Contract Engine ──────────────────────────────────────────────
   590→// Immutable contracts generated before ANY irreversible tool execution.
   591→// Every approval references a Contract (not mutable task state).
   592→// Contracts are versioned — modifications create V2, V3, etc.
   593→// Approved contracts become immutable. Rejected contracts remain searchable.
   594→
   595→model ExecutionContract {
   596→  id              String   @id @default(cuid())
   597→  workspaceId     String
   598→  taskId          String
   599→  employeeId      String
   600→  approvalId      String?  // linked approval (null until approval is created)
   601→
   602→  // Contract identity
   603→  contractNumber  String   // e.g., "EC-0001"
   604→  version         Int      @default(1) // 1, 2, 3... each modification = new version
   605→  parentContractId String? // ID of the previous version (for version chain)
   606→
   607→  // Contract content (immutable once approved)
   608→  status          String   @default("draft") // draft | pending_approval | approved | rejected | expired | superseded
   609→
   610→  // The 17 required fields
   611→  goal            String   // What the employee is trying to achieve
   612→  proposedAction  String   // JSON — the exact action to be executed
   613→  confidence      Float    // 0.0–1.0 — employee's confidence
   614→  evidence        String   @default("[]") // JSON array of evidence items
   615→  memoriesUsed    String   @default("[]") // JSON array of memory IDs used
   616→  policiesUsed    String   @default("[]") // JSON array of policy codes
   617→  businessImpact  String   // text — financial/reputational/operational impact
   618→  affectedSystems String   @default("[]") // JSON array of systems affected (e.g., ["invoices", "reminders"])
   619→  rollbackPlan    String   // text — how to undo this action if needed
   620→  estimatedBusinessOutcome String // text — expected business result
   621→  estimatedTokenCost Int   @default(0) // estimated tokens for this action
   622→  estimatedExecutionTime Int @default(0) // estimated milliseconds
   623→  requiredAuthority String @default("owner") // owner | manager | department_head
   624→
   625→  // Hash (SHA-256 of the canonical contract content)
   626→  contractHash    String   @unique
   627→
   628→  // Lifecycle
   629→  generatedAt     DateTime @default(now())
   630→  approvedAt      DateTime?
   631→  approvedBy      String?
   632→  rejectedAt      DateTime?
   633→  rejectedBy      String?
   634→  rejectionReason String?
   635→  supersededAt    DateTime?
   636→
   637→  createdAt       DateTime @default(now())
   638→  updatedAt       DateTime @updatedAt
   639→
   640→  workspace       Workspace @relation(fields: [workspaceId], references: [id])
   641→
   642→  @@unique([workspaceId, contractNumber, version])
   643→  @@index([workspaceId, status])
   644→  @@index([taskId])
   645→  @@index([employeeId])
   646→  @@index([contractHash])
   647→}
