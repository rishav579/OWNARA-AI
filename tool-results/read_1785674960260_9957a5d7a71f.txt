   122→   122→  // ─── CSV parsing ─────────────────────────────────────────────────────────
   123→   123→  function parseCsv(text: string): ParsedInvoice[] {
   124→   124→    const lines = text.trim().split(/\r?\n/).filter((l) => l.trim());
   125→   125→    if (lines.length < 2) {
   126→   126→      throw new Error("CSV must have a header row and at least one data row.");
   127→   127→    }
   128→   128→
   129→   129→    const headers = lines[0].split(",").map((h) => h.trim().toLowerCase());
   130→   130→    const required = ["customer_name", "customer_email", "invoice_number", "issue_date", "due_date", "subtotal", "tax"];
   131→   131→    for (const req of required) {
   132→   132→      if (!headers.includes(req)) {
   133→   133→        throw new Error(`Missing required column: ${req}. Expected columns: ${required.join(", ")}`);
   134→   134→      }
   135→   135→    }
   136→   136→
   137→   137→    const invoices: ParsedInvoice[] = [];
   138→   138→    for (let i = 1; i < lines.length; i++) {
   139→   139→      const cols = parseCsvLine(lines[i]);
   140→   140→      if (cols.length < headers.length) {
   141→   141→        throw new Error(`Row ${i + 1} has ${cols.length} columns, expected ${headers.length}.`);
   142→   142→      }
   143→   143→
   144→   144→      const row: Record<string, string> = {};
   145→   145→      headers.forEach((h, idx) => { row[h] = cols[idx]?.trim() || ""; });
   146→   146→
   147→   147→      // Convert amounts from rupees to paise (1 INR = 100 paise)
   148→   148→      const subtotal = Math.round(parseFloat(row.subtotal) * 100);
   149→   149→      const tax = Math.round(parseFloat(row.tax) * 100);
   150→   150→
   151→   151→      if (isNaN(subtotal) || isNaN(tax)) {
   152→   152→        throw new Error(`Row ${i + 1}: subtotal and tax must be numbers.`);
   153→   153→      }
   154→   154→
   155→   155→      invoices.push({
   156→   156→        customerName: row.customer_name,
   157→   157→        customerEmail: row.customer_email,
   158→   158→        invoiceNumber: row.invoice_number,
   159→   159→        issueDate: row.issue_date,
   160→   160→        dueDate: row.due_date,
   161→   161→        subtotal,
   162→   162→        tax,
   163→   163→      });
   164→   164→    }
   165→   165→
   166→   166→    return invoices;
   167→   167→  }
   168→   168→
   169→   169→  function parseCsvLine(line: string): string[] {
   170→   170→    const result: string[] = [];
   171→   171→    let current = "";
   172→   172→    let inQuotes = false;
   173→   173→    for (let i = 0; i < line.length; i++) {
   174→   174→      const char = line[i];
   175→   175→      if (char === '"') {
   176→   176→        inQuotes = !inQuotes;
   177→   177→      } else if (char === "," && !inQuotes) {
   178→   178→        result.push(current);
   179→   179→        current = "";
   180→   180→      } else {
   181→   181→        current += char;
   182→   182→      }
   183→   183→    }
   184→   184→    result.push(current);
   185→   185→    return result;
   186→   186→  }
