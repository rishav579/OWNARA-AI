    36→export interface Employee {
    37→  id: string;
    38→  name: string;
    39→  role: RoleKey;
    40→  roleName: string;
    41→  templateId: string;
    42→  status: EmployeeStatus;
    43→  state: EmployeeState;
    44→  avatarColor: string;
    45→  createdAt: string;
    46→  activatedAt: string | null;
    47→  taskCount: number;
    48→  completedTasks: number;
    49→  tokenUsage: number;
    50→  tokenCap: number;
    51→  tools: string[];
    52→  pendingApprovals: number;
    53→  jobDescription: string;
    54→  boundaries: string[];
    55→  approvalRules: Record<string, "critical" | "non_critical">;
    56→}
    57→
    58→export interface TaskStep {
    59→  stepNumber: number;
    60→  stepType: "plan" | "reasoning" | "tool_call" | "approval_gate";
    61→  reasoning: string;
    62→  tool?: string;
    63→  output?: string;
    64→  status: "completed" | "running" | "pending" | "failed";
    65→  tokens: number;
    66→  durationMs: number;
    67→}
    68→
    69→export interface Task {
    70→  id: string;
    71→  title: string;
    72→  description: string;
    73→  employeeId: string;
    74→  employeeName: string;
    75→  status: TaskStatus;
    76→  priority: "low" | "medium" | "high";
    77→  stepCount: number;
    78→  stepCap: number;
    79→  tokenUsage: number;
    80→  tokenCap: number;
    81→  assignedBy: string;
    82→  startedAt: string;
    83→  completedAt: string | null;
    84→  steps?: TaskStep[];
    85→}
    86→
    87→export interface Approval {
    88→  id: string;
    89→  taskId: string;
    90→  taskTitle: string;
    91→  employeeId: string;
    92→  employeeName: string;
    93→  tool: string;
    94→  toolDisplayName: string;
    95→  proposedAction: Record<string, string>;
    96→  status: ApprovalStatus;
    97→  createdAt: string;
    98→  timeoutAt: string;
    99→  decidedBy?: string;
   100→  decidedAt?: string;
   101→  decision?: ApprovalDecision;
   102→  reason?: string;
   103→  criticality: "critical" | "non_critical";
   104→}

   107→  id: string;
   108→  sequenceNumber: number;
   109→  entryType: string;
   110→  actorType: "user" | "employee" | "system";
   111→  actorName: string;
   112→  targetType: string;
   113→  targetId: string;
   114→  payload: Record<string, string>;
   115→  previousHash: string;
   116→  entryHash: string;
   117→  createdAt: string;
   118→}

   123→  contentType: string;
   124→  sizeBytes: number;
   125→  status: DocumentStatus;
   126→  chunkCount: number;

   128→  employeeName: string | null;
   129→  uploadedBy: string;
   130→  createdAt: string;
   131→}
